10//! # Cedarling
11//! The Cedarling is a performant local authorization service that runs the Rust Cedar Engine.
12//! Cedar policies and schema are loaded at startup from a locally cached "Policy Store".
13//! In simple terms, the Cedarling returns the answer: should the application allow this action on this resource given these JWT tokens.
14//! "Fit for purpose" policies help developers build a better user experience.
15//! For example, why display form fields that a user is not authorized to see?
16//! The Cedarling is a more productive and flexible way to handle authorization.

109/// Errors that can occur during initialization Cedarling.
110#[derive(Debug, thiserror::Error)]
111pub enum InitCedarlingError {
112    /// Error while preparing config for internal services
113    #[error(transparent)]
114    ServiceConfig(#[from] ServiceConfigError),
115    /// Error while initializing a Service
116    #[error(transparent)]
117    ServiceInit(#[from] ServiceInitError),
118    /// Error while parse [`BootstrapConfigRaw`]
119    #[error(transparent)]
120    BootstrapConfigLoading(#[from] BootstrapConfigLoadingError),
121    /// Error while initializing the `DataStore` (invalid configuration)
122    #[error(transparent)]
123    DataStoreInit(#[from] ConfigValidationError),
124    #[cfg(feature = "blocking")]
125    /// Error while init tokio runtime
126    #[error(transparent)]
127    RuntimeInit(std::io::Error),
128    /// Error returned when Cedarling fails to obtain client credentials for sending
129    /// logs to the Lock Server.
130    #[error("failed to initialize the Lock Service: {0}")]
131    InitLockService(#[from] InitLockServiceError),
132}

138/// The instance of the Cedarling application.
139/// It is safe to share between threads.
140#[derive(Clone)]
141pub struct Cedarling {
142    log: log::Logger,
143    /// Wrapped in [`ArcSwap`] so the policy-store refresh worker can publish a
144    /// freshly built [`Authz`] (with new policy store, rebuilt JWT service and
145    /// entity builder) atomically. Every public method snapshots via
146    /// [`ArcSwap::load`] so an in-flight authorization keeps using the
147    /// pre-swap instance.
148    authz: Arc<arc_swap::ArcSwap<Authz>>,
149    /// Optional custom token processor, held on `Cedarling` (never swapped) so it
150    /// survives policy-store refreshes that rebuild `authz`. Registered live via
151    /// [`Self::set_custom_token_processor`].
152    custom_token_processor: Arc<arc_swap::ArcSwapOption<CustomTokenProcessorHolder>>,
153    data: Arc<DataStore>,
154    /// Held purely for its `Drop` side effect: dropping the last `Arc` closes
155    /// the worker's `oneshot` shutdown channel so the background refresh loop
156    /// exits when [`Cedarling`] goes away. The leading `_` tells the compiler
157    /// the field is intentionally not read.
158    _refresh_handle: Option<Arc<PolicyStoreRefreshHandle>>,
159}

173    /// Create a new instance of the Cedarling application.
174    pub async fn new(config: &BootstrapConfig) -> Result<Cedarling, InitCedarlingError> {
175        let pdp_id = app_types::PdpID::new();
176        let app_name = (!config.application_name.is_empty())
177            .then(|| ApplicationName::from(config.application_name.clone()));
178
179        let metrics = Arc::new(
180            if config
181                .lock_config
182                .as_ref()
183                .is_some_and(|c| c.telemetry_interval.is_some())
184            {
185                MetricsCollector::new(0)
186            } else {
187                MetricsCollector::disabled()
188            },
189        );
190
191        let log = crate::log::init_logger(
192            &config.log_config,
193            pdp_id,
194            app_name,
195            config.lock_config.as_ref(),
196            metrics.clone(),
197            config.http_client_config,
198        )
199        .await?;
200
201        log.log_any(
202            LogEntry::new(BaseLogEntry::new_system_opt_request_id(
203                LogLevel::INFO,
204                None,
205            ))
206            .set_message("Cedarling initialization started".to_string())
207            .set_build_info(BUILD_COMMIT, BUILD_TIMESTAMP),
208        );
209
210        // Bootstrap-load: build HttpClient + load policy store. Returns the
211        // service config plus the refresh-worker seed in one shot so the
212        // seed values (initial body hash, initial cache validators) stay in
213        // lexical scope right next to the refresh-worker spawn below.
214        let (service_config, refresh_seed) = perform_bootstrap_load(config, &log).await?;
215
216        let policy_count = service_config
217            .policy_store
218            .policies
219            .get_set()
220            .num_of_policies();
221        metrics.set_policy_count(policy_count);
222
223        // Initialize data store first so it can be passed to authz service
224        let data = Arc::new(DataStore::new(
225            config.data_store_config.clone(),
226            metrics.clone(),
227        )?);
228
229        let mut service_factory = ServiceFactory::new(
230            config,
231            service_config,
232            log.clone(),
233            data.clone(),
234            metrics.clone(),
235        );
236
237        // Log policy store metadata if available (new format only)
238        if let Some(metadata) = service_factory.policy_store_metadata() {
239            log_policy_store_metadata(&log, metadata);
240        }
241
242        if let Some(registry) = log.health_registry() {
243            registry.register("core", || HealthStatus::Success);
244            registry.register("policy_load", move || {
245                if policy_count > 0 {
246                    HealthStatus::Success
247                } else {
248                    HealthStatus::Failure
249                }
250            });
251        }
252
253        let authz = service_factory.authz_service().await?;
254        let authz_swap = Arc::new(arc_swap::ArcSwap::from(authz));
255
256        let refresh_handle = maybe_spawn_refresh_worker(
257            config,
258            &service_factory,
259            authz_swap.clone(),
260            log.clone(),
261            data.clone(),
262            metrics.clone(),
263            refresh_seed,
264        );
265
266        Ok(Cedarling {
267            log,
268            authz: authz_swap,
269            custom_token_processor: Arc::new(arc_swap::ArcSwapOption::const_empty()),
270            data,
271            _refresh_handle: refresh_handle,
272        })
273    }

447            .clear_custom_token_cache();
448    }

484/// Build the HTTP client and load the policy store from the bootstrap config.
485/// Returns the parts the rest of `Cedarling::new` needs: the [`ServiceConfig`]
486/// for service-factory construction, and the [`RefreshWorkerSeed`] for the
487/// refresh worker's first-tick short-circuit. Wraps both fallible steps so
488/// the existing "configuration parsed successfully" / "...with error" log
489/// behavior covers the whole bootstrap unit.
490async fn perform_bootstrap_load(
491    config: &BootstrapConfig,
492    log: &log::Logger,
493) -> Result<(ServiceConfig, RefreshWorkerSeed), ServiceConfigError> {
494    let raw_load: Result<(http::HttpClient, LoadedPolicyStore), ServiceConfigError> = async {
495        let http_client = http::HttpClient::new(config.http_client_config)?;
496        let loaded = load_policy_store(
497            &config.policy_store_config,
498            &http_client,
499            config.authorization_config.strict_schema_validation,
500        )
501        .await?;
502        Ok((http_client, loaded))
503    }
504    .await;
505
506    let (http_client, loaded) = raw_load
507        .inspect(|_| {
508            log.log_any(
509                LogEntry::new(BaseLogEntry::new_system_opt_request_id(
510                    LogLevel::DEBUG,
511                    None,
512                ))
513                .set_message("configuration parsed successfully".to_string()),
514            );
515        })
516        .inspect_err(|err| {
517            log.log_any(
518                LogEntry::new(BaseLogEntry::new_system_opt_request_id(
519                    LogLevel::ERROR,
520                    None,
521                ))
522                .set_error(err.to_string())
523                .set_message("configuration parsed with error".to_string()),
524            );
525        })?;
526
527    let LoadedPolicyStore {
528        store: policy_store,
529        body_hash,
530        validators,
531    } = loaded;
532    Ok((
533        ServiceConfig {
534            policy_store,
535            http_client,
536        },
537        RefreshWorkerSeed {
538            initial_body_hash: body_hash,
539            initial_validators: validators,
540        },
541    ))
542}

544/// Spawn the background policy-store refresh worker if the source is a remote
545/// URL and a non-zero refresh interval was configured. Returns `None` for
546/// local sources or when refresh is disabled. The `seed` carries the
547/// `body_hash` and `validators` captured during initial bootstrap so the
548/// first periodic tick can short-circuit — passed in directly from the
549/// bootstrap-load result rather than detoured through `ServiceFactory`.
550fn maybe_spawn_refresh_worker(
551    config: &BootstrapConfig,
552    service_factory: &ServiceFactory<'_>,
553    authz_swap: Arc<arc_swap::ArcSwap<authz::Authz>>,
554    log: log::Logger,
555    data: Arc<context_data_api::DataStore>,
556    metrics: Arc<authz::metrics::MetricsCollector>,
557    seed: RefreshWorkerSeed,
558) -> Option<Arc<PolicyStoreRefreshHandle>> {
559    if !config.policy_store_config.refresh_enabled() {
560        return None;
561    }
562    let source = RefreshSource::from_policy_store_source(&config.policy_store_config.source)?;
563    let (interval_secs, clamped) = config.policy_store_config.effective_refresh_interval();
564    if clamped {
565        log.log_any(
566            LogEntry::new(BaseLogEntry::new_system_opt_request_id(LogLevel::WARN, None))
567                .set_message(format!(
568                    "CEDARLING_POLICY_STORE_REFRESH_INTERVAL={} is below the minimum; clamped to {} seconds",
569                    config.policy_store_config.refresh_interval_secs,
570                    interval_secs,
571                )),
572        );
573    }
574    let rebuilder = AuthzRebuilder {
575        jwt_config: config.jwt_config.clone(),
576        authorization_config: config.authorization_config.clone(),
577        http_client: service_factory.http_client_for_refresh(),
578        log: log.clone(),
579        data_store: data,
580        metrics: metrics.clone(),
581    };
582    let ctx = WorkerContext {
583        source,
584        interval_secs,
585        http_client: service_factory.http_client_for_refresh(),
586        rebuilder,
587        authz_swap,
588        metrics,
589        log,
590        initial_body_hash: seed.initial_body_hash,
591        initial_validators: seed.initial_validators,
592        strict_schema_validation: config.authorization_config.strict_schema_validation,
593        archive_limits: config.policy_store_config.archive_limits(),
594    };
595    Some(Arc::new(spawn_refresh_worker(ctx)))
596}

598/// Log detailed information about the loaded policy store metadata, including
599/// ID, version, description, Cedar version, timestamps, and compatibility with
600/// the runtime Cedar version.
601fn log_policy_store_metadata(
602    log: &log::Logger,
603    metadata: &crate::common::policy_store::PolicyStoreMetadata,
604) {
605    // Build detailed log message using accessor methods
606    let mut details = format!(
607        "Policy store '{}' (ID: {}) v{} loaded",
608        metadata.name(),
609        if metadata.id().is_empty() {
610            "<auto>"
611        } else {
612            metadata.id()
613        },
614        metadata.version()
615    );
616
617    // Add description if available
618    if let Some(desc) = metadata.description() {
619        let _ = write!(details, " - {desc}");
620    }
621
622    // Add Cedar version info
623    let _ = write!(details, " [Cedar {}]", metadata.cedar_version());
624
625    // Add timestamp info if available
626    if let Some(created) = metadata.created_date() {
627        let _ = write!(details, " (created: {})", created.format("%Y-%m-%d"));
628    }
629    if let Some(updated) = metadata.updated_date() {
630        let _ = write!(details, " (updated: {})", updated.format("%Y-%m-%d"));
631    }
632
633    log.log_any(
634        LogEntry::new(BaseLogEntry::new_system_opt_request_id(
635            LogLevel::DEBUG,
636            None,
637        ))
638        .set_message(details),
639    );
640
641    // Log version compatibility check with current Cedar
642    let current_cedar_version: Version = cedar_policy::get_lang_version();
643    match metadata.is_compatible_with_cedar(&current_cedar_version) {
644        Ok(true) => {
645            log.log_any(
646                LogEntry::new(BaseLogEntry::new_system_opt_request_id(
647                    LogLevel::DEBUG,
648                    None,
649                ))
650                .set_message(format!(
651                    "Policy store Cedar version {} is compatible with runtime version {}",
652                    metadata.cedar_version(),
653                    current_cedar_version
654                )),
655            );
656        },
657        Ok(false) => {
658            log.log_any(
659                LogEntry::new(BaseLogEntry::new_system_opt_request_id(
660                    LogLevel::WARN,
661                    None,
662                ))
663                .set_message(format!(
664                    "Policy store Cedar version {} may not be compatible with runtime version {}",
665                    metadata.cedar_version(),
666                    current_cedar_version
667                )),
668            );
669        },
670        Err(e) => {
671            log.log_any(
672                LogEntry::new(BaseLogEntry::new_system_opt_request_id(
673                    LogLevel::WARN,
674                    None,
675                ))
676                .set_message(format!("Could not check Cedar version compatibility: {e}")),
677            );
678        },
679    }
680
681    // Log parsed version for debugging if available
682    if let Some(parsed_version) = metadata.version_parsed() {
683        log.log_any(
684            LogEntry::new(BaseLogEntry::new_system_opt_request_id(
685                LogLevel::TRACE,
686                None,
687            ))
688            .set_message(format!(
689                "Policy store semantic version: {}.{}.{}",
690                parsed_version.major, parsed_version.minor, parsed_version.patch
691            )),
692        );
693    }
694}

724// implements DataApi for Cedarling
725// Helper function to calculate capacity usage and check memory alert threshold
726fn calculate_capacity_usage(
727    entry_count: usize,
728    max_entries: usize,
729    memory_alert_threshold: f64,
730) -> (f64, bool) {
731    // Precision loss is acceptable for percentage calculation
732    #[allow(clippy::cast_precision_loss)]
733    let capacity_usage_percent = if max_entries > 0 {
734        (entry_count as f64 / max_entries as f64) * 100.0
735    } else {
736        0.0 // Unlimited capacity, no percentage
737    };
738    let memory_alert_triggered = capacity_usage_percent >= memory_alert_threshold;
739    (capacity_usage_percent, memory_alert_triggered)
740}

742// provides public interface for pushing and retrieving data
743impl DataApi for Cedarling {
744    fn push_data_ctx(
745        &self,
746        key: &str,
747        value: serde_json::Value,
748        ttl: Option<std::time::Duration>,
749    ) -> Result<(), DataError> {
750        self.data.push(key, value, ttl)?;
751
752        // Check memory usage and log warning if threshold is exceeded
753        let config = self.data.config();
754        if config.max_entries > 0 {
755            let entry_count = self.data.count();
756            let (capacity_usage_percent, memory_alert_triggered) = calculate_capacity_usage(
757                entry_count,
758                config.max_entries,
759                config.memory_alert_threshold,
760            );
761            if memory_alert_triggered {
762                let log_entry = LogEntry::new(BaseLogEntry::new_system_opt_request_id(
763                    LogLevel::WARN,
764                    None,
765                ))
766                .set_message(format!(
767                    "DataStore memory usage alert: {:.1}% capacity used ({}/{} entries), threshold: {:.1}%",
768                    capacity_usage_percent,
769                    entry_count,
770                    config.max_entries,
771                    config.memory_alert_threshold
772                ));
773                self.log.log_any(log_entry);
774            }
775        }
776
777        Ok(())
778    }
779
780    fn get_data_ctx(&self, key: &str) -> Result<Option<serde_json::Value>, DataError> {
781        Ok(self.data.get(key))
782    }
783
784    fn get_data_entry_ctx(&self, key: &str) -> Result<Option<DataEntry>, DataError> {
785        Ok(self.data.get_entry(key))
786    }
787
788    fn remove_data_ctx(&self, key: &str) -> Result<bool, DataError> {
789        Ok(self.data.remove(key))
790    }
791
792    fn clear_data_ctx(&self) -> Result<(), DataError> {
793        self.data.clear();
794        Ok(())
795    }
796
797    fn list_data_ctx(&self) -> Result<Vec<DataEntry>, DataError> {
798        Ok(self.data.list_entries())
799    }
800
801    fn get_stats_ctx(&self) -> Result<DataStoreStats, DataError> {
802        let config = self.data.config();
803        let entry_count = self.data.count();
804        let total_size_bytes = self.data.total_size();
805        let avg_entry_size_bytes = total_size_bytes.checked_div(entry_count).unwrap_or(0);
806
807        // Calculate capacity usage percentage
808        let (capacity_usage_percent, memory_alert_triggered) = calculate_capacity_usage(
809            entry_count,
810            config.max_entries,
811            config.memory_alert_threshold,
812        );
813
814        Ok(DataStoreStats {
815            entry_count,
816            max_entries: config.max_entries,
817            max_entry_size: config.max_entry_size,
818            metrics_enabled: config.enable_metrics,
819            total_size_bytes,
820            avg_entry_size_bytes,
821            capacity_usage_percent,
822            memory_alert_threshold: config.memory_alert_threshold,
823            memory_alert_triggered,
824        })
825    }

828// Tooling & Diagnostics
829#[cfg(feature = "tools")]
830impl Cedarling {
831    /// Return metadata for every policy in the store.
832    ///
833    /// Intended for tooling that needs to enumerate the full policy set
834    /// (coverage reports, dashboards). Ordering is unspecified.
835    #[must_use]
836    pub fn all_policy_metadata(&self) -> Vec<PolicyMetadata> {
837        self.authz.load().all_policy_metadata()
838    }
839
840    /// Run parse / schema / metadata validation against a policy-store source
841    /// without initializing the authorization engine. Never returns Err for
842    /// validation failures — those land in the returned report; only
843    /// returns Err for underlying I/O or network failures.
844    #[allow(clippy::too_many_lines)]
845    pub async fn validate_policy_store(
846        config: &PolicyStoreConfig,
847        http_config: &crate::http::HttpClientConfig,
848    ) -> Result<ValidationReport, ValidateInfraError> {
849        // 1. Build a fresh HttpClient
850        let http_client = crate::http::HttpClient::new(*http_config)?;
851
852        // 2. Call load_policy_store
853        let load_result =
854            crate::init::policy_store::load_policy_store(config, &http_client, false).await;
855
856        match load_result {
857            Err(e) => {
858                use crate::init::policy_store::PolicyStoreLoadError;
859
860                let err_str = e.to_string();
861
862                let is_metadata = matches!(&e, PolicyStoreLoadError::Validation(_));
863
864                let is_parse = matches!(
865                    &e,
866                    PolicyStoreLoadError::LegacyJsonNotSupported
867                        | PolicyStoreLoadError::ParseYaml(_)
868                        | PolicyStoreLoadError::Conversion(_)
869                        | PolicyStoreLoadError::InvalidStore(_)
870                );
871
872                match e {
873                    PolicyStoreLoadError::FetchFromLockServer(_)
874                    | PolicyStoreLoadError::Archive(_)
875                    | PolicyStoreLoadError::Directory(_) => {
876                        Err(ValidateInfraError::Io(std::io::Error::other(err_str)))
877                    },
878                    PolicyStoreLoadError::ParseFile(_, io_err) => {
879                        Err(ValidateInfraError::Io(io_err))
880                    },
881                    _ if is_metadata => {
882                        let diag = Diagnostic {
883                            file: "<policy-store>".into(),
884                            line: None,
885                            column: None,
886                            message: err_str,
887                        };
888                        Ok(ValidationReport {
889                            parse: LevelResult::Skipped {
890                                reason: "metadata check failed".into(),
891                            },
892                            schema: LevelResult::Skipped {
893                                reason: "metadata check failed".into(),
894                            },
895                            metadata: LevelResult::Failed { errors: vec![diag] },
896                        })
897                    },
898                    _ if is_parse => {
899                        let diag = Diagnostic {
900                            file: "<policy-store>".into(),
901                            line: None,
902                            column: None,
903                            message: err_str,
904                        };
905                        Ok(ValidationReport {
906                            parse: LevelResult::Failed { errors: vec![diag] },
907                            schema: LevelResult::Skipped {
908                                reason: "parse failed".into(),
909                            },
910                            metadata: LevelResult::Skipped {
911                                reason: "parse failed".into(),
912                            },
913                        })
914                    },
915                    _ => Err(ValidateInfraError::Io(std::io::Error::other(err_str))),
916                }
917            },
918            Ok(loaded) => {
919                // Parse succeeded! Now run schema and metadata independently.
920
921                // Schema Level
922                let schema_res = if let Some(schema) = &loaded.store.store.schema {
923                    let validator = cedar_policy::Validator::new(schema.schema.clone());
924                    let result = validator.validate(
925                        loaded.store.store.policies.get_set(),
926                        cedar_policy::ValidationMode::Strict,
927                    );
928                    if result.validation_passed() {
929                        LevelResult::Ok
930                    } else {
931                        let errors = result
932                            .validation_errors()
933                            .map(|e| Diagnostic {
934                                file: e.policy_id().to_string(),
935                                line: None,
936                                column: None,
937                                message: e.to_string(),
938                            })
939                            .collect();
940                        LevelResult::Failed { errors }
941                    }
942                } else {
943                    LevelResult::Skipped {
944                        reason: "no schema present".into(),
945                    }
946                };
947
948                // Metadata Level
949                // For directory/archive, the loader already ran MetadataValidator.
950                // For legacy YAML/JSON stores, we run validate_legacy_metadata.
951                let metadata_res = match &loaded.store.metadata {
952                    Some(metadata) => {
953                        use crate::common::policy_store::validator::MetadataValidator;
954                        match MetadataValidator::validate(metadata) {
955                            Ok(()) => LevelResult::Ok,
956                            Err(e) => LevelResult::Failed {
957                                errors: vec![Diagnostic {
958                                    file: "<metadata>".into(),
959                                    line: None,
960                                    column: None,
961                                    message: e.to_string(),
962                                }],
963                            },
964                        }
965                    },
966                    None => {
967                        match crate::common::policy_store::validator::validate_legacy_metadata(
968                            &loaded.store.store,
969                        ) {
970                            Ok(()) => LevelResult::Ok,
971                            Err(e) => LevelResult::Failed {
972                                errors: vec![Diagnostic {
973                                    file: "<inline>".into(),
974                                    line: None,
975                                    column: None,
976                                    message: e.to_string(),
977                                }],
978                            },
979                        }
980                    },
981                };
982
983                Ok(ValidationReport {
984                    parse: LevelResult::Ok,
985                    schema: schema_res,
986                    metadata: metadata_res,
987                })
988            },
989        }
990    }
