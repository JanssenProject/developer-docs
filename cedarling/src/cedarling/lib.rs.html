10//! # Cedarling
11//! The Cedarling is a performant local authorization service that runs the Rust Cedar Engine.
12//! Cedar policies and schema are loaded at startup from a locally cached "Policy Store".
13//! In simple terms, the Cedarling returns the answer: should the application allow this action on this resource given these JWT tokens.
14//! "Fit for purpose" policies help developers build a better user experience.
15//! For example, why display form fields that a user is not authorized to see?
16//! The Cedarling is a more productive and flexible way to handle authorization.

110/// Errors that can occur during initialization Cedarling.
111#[derive(Debug, thiserror::Error)]
112pub enum InitCedarlingError {
113    /// Error while preparing config for internal services
114    #[error(transparent)]
115    ServiceConfig(#[from] ServiceConfigError),
116    /// Error while initializing a Service
117    #[error(transparent)]
118    ServiceInit(#[from] ServiceInitError),
119    /// Error while parse [`BootstrapConfigRaw`]
120    #[error(transparent)]
121    BootstrapConfigLoading(#[from] BootstrapConfigLoadingError),
122    /// Error while initializing the `DataStore` (invalid configuration)
123    #[error(transparent)]
124    DataStoreInit(#[from] ConfigValidationError),
125    #[cfg(feature = "blocking")]
126    /// Error while init tokio runtime
127    #[error(transparent)]
128    RuntimeInit(std::io::Error),
129    /// Error returned when Cedarling fails to obtain client credentials for sending
130    /// logs to the Lock Server.
131    #[error("failed to initialize the Lock Service: {0}")]
132    InitLockService(#[from] InitLockServiceError),
133}

139/// The instance of the Cedarling application.
140/// It is safe to share between threads.
141#[derive(Clone)]
142pub struct Cedarling {
143    log: log::Logger,
144    /// Wrapped in [`ArcSwap`] so the policy-store refresh worker can publish a
145    /// freshly built [`Authz`] (with new policy store, rebuilt JWT service and
146    /// entity builder) atomically. Every public method snapshots via
147    /// [`ArcSwap::load`] so an in-flight authorization keeps using the
148    /// pre-swap instance.
149    authz: Arc<arc_swap::ArcSwap<Authz>>,
150    /// Optional custom token processor, held on `Cedarling` (never swapped) so it
151    /// survives policy-store refreshes that rebuild `authz`. Registered live via
152    /// [`Self::set_custom_token_processor`].
153    custom_token_processor: Arc<arc_swap::ArcSwapOption<CustomTokenProcessorHolder>>,
154    data: Arc<DataStore>,
155    /// Metrics collector shared with log, data store, authz and refresh worker.
156    /// Owned locally when [`MetricsMode::Local`]; otherwise held only so the
157    /// injected callers can record into it.
158    metrics: Arc<MetricsCollector>,
159    /// Held purely for its `Drop` side effect: dropping the last `Arc` closes
160    /// the worker's `oneshot` shutdown channel so the background refresh loop
161    /// exits when [`Cedarling`] goes away. The leading `_` tells the compiler
162    /// the field is intentionally not read.
163    _refresh_handle: Option<Arc<PolicyStoreRefreshHandle>>,
164}

178    /// Create a new instance of the Cedarling application.
179    pub async fn new(config: &BootstrapConfig) -> Result<Cedarling, InitCedarlingError> {
180        let pdp_id = app_types::PdpID::new();
181        let app_name = (!config.application_name.is_empty())
182            .then(|| ApplicationName::from(config.application_name.clone()));
183
184        let telemetry_active = config
185            .lock_config
186            .as_ref()
187            .is_some_and(|c| c.telemetry_interval.is_some());
188        let metrics_mode = resolve_metrics_mode(
189            telemetry_active,
190            config.authorization_config.metrics_collection,
191        );
192        let metrics = Arc::new(MetricsCollector::new(metrics_mode));
193
194        let log = crate::log::init_logger(
195            &config.log_config,
196            pdp_id,
197            app_name,
198            config.lock_config.as_ref(),
199            metrics.clone(),
200            config.http_client_config,
201        )
202        .await?;
203
204        log.log_any(
205            LogEntry::new(BaseLogEntry::new_system_opt_request_id(
206                LogLevel::INFO,
207                None,
208            ))
209            .set_message("Cedarling initialization started".to_string())
210            .set_build_info(BUILD_COMMIT, BUILD_TIMESTAMP),
211        );
212
213        // Bootstrap-load: build HttpClient + load policy store. Returns the
214        // service config plus the refresh-worker seed in one shot so the
215        // seed values (initial body hash, initial cache validators) stay in
216        // lexical scope right next to the refresh-worker spawn below.
217        let (service_config, refresh_seed) = perform_bootstrap_load(config, &log).await?;
218
219        let policy_count = service_config
220            .policy_store
221            .policies
222            .get_set()
223            .num_of_policies();
224        metrics.set_policy_count(policy_count);
225
226        // Initialize data store first so it can be passed to authz service
227        let data = Arc::new(DataStore::new(
228            config.data_store_config.clone(),
229            metrics.clone(),
230        )?);
231
232        let mut service_factory = ServiceFactory::new(
233            config,
234            service_config,
235            log.clone(),
236            data.clone(),
237            metrics.clone(),
238        );
239
240        // Log policy store metadata if available (new format only)
241        if let Some(metadata) = service_factory.policy_store_metadata() {
242            log_policy_store_metadata(&log, metadata);
243        }
244
245        if let Some(registry) = log.health_registry() {
246            registry.register("core", || HealthStatus::Success);
247            registry.register("policy_load", move || {
248                if policy_count > 0 {
249                    HealthStatus::Success
250                } else {
251                    HealthStatus::Failure
252                }
253            });
254        }
255
256        let authz = service_factory.authz_service().await?;
257        let authz_swap = Arc::new(arc_swap::ArcSwap::from(authz));
258
259        let refresh_handle = maybe_spawn_refresh_worker(
260            config,
261            &service_factory,
262            authz_swap.clone(),
263            log.clone(),
264            data.clone(),
265            metrics.clone(),
266            refresh_seed,
267        );
268
269        Ok(Cedarling {
270            log,
271            authz: authz_swap,
272            custom_token_processor: Arc::new(arc_swap::ArcSwapOption::const_empty()),
273            data,
274            metrics,
275            _refresh_handle: refresh_handle,
276        })
277    }

482            .clear_custom_token_cache();
483    }

519/// Build the HTTP client and load the policy store from the bootstrap config.
520/// Returns the parts the rest of `Cedarling::new` needs: the [`ServiceConfig`]
521/// for service-factory construction, and the [`RefreshWorkerSeed`] for the
522/// refresh worker's first-tick short-circuit. Wraps both fallible steps so
523/// the existing "configuration parsed successfully" / "...with error" log
524/// behavior covers the whole bootstrap unit.
525async fn perform_bootstrap_load(
526    config: &BootstrapConfig,
527    log: &log::Logger,
528) -> Result<(ServiceConfig, RefreshWorkerSeed), ServiceConfigError> {
529    let raw_load: Result<(http::HttpClient, LoadedPolicyStore), ServiceConfigError> = async {
530        let http_client = http::HttpClient::new(config.http_client_config)?;
531        let loaded = load_policy_store(
532            &config.policy_store_config,
533            &http_client,
534            config.authorization_config.strict_schema_validation,
535        )
536        .await?;
537        Ok((http_client, loaded))
538    }
539    .await;
540
541    let (http_client, loaded) = raw_load
542        .inspect(|_| {
543            log.log_any(
544                LogEntry::new(BaseLogEntry::new_system_opt_request_id(
545                    LogLevel::DEBUG,
546                    None,
547                ))
548                .set_message("configuration parsed successfully".to_string()),
549            );
550        })
551        .inspect_err(|err| {
552            log.log_any(
553                LogEntry::new(BaseLogEntry::new_system_opt_request_id(
554                    LogLevel::ERROR,
555                    None,
556                ))
557                .set_error(err.to_string())
558                .set_message("configuration parsed with error".to_string()),
559            );
560        })?;
561
562    let LoadedPolicyStore {
563        store: policy_store,
564        body_hash,
565        validators,
566    } = loaded;
567    Ok((
568        ServiceConfig {
569            policy_store,
570            http_client,
571        },
572        RefreshWorkerSeed {
573            initial_body_hash: body_hash,
574            initial_validators: validators,
575        },
576    ))
577}

579/// Spawn the background policy-store refresh worker if the source is a remote
580/// URL and a non-zero refresh interval was configured. Returns `None` for
581/// local sources or when refresh is disabled. The `seed` carries the
582/// `body_hash` and `validators` captured during initial bootstrap so the
583/// first periodic tick can short-circuit — passed in directly from the
584/// bootstrap-load result rather than detoured through `ServiceFactory`.
585fn maybe_spawn_refresh_worker(
586    config: &BootstrapConfig,
587    service_factory: &ServiceFactory<'_>,
588    authz_swap: Arc<arc_swap::ArcSwap<authz::Authz>>,
589    log: log::Logger,
590    data: Arc<context_data_api::DataStore>,
591    metrics: Arc<authz::metrics::MetricsCollector>,
592    seed: RefreshWorkerSeed,
593) -> Option<Arc<PolicyStoreRefreshHandle>> {
594    if !config.policy_store_config.refresh_enabled() {
595        return None;
596    }
597    let source = RefreshSource::from_policy_store_source(&config.policy_store_config.source)?;
598    let (interval_secs, clamped) = config.policy_store_config.effective_refresh_interval();
599    if clamped {
600        log.log_any(
601            LogEntry::new(BaseLogEntry::new_system_opt_request_id(LogLevel::WARN, None))
602                .set_message(format!(
603                    "CEDARLING_POLICY_STORE_REFRESH_INTERVAL={} is below the minimum; clamped to {} seconds",
604                    config.policy_store_config.refresh_interval_secs,
605                    interval_secs,
606                )),
607        );
608    }
609    let rebuilder = AuthzRebuilder {
610        jwt_config: config.jwt_config.clone(),
611        authorization_config: config.authorization_config.clone(),
612        http_client: service_factory.http_client_for_refresh(),
613        log: log.clone(),
614        data_store: data,
615        metrics: metrics.clone(),
616    };
617    let ctx = WorkerContext {
618        source,
619        interval_secs,
620        http_client: service_factory.http_client_for_refresh(),
621        rebuilder,
622        authz_swap,
623        metrics,
624        log,
625        initial_body_hash: seed.initial_body_hash,
626        initial_validators: seed.initial_validators,
627        strict_schema_validation: config.authorization_config.strict_schema_validation,
628        archive_limits: config.policy_store_config.archive_limits(),
629    };
630    Some(Arc::new(spawn_refresh_worker(ctx)))
631}

633/// Log detailed information about the loaded policy store metadata, including
634/// ID, version, description, Cedar version, timestamps, and compatibility with
635/// the runtime Cedar version.
636fn log_policy_store_metadata(
637    log: &log::Logger,
638    metadata: &crate::common::policy_store::PolicyStoreMetadata,
639) {
640    // Build detailed log message using accessor methods
641    let mut details = format!(
642        "Policy store '{}' (ID: {}) v{} loaded",
643        metadata.name(),
644        if metadata.id().is_empty() {
645            "<auto>"
646        } else {
647            metadata.id()
648        },
649        metadata.version()
650    );
651
652    // Add description if available
653    if let Some(desc) = metadata.description() {
654        let _ = write!(details, " - {desc}");
655    }
656
657    // Add Cedar version info
658    let _ = write!(details, " [Cedar {}]", metadata.cedar_version());
659
660    // Add timestamp info if available
661    if let Some(created) = metadata.created_date() {
662        let _ = write!(details, " (created: {})", created.format("%Y-%m-%d"));
663    }
664    if let Some(updated) = metadata.updated_date() {
665        let _ = write!(details, " (updated: {})", updated.format("%Y-%m-%d"));
666    }
667
668    log.log_any(
669        LogEntry::new(BaseLogEntry::new_system_opt_request_id(
670            LogLevel::DEBUG,
671            None,
672        ))
673        .set_message(details),
674    );
675
676    // Log version compatibility check with current Cedar
677    let current_cedar_version: Version = cedar_policy::get_lang_version();
678    match metadata.is_compatible_with_cedar(&current_cedar_version) {
679        Ok(true) => {
680            log.log_any(
681                LogEntry::new(BaseLogEntry::new_system_opt_request_id(
682                    LogLevel::DEBUG,
683                    None,
684                ))
685                .set_message(format!(
686                    "Policy store Cedar version {} is compatible with runtime version {}",
687                    metadata.cedar_version(),
688                    current_cedar_version
689                )),
690            );
691        },
692        Ok(false) => {
693            log.log_any(
694                LogEntry::new(BaseLogEntry::new_system_opt_request_id(
695                    LogLevel::WARN,
696                    None,
697                ))
698                .set_message(format!(
699                    "Policy store Cedar version {} may not be compatible with runtime version {}",
700                    metadata.cedar_version(),
701                    current_cedar_version
702                )),
703            );
704        },
705        Err(e) => {
706            log.log_any(
707                LogEntry::new(BaseLogEntry::new_system_opt_request_id(
708                    LogLevel::WARN,
709                    None,
710                ))
711                .set_message(format!("Could not check Cedar version compatibility: {e}")),
712            );
713        },
714    }
715
716    // Log parsed version for debugging if available
717    if let Some(parsed_version) = metadata.version_parsed() {
718        log.log_any(
719            LogEntry::new(BaseLogEntry::new_system_opt_request_id(
720                LogLevel::TRACE,
721                None,
722            ))
723            .set_message(format!(
724                "Policy store semantic version: {}.{}.{}",
725                parsed_version.major, parsed_version.minor, parsed_version.patch
726            )),
727        );
728    }
729}

759// implements DataApi for Cedarling
760// Helper function to calculate capacity usage and check memory alert threshold
761fn calculate_capacity_usage(
762    entry_count: usize,
763    max_entries: usize,
764    memory_alert_threshold: f64,
765) -> (f64, bool) {
766    // Precision loss is acceptable for percentage calculation
767    #[allow(clippy::cast_precision_loss)]
768    let capacity_usage_percent = if max_entries > 0 {
769        (entry_count as f64 / max_entries as f64) * 100.0
770    } else {
771        0.0 // Unlimited capacity, no percentage
772    };
773    let memory_alert_triggered = capacity_usage_percent >= memory_alert_threshold;
774    (capacity_usage_percent, memory_alert_triggered)
775}

777// provides public interface for pushing and retrieving data
778impl DataApi for Cedarling {
779    fn push_data_ctx(
780        &self,
781        key: &str,
782        value: serde_json::Value,
783        ttl: Option<std::time::Duration>,
784    ) -> Result<(), DataError> {
785        self.data.push(key, value, ttl)?;
786
787        // Check memory usage and log warning if threshold is exceeded
788        let config = self.data.config();
789        if config.max_entries > 0 {
790            let entry_count = self.data.count();
791            let (capacity_usage_percent, memory_alert_triggered) = calculate_capacity_usage(
792                entry_count,
793                config.max_entries,
794                config.memory_alert_threshold,
795            );
796            if memory_alert_triggered {
797                let log_entry = LogEntry::new(BaseLogEntry::new_system_opt_request_id(
798                    LogLevel::WARN,
799                    None,
800                ))
801                .set_message(format!(
802                    "DataStore memory usage alert: {:.1}% capacity used ({}/{} entries), threshold: {:.1}%",
803                    capacity_usage_percent,
804                    entry_count,
805                    config.max_entries,
806                    config.memory_alert_threshold
807                ));
808                self.log.log_any(log_entry);
809            }
810        }
811
812        Ok(())
813    }
814
815    fn get_data_ctx(&self, key: &str) -> Result<Option<serde_json::Value>, DataError> {
816        Ok(self.data.get(key))
817    }
818
819    fn get_data_entry_ctx(&self, key: &str) -> Result<Option<DataEntry>, DataError> {
820        Ok(self.data.get_entry(key))
821    }
822
823    fn remove_data_ctx(&self, key: &str) -> Result<bool, DataError> {
824        Ok(self.data.remove(key))
825    }
826
827    fn clear_data_ctx(&self) -> Result<(), DataError> {
828        self.data.clear();
829        Ok(())
830    }
831
832    fn list_data_ctx(&self) -> Result<Vec<DataEntry>, DataError> {
833        Ok(self.data.list_entries())
834    }
835
836    fn get_stats_ctx(&self) -> Result<DataStoreStats, DataError> {
837        let config = self.data.config();
838        let entry_count = self.data.count();
839        let total_size_bytes = self.data.total_size();
840        let avg_entry_size_bytes = total_size_bytes.checked_div(entry_count).unwrap_or(0);
841
842        // Calculate capacity usage percentage
843        let (capacity_usage_percent, memory_alert_triggered) = calculate_capacity_usage(
844            entry_count,
845            config.max_entries,
846            config.memory_alert_threshold,
847        );
848
849        Ok(DataStoreStats {
850            entry_count,
851            max_entries: config.max_entries,
852            max_entry_size: config.max_entry_size,
853            metrics_enabled: config.enable_metrics,
854            total_size_bytes,
855            avg_entry_size_bytes,
856            capacity_usage_percent,
857            memory_alert_threshold: config.memory_alert_threshold,
858            memory_alert_triggered,
859        })
860    }

863// Tooling & Diagnostics
864#[cfg(feature = "tools")]
865impl Cedarling {
866    /// Return metadata for every policy in the store.
867    ///
868    /// Intended for tooling that needs to enumerate the full policy set
869    /// (coverage reports, dashboards). Ordering is unspecified.
870    #[must_use]
871    pub fn all_policy_metadata(&self) -> Vec<PolicyMetadata> {
872        self.authz.load().all_policy_metadata()
873    }
874
875    /// Run parse / schema / metadata validation against a policy-store source
876    /// without initializing the authorization engine. Never returns Err for
877    /// validation failures — those land in the returned report; only
878    /// returns Err for underlying I/O or network failures.
879    #[allow(clippy::too_many_lines)]
880    pub async fn validate_policy_store(
881        config: &PolicyStoreConfig,
882        http_config: &crate::http::HttpClientConfig,
883    ) -> Result<ValidationReport, ValidateInfraError> {
884        // 1. Build a fresh HttpClient
885        let http_client = crate::http::HttpClient::new(*http_config)?;
886
887        // 2. Call load_policy_store
888        let load_result =
889            crate::init::policy_store::load_policy_store(config, &http_client, false).await;
890
891        match load_result {
892            Err(e) => {
893                use crate::init::policy_store::PolicyStoreLoadError;
894
895                let err_str = e.to_string();
896
897                let is_metadata = matches!(&e, PolicyStoreLoadError::Validation(_));
898
899                let is_parse = matches!(
900                    &e,
901                    PolicyStoreLoadError::LegacyJsonNotSupported
902                        | PolicyStoreLoadError::ParseYaml(_)
903                        | PolicyStoreLoadError::Conversion(_)
904                        | PolicyStoreLoadError::InvalidStore(_)
905                );
906
907                match e {
908                    PolicyStoreLoadError::FetchFromLockServer(_)
909                    | PolicyStoreLoadError::Archive(_)
910                    | PolicyStoreLoadError::Directory(_) => {
911                        Err(ValidateInfraError::Io(std::io::Error::other(err_str)))
912                    },
913                    PolicyStoreLoadError::ParseFile(_, io_err) => {
914                        Err(ValidateInfraError::Io(io_err))
915                    },
916                    _ if is_metadata => {
917                        let diag = Diagnostic {
918                            file: "<policy-store>".into(),
919                            line: None,
920                            column: None,
921                            message: err_str,
922                        };
923                        Ok(ValidationReport {
924                            parse: LevelResult::Skipped {
925                                reason: "metadata check failed".into(),
926                            },
927                            schema: LevelResult::Skipped {
928                                reason: "metadata check failed".into(),
929                            },
930                            metadata: LevelResult::Failed { errors: vec![diag] },
931                        })
932                    },
933                    _ if is_parse => {
934                        let diag = Diagnostic {
935                            file: "<policy-store>".into(),
936                            line: None,
937                            column: None,
938                            message: err_str,
939                        };
940                        Ok(ValidationReport {
941                            parse: LevelResult::Failed { errors: vec![diag] },
942                            schema: LevelResult::Skipped {
943                                reason: "parse failed".into(),
944                            },
945                            metadata: LevelResult::Skipped {
946                                reason: "parse failed".into(),
947                            },
948                        })
949                    },
950                    _ => Err(ValidateInfraError::Io(std::io::Error::other(err_str))),
951                }
952            },
953            Ok(loaded) => {
954                // Parse succeeded! Now run schema and metadata independently.
955
956                // Schema Level
957                let schema_res = if let Some(schema) = &loaded.store.store.schema {
958                    let validator = cedar_policy::Validator::new(schema.schema.clone());
959                    let result = validator.validate(
960                        loaded.store.store.policies.get_set(),
961                        cedar_policy::ValidationMode::Strict,
962                    );
963                    if result.validation_passed() {
964                        LevelResult::Ok
965                    } else {
966                        let errors = result
967                            .validation_errors()
968                            .map(|e| Diagnostic {
969                                file: e.policy_id().to_string(),
970                                line: None,
971                                column: None,
972                                message: e.to_string(),
973                            })
974                            .collect();
975                        LevelResult::Failed { errors }
976                    }
977                } else {
978                    LevelResult::Skipped {
979                        reason: "no schema present".into(),
980                    }
981                };
982
983                // Metadata Level
984                // Legacy YAML sources synthesize `metadata: Some(..)` with the
985                // user-chosen `policy_stores` key as `id`, so matching on
986                // `Option` cannot distinguish strict vs legacy. Branch on
987                // `config.source` instead; every other source keeps strict
988                // `MetadataValidator` checks.
989                let metadata_res = if matches!(
990                    &config.source,
991                    PolicyStoreSource::Yaml(_) | PolicyStoreSource::FileYaml(_)
992                ) {
993                    match crate::common::policy_store::validator::validate_legacy_metadata(
994                        &loaded.store.store,
995                    ) {
996                        Ok(()) => LevelResult::Ok,
997                        Err(e) => LevelResult::Failed {
998                            errors: vec![Diagnostic {
999                                file: "<inline>".into(),
1000                                line: None,
1001                                column: None,
1002                                message: e.to_string(),
1003                            }],
1004                        },
1005                    }
1006                } else {
1007                    match &loaded.store.metadata {
1008                        Some(metadata) => {
1009                            use crate::common::policy_store::validator::MetadataValidator;
1010                            match MetadataValidator::validate(metadata) {
1011                                Ok(()) => LevelResult::Ok,
1012                                Err(e) => LevelResult::Failed {
1013                                    errors: vec![Diagnostic {
1014                                        file: "<metadata>".into(),
1015                                        line: None,
1016                                        column: None,
1017                                        message: e.to_string(),
1018                                    }],
1019                                },
1020                            }
1021                        },
1022                        None => LevelResult::Skipped {
1023                            reason: "no metadata present".into(),
1024                        },
1025                    }
1026                };
1027
1028                Ok(ValidationReport {
1029                    parse: LevelResult::Ok,
1030                    schema: schema_res,
1031                    metadata: metadata_res,
1032                })
1033            },
1034        }
1035    }
