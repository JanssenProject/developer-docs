38#[derive(Error, Debug)]
39pub(crate) enum TrustedIssuerLoaderError {
40    #[error(
41        "failed to acquire semaphore permit for concurrent issuer loading - this indicates a serious concurrency issue or resource exhaustion"
42    )]
43    SemaphoreAcquire,
44    #[error(
45        "discovery document at '{configuration}' declares issuer '{found}', \
46         but the configured iss is '{expected}'"
47    )]
48    DiscoveryIssuerMismatch {
49        configuration: String,
50        expected: IssClaim,
51        found: IssClaim,
52    },
53    #[error(
54        "issuers '{}' resolve to iss '{iss}' with different JWKS ('{}'); none of them is loaded",
55        .issuers.join("', '"),
56        .jwks.join("', '")
57    )]
58    ConflictingJwks {
59        iss: IssClaim,
60        issuers: Vec<String>,
61        jwks: Vec<String>,
62    },
63}

81impl TrustedIssuerLoader {
82    /// Load trusted issuers either synchronously or asynchronously based on configuration.
83    pub(super) async fn load_trusted_issuers(
84        &self,
85        token_types: &TokenTypes,
86    ) -> Result<(), JwtServiceInitError> {
87        let trusted_issuers = unresolved_issuers(token_types);
88        match self.jwt_config.trusted_issuer_loader {
89            TrustedIssuerLoaderConfig::Sync { workers } => {
90                load_trusted_issuers(self, trusted_issuers, workers.get())
91                    .await
92                    .inspect_err(|e| {
93                        log_load_trusted_issuers_error(self.logger.as_ref(), e);
94                    })
95            },
96            TrustedIssuerLoaderConfig::Async { workers } => {
97                let loader = self.clone();
98                spawn_task(async move {
99                    let _ = load_trusted_issuers(&loader, trusted_issuers, workers.get())
100                        .await
101                        .inspect_err(|e| {
102                            log_load_trusted_issuers_error(loader.logger.as_ref(), e);
103                        });
104                });
105                Ok(())
106            },
107        }
108    }
109
110    /// Quick check so we don't get surprised if the program runs but can't validate
111    /// if signed authorization is unavailable and signature validation is enabled, log a warning
112    pub(crate) fn check_keys_loaded(&self) {
113        let signed_authz_available = self.key_service.has_keys();
114        if !signed_authz_available && self.jwt_config.jwt_sig_validation {
115            self.logger.log_any(JwtLogEntry::new(
116                "signed authorization is unavailable because no trusted issuers or JWKS were configured".to_string(),
117                Some(LogLevel::WARN),
118            ));
119        }
120    }
121}

123/// Every JWT issuer of `token_types` with its token types, not yet resolved.
124fn unresolved_issuers(token_types: &TokenTypes) -> Vec<IssuerConfig> {
125    token_types
126        .jwt_issuers()
127        .iter()
128        .map(|(name, issuer)| IssuerConfig {
129            policy: issuer.clone(),
130            tokens: token_types
131                .jwt_tokens_of(name)
132                .map(|token| (token.entity_type.clone(), token.metadata.clone()))
133                .collect(),
134            openid_config: None,
135        })
136        .collect()
137}

139/// Loads every issuer in two phases. Key sources are resolved first, so that
140/// issuers whose keys would collide under one `iss` are all rejected before any
141/// of them is installed; the outcome then does not depend on task order.
142async fn load_trusted_issuers(
143    loader: &TrustedIssuerLoader,
144    trusted_issuers: Vec<IssuerConfig>,
145    workers: usize,
146) -> Result<(), JwtServiceInitError> {
147    let semaphore = Arc::new(tokio::sync::Semaphore::new(workers));
148    let mut errors = Vec::new();
149
150    // Load local JWKS once before processing issuers to avoid redundant work
151    if loader.jwt_config.jwt_sig_validation
152        && let Some(jwks) = loader.jwt_config.jwks.as_ref()
153    {
154        loader
155            .key_service
156            .insert_keys_from_str(jwks)
157            .map_err(JwtServiceInitError::PrepareKeys)?;
158    }
159
160    let mut handles = Vec::new();
161    for issuer in trusted_issuers {
162        let permit = acquire(&semaphore).await?;
163        let loader = loader.clone();
164        handles.push(spawn_task(async move {
165            let http_client = loader.http_client.clone();
166            let issuer_id = issuer.policy.name.to_string();
167            let resolved = resolve_trusted_issuer(&loader, issuer, &http_client).await;
168            drop(permit);
169            (issuer_id, resolved)
170        }));
171    }
172    let mut resolved = Vec::with_capacity(handles.len());
173    for handle in handles {
174        match handle.await_result().await {
175            (_, Ok(issuer)) => resolved.push(issuer),
176            (issuer_id, Err(error)) => {
177                record_failure(loader, &issuer_id, &error);
178                errors.push(error);
179            },
180        }
181    }
182
183    let (resolved, rejected) = if loader.jwt_config.jwt_sig_validation {
184        reject_conflicting_key_sources(resolved)
185    } else {
186        // No keys are loaded, so a shared `iss` cannot plant any.
187        (resolved, Vec::new())
188    };
189    for (issuer_id, error) in rejected {
190        loader.logger.log_any(JwtLogEntry::new(
191            format!("trusted issuer '{issuer_id}' not loaded: {error}"),
192            Some(LogLevel::ERROR),
193        ));
194        loader.loading_state.add_trusted_issuer_failed(issuer_id);
195        errors.push(error.into());
196    }
197
198    let mut handles = Vec::new();
199    for issuer in resolved {
200        let permit = acquire(&semaphore).await?;
201        let loader = loader.clone();
202        handles.push(spawn_task(async move {
203            let issuer_id = issuer.config.policy.name.to_string();
204            let installed = install_trusted_issuer(&loader, issuer).await;
205            drop(permit);
206            (issuer_id, installed)
207        }));
208    }
209    for handle in handles {
210        if let (issuer_id, Err(error)) = handle.await_result().await {
211            record_failure(loader, &issuer_id, &error);
212            errors.push(error);
213        }
214    }
215
216    loader.check_keys_loaded();
217
218    if !errors.is_empty() {
219        return Err(JwtServiceInitError::LoadTrustedIssuersAggregate(
220            errors.into(),
221        ));
222    }
223
224    Ok(())
225}

253/// An issuer whose key source is known, ready to be installed.
254struct ResolvedIssuer {
255    config: IssuerConfig,
256    /// What tokens of this issuer carry in `iss`.
257    iss: IssClaim,
258    /// Unknown when discovery was skipped because no validation needs it.
259    jwks_uri: Option<url::Url>,
260}

262/// Splits off every issuer whose `iss` is shared with another issuer that loads
263/// its keys from a different JWKS. Keys are stored per `iss`, so either key set
264/// would also validate the other issuer's tokens; with no order between them,
265/// none of them is trusted.
266fn reject_conflicting_key_sources(
267    resolved: Vec<ResolvedIssuer>,
268) -> (Vec<ResolvedIssuer>, Vec<(String, TrustedIssuerLoaderError)>) {
269    let mut sources_by_iss: HashMap<&IssClaim, Vec<(&str, &url::Url)>> = HashMap::new();
270    for issuer in &resolved {
271        if let Some(jwks_uri) = &issuer.jwks_uri {
272            sources_by_iss
273                .entry(&issuer.iss)
274                .or_default()
275                .push((issuer.config.policy.name.as_str(), jwks_uri));
276        }
277    }
278
279    let mut conflicts: HashMap<IssClaim, (Vec<String>, Vec<String>)> = HashMap::new();
280    for (iss, sources) in sources_by_iss {
281        let mut jwks: Vec<String> = sources.iter().map(|(_, uri)| uri.to_string()).collect();
282        jwks.sort();
283        jwks.dedup();
284        if jwks.len() > 1 {
285            let mut issuers: Vec<String> =
286                sources.iter().map(|(id, _)| (*id).to_string()).collect();
287            issuers.sort();
288            conflicts.insert(iss.clone(), (issuers, jwks));
289        }
290    }
291
292    let (rejected, kept): (Vec<_>, Vec<_>) = resolved
293        .into_iter()
294        .partition(|issuer| issuer.jwks_uri.is_some() && conflicts.contains_key(&issuer.iss));
295    let rejected = rejected
296        .into_iter()
297        .map(|issuer| {
298            let (issuers, jwks) = conflicts[&issuer.iss].clone();
299            let error = TrustedIssuerLoaderError::ConflictingJwks {
300                iss: issuer.iss,
301                issuers,
302                jwks,
303            };
304            (issuer.config.policy.name.to_string(), error)
305        })
306        .collect();
307    (kept, rejected)
308}

321/// Finds where an issuer's keys come from: its discovery document, checked
322/// against the expected `iss`, or its static JWKS.
323async fn resolve_trusted_issuer(
324    loader: &TrustedIssuerLoader,
325    mut config: IssuerConfig,
326    http_client: &HttpClient,
327) -> Result<ResolvedIssuer, JwtServiceInitError> {
328    // this is what we expect to find in the JWT `iss` claim
329    let iss_claim = config.policy.iss_claim().clone();
330
331    if let KeySource::Discovery { configuration } = &config.policy.key_source
332        && (loader.jwt_config.jwt_sig_validation || loader.jwt_config.jwt_status_validation)
333    {
334        config.openid_config = Some(
335            discover(
336                configuration,
337                &iss_claim,
338                config.policy.name.as_str(),
339                loader.logger.as_ref(),
340                http_client,
341            )
342            .await?,
343        );
344    }
345
346    let jwks_uri = match &config.policy.key_source {
347        KeySource::Discovery { .. } => config
348            .openid_config
349            .as_ref()
350            .map(|openid| openid.jwks_uri.clone()),
351        KeySource::Static { jwks } => Some(jwks.clone()),
352    };
353
354    Ok(ResolvedIssuer {
355        config,
356        iss: iss_claim,
357        jwks_uri,
358    })
359}

361/// Fetches the keys of a resolved issuer and makes it available for validation.
362async fn install_trusted_issuer(
363    loader: &TrustedIssuerLoader,
364    issuer: ResolvedIssuer,
365) -> Result<(), JwtServiceInitError> {
366    let ResolvedIssuer {
367        config: iss_config,
368        iss: iss_claim,
369        jwks_uri,
370    } = issuer;
371    let http_client = loader.http_client.clone();
372
373    if loader.jwt_config.jwt_sig_validation
374        && let Some(jwks_uri) = &jwks_uri
375    {
376        loader
377            .key_service
378            .fetch_keys(
379                &iss_claim,
380                jwks_uri,
381                loader.logger.as_ref(),
382                http_client.clone(),
383            )
384            .await?;
385    }
386
387    loader.validators.init_for_iss(
388        &iss_config,
389        &loader.jwt_config,
390        &loader.status_lists,
391        loader.logger.as_ref(),
392    );
393
394    if loader.jwt_config.jwt_status_validation {
395        loader
396            .status_lists
397            .init_for_iss(
398                &iss_config,
399                InitForIssArgs {
400                    validators: &loader.validators,
401                    key_service: &loader.key_service,
402                    token_cache: loader.token_cache.clone(),
403                    logger: loader.logger.clone(),
404                    http_client,
405                    // `JwtConfig` has already been normalized in `JwtService::new`,
406                    // so this value is guaranteed to be in the safe range.
407                    refresh_interval_max: std::time::Duration::from_secs(
408                        loader.jwt_config.status_list_refresh_interval_max,
409                    ),
410                    cancel_tkn: loader.jwks_cancel_token.clone(),
411                },
412            )
413            .await?;
414    }
415
416    if loader.jwt_config.jwt_sig_validation
417        && let Some(jwks_uri) = jwks_uri
418    {
419        spawn_jwks_refresh(loader, iss_claim.clone(), jwks_uri);
420    }
421
422    loader.issuer_configs.insert(iss_claim, iss_config);
423
424    Ok(())
425}

427/// Starts the background task that keeps the JWKS of `iss` fresh, once per `iss`:
428/// issuers sharing an `iss` share its keys, as conflicting JWKS were rejected.
429fn spawn_jwks_refresh(loader: &TrustedIssuerLoader, iss: IssClaim, jwks_uri: url::Url) {
430    let notify = Arc::new(Notify::new());
431
432    let initial_interval = Duration::from_secs(
433        loader
434            .jwt_config
435            .jwks_refresh_interval
436            .unwrap_or(DEFAULT_JWKS_REFRESH_INTERVAL_SECS),
437    );
438    let min_interval = Duration::from_secs(loader.jwt_config.jwks_refresh_min_interval);
439    let config_override = loader.jwt_config.jwks_refresh_interval;
440
441    {
442        let mut notifiers = loader
443            .jwks_refresh_notifiers
444            .lock()
445            .expect("acquire jwks_refresh_notifiers lock");
446        match notifiers.entry(iss.clone()) {
447            Entry::Occupied(_) => return,
448            Entry::Vacant(slot) => {
449                slot.insert(notify.clone());
450            },
451        }
452    }
453
454    spawn_task(keep_jwks_updated(JwksRefreshParams {
455        key_service: loader.key_service.clone(),
456        iss,
457        jwks_uri,
458        initial_interval,
459        min_interval,
460        config_override,
461        notify,
462        logger: loader.logger.clone(),
463        cancel_tkn: loader.jwks_cancel_token.clone(),
464        http_client: loader.http_client.clone(),
465    }));
466}

468/// Fetches the `OpenID` configuration of a discovery issuer and checks that it
469/// is for the expected `iss` (OIDC Discovery §4.3).
470async fn discover(
471    configuration: &url::Url,
472    expected: &IssClaim,
473    issuer_id: &str,
474    logger: Option<&Logger>,
475    http_client: &HttpClient,
476) -> Result<OpenIdConfig, JwtServiceInitError> {
477    let openid_config = OpenIdConfig::get_from_url(configuration, http_client)
478        .await
479        .inspect_err(|e| {
480            logger.log_any(JwtLogEntry::new(
481                format!(
482                    "failed to get openid configuration for trusted issuer: '{issuer_id}': {e}"
483                ),
484                Some(LogLevel::ERROR),
485            ));
486        })?;
487    if openid_config.issuer != *expected {
488        return Err(TrustedIssuerLoaderError::DiscoveryIssuerMismatch {
489            configuration: configuration.to_string(),
490            expected: expected.clone(),
491            found: openid_config.issuer,
492        }
493        .into());
494    }
495    Ok(openid_config)
496}

510struct JwksRefreshParams {
511    key_service: Arc<KeyService>,
512    iss: IssClaim,
513    jwks_uri: url::Url,
514    initial_interval: Duration,
515    min_interval: Duration,
516    config_override: Option<u64>,
517    notify: Arc<Notify>,
518    logger: Option<Logger>,
519    cancel_tkn: CancellationToken,
520    http_client: HttpClient,
521}
522
523/// Background task that periodically re-fetches JWKS for a single trusted issuer
524async fn keep_jwks_updated(params: JwksRefreshParams) {
525    let JwksRefreshParams {
526        key_service,
527        iss,
528        jwks_uri,
529        initial_interval,
530        min_interval,
531        config_override,
532        notify,
533        logger,
534        cancel_tkn,
535        http_client,
536    } = params;
537
538    let mut interval = initial_interval;
539    let mut last_refresh = Utc::now() - interval;
540    let mut backoff = Backoff::default_fixed();
541
542    loop {
543        tokio::select! {
544            () = sleep(interval) => {},
545            () = notify.notified() => {
546                let elapsed = Utc::now().signed_duration_since(last_refresh).num_seconds().cast_unsigned();
547                if elapsed < min_interval.as_secs() {
548                    sleep(Duration::from_secs(min_interval.as_secs() - elapsed)).await;
549                }
550            },
551            () = cancel_tkn.cancelled() => { break; },
552        }
553
554        logger.log_any(JwtLogEntry::new(
555            format!("refreshing JWKS for issuer '{}'", iss.as_str()),
556            Some(LogLevel::DEBUG),
557        ));
558
559        match key_service
560            .refresh_keys(&iss, &jwks_uri, logger.as_ref(), &http_client)
561            .await
562        {
563            Ok(max_age) => {
564                last_refresh = Utc::now();
565                backoff.reset();
566
567                if let Some(config_secs) = config_override {
568                    interval = Duration::from_secs(config_secs);
569                } else if let Some(max_age_secs) = max_age {
570                    interval = if max_age_secs == 0 {
571                        min_interval
572                    } else {
573                        Duration::from_secs(max_age_secs)
574                    };
575                }
576
577                logger.log_any(JwtLogEntry::new(
578                    format!(
579                        "refreshed JWKS for issuer '{}', next refresh in {}s",
580                        iss.as_str(),
581                        interval.as_secs(),
582                    ),
583                    Some(LogLevel::INFO),
584                ));
585            },
586            Err(err) => {
587                logger.log_any(JwtLogEntry::new(
588                    format!(
589                        "failed to refresh JWKS for issuer '{}': {}",
590                        iss.as_str(),
591                        err,
592                    ),
593                    Some(LogLevel::ERROR),
594                ));
595
596                tokio::select! {
597                    _ = backoff.snooze() => {},
598                    () = cancel_tkn.cancelled() => { break; },
599                }
600            },
601        }
602    }
603}
