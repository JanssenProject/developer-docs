6use crate::common::{issuer_utils::IssClaim, policy_store::TrustedIssuer};
7use serde::Deserialize;
8use serde_json::Value;
9use smol_str::SmolStr;
10use std::{borrow::Cow, collections::HashMap, sync::Arc};
11
12/// Identifies where a validated [`Token`] came from.
13///
14/// A token is either backed by a JWT [`TrustedIssuer`] (the standard path) or by
15/// a custom issuer resolved by a [`CustomTokenProcessor`](crate::CustomTokenProcessor).
16#[derive(Debug, PartialEq, Clone)]
17pub(crate) enum TokenIssuer {
18    /// A JWT trusted issuer from the policy store.
19    Jwt(Arc<TrustedIssuer>),
20    /// A custom (non-JWT) issuer.
21    Custom(CustomTokenIssuerMeta),
22}
23
24/// Issuer metadata for a custom (non-JWT) token, carried on the built [`Token`]
25/// so the entity builder can resolve its entity type, id, and context key without
26/// a [`TrustedIssuer`].
27#[derive(Debug, PartialEq, Clone)]
28pub(crate) struct CustomTokenIssuerMeta {
29    /// The issuer name lowercased, given as the token's `iss`.
30    pub issuer_id: SmolStr,
31    /// The id used as the token entity id (supplied by the processor).
32    pub token_id: String,
33    /// Effective expiration (unix seconds) the processor reported, via
34    /// `ProcessedTokenClaims::expiration` or an `exp` claim. Carried here so the
35    /// entity builder can populate the `exp` attribute (and satisfy a schema that
36    /// declares `exp` required) even when the processor kept it out of `claims`.
37    pub expiration: Option<i64>,
38}
39
40/// Structure representing a validated token, used to derive a Cedar token entity.
41/// The `iss` field carries the resolved issuer; a JWT-backed token references a
42/// `TrustedIssuer`, a custom token carries [`CustomTokenIssuerMeta`]. Every
43/// validated token has one: a token is only accepted for the issuer its mapping
44/// names.
45#[derive(Debug, PartialEq, Clone)]
46pub(crate) struct Token {
47    pub iss: TokenIssuer,
48    pub(crate) claims: TokenClaims,
49}
50
51impl Token {
52    pub(crate) fn new(claims: TokenClaims, iss: TokenIssuer) -> Token {
53        Self { iss, claims }
54    }
55
56    pub(crate) fn get_claim_val(&self, name: &str) -> Option<&Value> {
57        self.claims.claims.get(name)
58    }
59
60    pub(crate) fn logging_info(&self, claim: &str) -> HashMap<String, serde_json::Value> {
61        self.claims.logging_info(claim)
62    }
63
64    pub(crate) fn claims_value(&self) -> &HashMap<String, Value> {
65        &self.claims.claims
66    }
67
68    /// The normalized issuer of the token: the configured `iss` of a JWT issuer,
69    /// which a validated JWT's `iss` claim equals, or a custom issuer's id.
70    pub(crate) fn normalized_issuer(&self) -> Cow<'_, IssClaim> {
71        match &self.iss {
72            TokenIssuer::Jwt(trusted_issuer) => Cow::Borrowed(trusted_issuer.iss_claim()),
73            TokenIssuer::Custom(meta) => Cow::Owned(IssClaim::new(meta.issuer_id.as_str())),
74        }
75    }
76}
77
78#[derive(Debug, PartialEq, Default, Deserialize, Clone)]
79pub(crate) struct TokenClaims {
80    #[serde(flatten)]
81    claims: HashMap<String, Value>,
82}
83
84impl From<HashMap<String, Value>> for TokenClaims {
85    fn from(claims: HashMap<String, Value>) -> Self {
86        Self { claims }
87    }
88}
89
90impl TryFrom<Value> for TokenClaims {
91    type Error = &'static str;
92
93    fn try_from(claims: Value) -> Result<Self, Self::Error> {
94        match claims {
95            Value::Object(map) => Ok(Self {
96                claims: map.into_iter().collect(),
97            }),
98            _ => Err("expected a JSON object for TokenClaims"),
99        }
100    }
101}
102
103impl TokenClaims {
104    pub(crate) fn get_claim(&self, name: &str) -> Option<TokenClaim<'_>> {
105        self.claims.get(name).map(|value| TokenClaim { value })
106    }
107
108    pub(crate) fn logging_info(&self, claim: &str) -> HashMap<String, serde_json::Value> {
109        let claim = if claim.is_empty() { "jti" } else { claim };
110
111        self.claims
112            .get(claim)
113            .map(|value| HashMap::from([(claim.to_string(), value.clone())]))
114            .unwrap_or_default()
115    }
116}
117
118pub(crate) struct TokenClaim<'a> {
119    value: &'a serde_json::Value,
120}
121
122impl TokenClaim<'_> {
123    pub(crate) fn value(&self) -> &Value {
124        self.value
125    }
