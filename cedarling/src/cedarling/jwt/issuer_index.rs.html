6use crate::common::issuer_utils::IssClaim;
7use crate::common::policy_store::IssuerName;
8
9use super::IssuerConfig;
10use ahash::{HashMap, HashMapExt};
11use std::{
12    collections::HashSet,
13    sync::{Arc, RwLock},
14};
15
16const MUTEX_POISONED_ERR: &str =
17    "IssuerIndex RwLock poisoned due to another thread panicking while holding the lock";
18
19/// The loaded issuers, by issuer id, and the `iss` values they cover. Several
20/// issuers may share one `iss`, so it is kept as a set rather than a key.
21///
22/// Thread-safe for concurrent reads and writes.
23pub(super) struct IssuerIndex {
24    issuer_idx: RwLock<HashMap<IssuerName, Arc<IssuerConfig>>>,
25    iss_index: RwLock<HashSet<IssClaim>>,
26}
27
28impl IssuerIndex {
29    /// Create a new, empty `IssuerIndex`
30    pub(super) fn new() -> Self {
31        Self {
32            issuer_idx: RwLock::new(HashMap::new()),
33            iss_index: RwLock::new(HashSet::new()),
34        }
35    }
36
37    /// Adds a loaded issuer under its id and records its `iss`.
38    pub(super) fn insert(&self, iss: IssClaim, config: IssuerConfig) {
39        self.issuer_idx
40            .write()
41            .expect(MUTEX_POISONED_ERR)
42            .insert(config.policy.name.clone(), Arc::new(config));
43
44        self.iss_index
45            .write()
46            .expect(MUTEX_POISONED_ERR)
47            .insert(iss);
48    }
49
50    /// Get the number of issuer configurations in the index; counted by issuer
51    /// id, since several issuers may share one `iss`.
52    pub(super) fn len(&self) -> usize {
53        let index = self.issuer_idx.read().expect(MUTEX_POISONED_ERR);
54        index.len()
55    }
56
57    /// Check if the index contains a specific iss claim
58    pub(super) fn contains_iss(&self, iss: &IssClaim) -> bool {
59        let index = self.iss_index.read().expect(MUTEX_POISONED_ERR);
60        index.contains(iss)
61    }
62
63    /// Find the iss claim for a given issuer ID (policy store key)
64    /// Returns `true` if issuer with the given ID is found
65    pub(super) fn is_issuer_id_present(&self, issuer_id: &str) -> bool {
66        self.issuer_idx
67            .read()
68            .expect(MUTEX_POISONED_ERR)
69            .get(issuer_id)
70            .is_some()
71    }
72
73    /// Get all issuer IDs in the index of loaded trusted issuers
74    pub(super) fn loaded_issuer_ids(&self) -> HashSet<String> {
75        let index = self.issuer_idx.read().expect(MUTEX_POISONED_ERR);
76        index.keys().map(ToString::to_string).collect()
77    }
78}
