6//! Custom token processing extension point.
7//!
8//! JWT validation is hardcoded in [`JwtService`](crate::jwt). This module adds a
9//! pluggable interface so Rust consumers can authorize on **non-JWT** tokens,
10//! opaque tokens, API keys, vendor-specific formats, or tokens whose validation
11//! uses crypto operations Cedarling does not implement (e.g. attenuation /
12//! chain-of-custody proofs).
13//!
14//! A consumer implements [`CustomTokenProcessor`] and registers it on a live
15//! instance via [`Cedarling::set_custom_token_processor`](crate::Cedarling::set_custom_token_processor).
16//! The processor turns a raw payload into [`ProcessedTokenClaims`]; those claims
17//! then flow through the *existing* entity-builder and context machinery, so a
18//! custom token becomes a Cedar entity under `context.tokens.*` exactly like a JWT.
19
20use async_trait::async_trait;
21use serde_json::Value;
22use std::collections::HashMap;
23use std::time::Duration;
24
25/// A consumer-supplied processor that validates a raw token payload and maps it
26/// to claims.
27///
28/// One processor handles all custom tokens and dispatches internally on
29/// `mapping`. Its output is authoritative: Cedarling performs no signature or
30/// issuer verification of its own on the payload, so the processor is fully
31/// trusted.
32#[cfg_attr(not(any(target_arch = "wasm32", target_arch = "wasm64")), async_trait)]
33#[cfg_attr(any(target_arch = "wasm32", target_arch = "wasm64"), async_trait(?Send))]
34pub trait CustomTokenProcessor: Send + Sync {
35    /// Validate `payload` and map it to claims.
36    ///
37    /// - `mapping`: the entity type mapping from the request's `TokenInput`
38    ///   (e.g. `"Acme::CustomToken"`); use it to dispatch between token formats.
39    /// - `payload`: the raw token string, opaque to Cedarling.
40    ///
41    /// Returns [`ProcessedTokenClaims`] on success. On error only this token is
42    /// dropped and the request continues with the others.
43    ///
44    /// The framework may race this future against a configured timeout deadline.
45    /// Cancellation still requires the future to yield, so a `process` that blocks
46    /// without an await point runs to completion regardless: treat the timeout as a
47    /// backstop, not as a guarantee that work stops.
48    async fn process(
49        &self,
50        mapping: &str,
51        payload: &str,
52    ) -> Result<ProcessedTokenClaims, CustomTokenError>;
53}
54
55/// The result of [`CustomTokenProcessor::process`].
56#[derive(Debug, Clone)]
57pub struct ProcessedTokenClaims {
58    /// Claims extracted from the token, used to build the Cedar token entity.
59    pub claims: HashMap<String, Value>,
60    /// The entity id for the resulting token entity.
61    pub token_id: String,
62    /// Optional expiration (unix seconds). When set, the token is rejected once
63    /// that time has passed and the value bounds the token-cache TTL, without the
64    /// processor having to put an `exp` into [`claims`](Self::claims). Falls back to
65    /// an `exp` claim when `None`; an explicit value wins over the claim.
66    ///
67    /// Beyond rejection and cache TTL, this also populates the token entity's `exp`
68    /// attribute — so it satisfies a schema that declares `exp` required and is
69    /// readable by a policy as `context.tokens.*.exp`, without a separate `exp` claim.
70    pub expiration: Option<i64>,
71    /// Whether this validation result may be cached. [`new`](Self::new) sets it to
72    /// `true`, so revocation-sensitive tokens must opt out explicitly via
73    /// [`with_cacheable(false)`](Self::with_cacheable) to make every request
74    /// re-run `process`.
75    pub cacheable: bool,
76}
77
78impl ProcessedTokenClaims {
79    /// Build a cacheable result with no expiration.
80    pub fn new(claims: HashMap<String, Value>, token_id: impl Into<String>) -> Self {
81        Self {
82            claims,
83            token_id: token_id.into(),
84            expiration: None,
85            cacheable: true,
86        }
87    }
88
89    /// Set whether this result may be cached. See [`cacheable`](Self::cacheable).
90    #[must_use]
91    pub fn with_cacheable(mut self, cacheable: bool) -> Self {
92        self.cacheable = cacheable;
93        self
94    }
95
96    /// Set the expiration (unix seconds). See [`expiration`](Self::expiration).
97    #[must_use]
98    pub fn with_expiration(mut self, exp: i64) -> Self {
99        self.expiration = Some(exp);
100        self
101    }
102}
103
104/// Errors returned by a [`CustomTokenProcessor`], plus the framework-side timeout
105/// variant.
106#[derive(Debug, thiserror::Error)]
107pub enum CustomTokenError {
108    /// The processor rejected the token or failed to validate it.
109    #[error("custom token processing failed: {0}")]
110    Processing(String),
111
112    /// `process` did not complete within the configured timeout. Kept distinct
113    /// from [`Processing`](Self::Processing) so telemetry can separate "slow" from
114    /// "rejected".
115    #[error("custom token processing timed out after {0:?}")]
116    Timeout(Duration),
117
118    /// No custom issuer declares the requested token type. Cedarling itself no
119    /// longer returns this: an undeclared mapping is dropped before any processor
120    /// runs. A processor may still return it; kept so existing processors build.
121    #[error("no custom issuer declares token type '{0}'")]
122    UnknownMapping(String),
123
124    /// The processor reported an expiration that has already passed, via
125    /// [`ProcessedTokenClaims::expiration`] or an `exp` claim. Cedarling does not
126    /// validate a custom payload, but it does honor an expiration the processor
127    /// chose to report, matching what the token cache already enforces.
128    #[error("custom token for '{mapping}' expired at {exp}")]
129    Expired {
130        /// The requested Cedar entity type name.
131        mapping: String,
132        /// The expiration (unix seconds) that had already passed.
133        exp: i64,
134    },
135
136    /// The processor returned an empty `token_id`, which would produce a degenerate
137    /// `EntityType::""` Cedar UID shared by every token from that processor.
138    #[error("custom token for '{0}' has an empty token_id")]
139    EmptyTokenId(String),
140
141    /// A required claim was absent from the processed output.
142    #[error("custom token missing required claim: {0}")]
143    MissingRequiredClaim(String),
144
145    /// A token routed to a custom issuer but no [`CustomTokenProcessor`] is registered.
146    #[error("no CustomTokenProcessor registered for custom mapping '{0}'")]
147    NoProcessorRegistered(String),
148}
