18// async_traits are Send by default but wasm-bindgen doesn't support those
19// so we opt out of it for the wasm bindings to compile.
20//
21// see this relevant discussion: https://github.com/rustwasm/wasm-bindgen/issues/2409
22#[cfg_attr(not(any(target_arch = "wasm32", target_arch = "wasm64")), async_trait)]
23#[cfg_attr(any(target_arch = "wasm32", target_arch = "wasm64"), async_trait(?Send))]
24pub(super) trait GetFromUrl<T: for<'de> serde::Deserialize<'de>> {
25    /// Send a get request to receive the resource from a URL
26    async fn get_from_url(url: &Url, client: &HttpClient) -> Result<T, HttpError> {
27        require_secure_url(url)?;
28
29        // add delay to simulate network latency and test async behavior
30        // it would be great to implement delay in mock server, but mockito doesn't support it.
31        #[cfg(test)]
32        {
33            use std::time::Duration;
34            use tokio::time::sleep;
35
36            sleep(Duration::from_millis(1)).await;
37        }
38
39        let result = client
40            .get_json::<T>(url.as_str())
41            .await
42            .map_err(HttpError::Request)?;
43
44        Ok(result)
45    }
46}

48/// Returns `Ok(())` if `url` is safe to fetch from a JWT-validation context.
49///
50/// "Safe" means either `https`, or `http` pointed at a loopback address
51/// (`localhost`, `127.0.0.1`, `::1`)
52pub(super) fn require_secure_url(url: &Url) -> Result<(), HttpError> {
53    if url.scheme().eq_ignore_ascii_case("https") {
54        return Ok(());
55    }
56
57    if url.scheme().eq_ignore_ascii_case("http") && is_loopback_host(url) {
58        return Ok(());
59    }
60
61    Err(HttpError::InsecureScheme {
62        url: url.to_string(),
63        scheme: url.scheme().to_string(),
64    })
65}
66
67fn is_loopback_host(url: &Url) -> bool {
68    match url.host() {
69        Some(url::Host::Ipv4(addr)) => addr.is_loopback(),
70        Some(url::Host::Ipv6(addr)) => addr.is_loopback(),
71        Some(url::Host::Domain(domain)) => domain.eq_ignore_ascii_case("localhost"),
72        None => false,
73    }
74}

85fn deserialize_url<'de, D>(deserializer: D) -> Result<Url, D::Error>
86where
87    D: Deserializer<'de>,
88{
89    let url_str = String::deserialize(deserializer)?;
90    let url = Url::parse(&url_str)
91        .map_err(|e| de::Error::custom(format!("invalid url '{url_str}': {e}")))?;
92    Ok(url)
93}
94
95fn deserialize_opt_url<'de, D>(deserializer: D) -> Result<Option<Url>, D::Error>
96where
97    D: Deserializer<'de>,
98{
99    let url_str = Option::<String>::deserialize(deserializer)?;
100    let url = url_str
101        .as_ref()
102        .map(|url_str| {
103            Url::parse(url_str)
104                .map_err(|e| de::Error::custom(format!("invalid url '{url_str}': {e}")))
105        })
106        .transpose()?;
107    Ok(url)
108}

114impl JwkSet {
115    pub(super) async fn get_from_url_with_max_age(
116        url: &Url,
117        client: &HttpClient,
118    ) -> Result<(Self, Option<u64>), HttpError> {
119        require_secure_url(url)?;
120
121        let response = client
122            .get_with_retry(url.as_str())
123            .await
124            .map_err(HttpError::Request)?;
125
126        let max_age = response
127            .headers()
128            .get(CACHE_CONTROL)
129            .and_then(|v| v.to_str().ok())
130            .and_then(parse_max_age);
131
132        // Read the body with the configured size cap so a hostile JWKS endpoint
133        // can't OOM the backend with an oversized response.
134        let bytes =
135            crate::http_utils::read_response_capped(response, client.max_response_size_bytes())
136                .await
137                .map_err(|e| HttpError::Request(HttpClientError::new(e, None)))?;
138        let jwk_set: JwkSet =
139            serde_json::from_slice(&bytes).map_err(HttpError::JsonDeserializeBytes)?;
140
141        Ok((jwk_set, max_age))
142    }
143}

145/// Parses `max-age=<seconds>` from a `Cache-Control` header value.
146fn parse_max_age(cache_control: &str) -> Option<u64> {
147    cache_control
148        .split(',')
149        .map(str::trim)
150        .filter_map(|directive| directive.split_once('='))
151        .find(|(name, _)| name.trim().eq_ignore_ascii_case("max-age"))
152        .and_then(|(_, value)| value.trim().parse::<u64>().ok())
153}

155// NOTE: we cant use the async_trait here since this is called from another async
156// function which requires this to be Send.
157impl StatusListJwtStr {
158    pub(super) async fn get_from_url(url: &Url, client: &HttpClient) -> Result<Self, HttpError> {
159        require_secure_url(url)?;
160
161        let response = client
162            .get_with_retry_with(url.as_str(), |b| {
163                b.header("Accept", "application/statuslist+jwt")
164            })
165            .await
166            .map_err(HttpError::Request)?;
167
168        if let Some(content_type) = response.headers().get("Content-Type") {
169            let content_type = content_type
170                .to_str()
171                .map_err(|e| HttpError::InvalidHeader("Content-Type".to_string(), e))?;
172            if content_type != "application/statuslist+jwt" {
173                return Err(HttpError::Unsupported(format!(
174                    "got unsupported status list type, '{0}', from '{1}'. Cedarling currently only supports 'application/statuslist+jwt'",
175                    content_type,
176                    url.as_str(),
177                )));
178            }
179        }
180
181        let bytes =
182            crate::http_utils::read_response_capped(response, client.max_response_size_bytes())
183                .await
184                .map_err(|e| HttpError::Request(HttpClientError::new(e, None)))?;
185        let status_list_jwt = String::from_utf8(bytes).map_err(HttpError::InvalidUtf8)?;
186
187        Ok(StatusListJwtStr::new(status_list_jwt))
188    }
189}

191#[derive(Debug, thiserror::Error)]
192pub(crate) enum HttpError {
193    #[error("HTTP request failed: {0}")]
194    Request(#[from] HttpClientError),
195    #[error("failed to deserialize response body bytes as JSON: {0}")]
196    JsonDeserializeBytes(#[source] serde_json::Error),
197    #[error("response body is not valid UTF-8: {0}")]
198    InvalidUtf8(#[source] std::string::FromUtf8Error),
199    #[error("the value of the '{0}' header is invalid: {1}")]
200    InvalidHeader(String, ToStrError),
201    #[error("{0}")]
202    Unsupported(String),
203    #[error(
204        "refusing to fetch '{url}' over insecure scheme '{scheme}'; \
205         JWT validation requires https (loopback http is allowed)"
206    )]
207    InsecureScheme { url: String, scheme: String },
208}

252    #[test]
253    fn allows_http_loopback_hosts() {
254        for raw in [
255            "http://localhost/.well-known/openid-configuration",
256            "http://localhost:8080/jwks",
257            "http://127.0.0.1:1234/jwks",

270        let url = Url::parse("http://idp.example.com/.well-known/openid-configuration").unwrap();
271        let err = require_secure_url(&url).expect_err("expected http remote to be rejected");
272        match err {
273            HttpError::InsecureScheme { scheme, .. } => assert_eq!(scheme, "http"),
274            other => panic!("expected InsecureScheme, got {other:?}"),
