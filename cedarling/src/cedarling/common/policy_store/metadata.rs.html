6//! Policy store metadata types for identification, versioning, and integrity validation.
7
8use chrono::{DateTime, Utc};
9
10/// Metadata for a policy store.
11///
12/// Contains identification, versioning, and descriptive information about a policy store.
13#[derive(Debug, Clone, PartialEq)]
14pub(crate) struct PolicyStoreMetadata {
15    /// The version of the Cedar policy language used in this policy store
16    pub(crate) cedar_version: String,
17    /// Policy store configuration
18    pub(crate) policy_store: PolicyStoreInfo,
19}
20
21/// Core information about a policy store.
22#[derive(Debug, Clone, PartialEq)]
23pub(crate) struct PolicyStoreInfo {
24    /// Unique identifier for the policy store (hex hash)
25    pub id: String,
26    /// Human-readable name for the policy store
27    pub name: String,
28    /// Optional description of the policy store
29    pub description: Option<String>,
30    /// Semantic version of the policy store content
31    pub version: String,
32    /// ISO 8601 timestamp when created
33    pub created_date: Option<DateTime<Utc>>,
34    /// ISO 8601 timestamp when last modified
35    pub updated_date: Option<DateTime<Utc>>,
36}
