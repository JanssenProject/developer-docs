6//! Policy store metadata types for identification, versioning, and integrity validation.

8use chrono::{DateTime, Utc};
9use serde::{Deserialize, Serialize};
10
11/// Helper module for serializing Optional [`DateTime`]
12mod datetime_option {
13    use chrono::{DateTime, Utc};
14    use serde::{Deserialize, Deserializer, Serializer};
15
16    #[allow(clippy::ref_option)]
17    pub(super) fn serialize<S>(
18        date: &Option<DateTime<Utc>>,
19        serializer: S,
20    ) -> Result<S::Ok, S::Error>
21    where
22        S: Serializer,
23    {
24        match date {
25            Some(dt) => serializer.serialize_some(&dt.to_rfc3339()),
26            None => serializer.serialize_none(),
27        }
28    }
29
30    pub(super) fn deserialize<'de, D>(deserializer: D) -> Result<Option<DateTime<Utc>>, D::Error>
31    where
32        D: Deserializer<'de>,
33    {
34        let opt: Option<String> = Option::deserialize(deserializer)?;
35        match opt {
36            Some(s) => DateTime::parse_from_rfc3339(&s)
37                .map(|dt| Some(dt.with_timezone(&Utc)))
38                .map_err(serde::de::Error::custom),
39            None => Ok(None),
40        }
41    }
42}

44/// Metadata for a policy store.
45///
46/// Contains identification, versioning, and descriptive information about a policy store.
47#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
48pub(crate) struct PolicyStoreMetadata {
49    /// The version of the Cedar policy language used in this policy store
50    pub(crate) cedar_version: String,
51    /// Policy store configuration
52    pub(crate) policy_store: PolicyStoreInfo,
53}
54
55/// Core information about a policy store.
56#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
57pub(crate) struct PolicyStoreInfo {
58    /// Unique identifier for the policy store (hex hash)
59    #[serde(default)]
60    pub id: String,
61    /// Human-readable name for the policy store
62    pub name: String,
63    /// Optional description of the policy store
64    #[serde(default, skip_serializing_if = "Option::is_none")]
65    pub description: Option<String>,
66    /// Semantic version of the policy store content
67    #[serde(default)]
68    pub version: String,
69    /// ISO 8601 timestamp when created
70    #[serde(
71        default,
72        skip_serializing_if = "Option::is_none",
73        with = "datetime_option"
74    )]
75    pub created_date: Option<DateTime<Utc>>,
76    /// ISO 8601 timestamp when last modified
77    #[serde(
78        default,
79        skip_serializing_if = "Option::is_none",
80        with = "datetime_option"
81    )]
82    pub updated_date: Option<DateTime<Utc>>,
83}

85#[cfg(test)]
86mod tests {
87    use super::*;
88
89    #[test]
90    fn test_policy_store_metadata_serialization() {
91        // Use a fixed timestamp for deterministic comparison
92        let created = DateTime::parse_from_rfc3339("2024-01-01T00:00:00Z")
93            .unwrap()
94            .with_timezone(&Utc);
95        let updated = DateTime::parse_from_rfc3339("2024-01-02T00:00:00Z")
96            .unwrap()
97            .with_timezone(&Utc);
98
99        let metadata = PolicyStoreMetadata {
100            cedar_version: "4.4.0".to_string(),
101            policy_store: PolicyStoreInfo {
102                id: "abc123".to_string(),
103                name: "test_store".to_string(),
104                description: Some("A test policy store".to_string()),
105                version: "1.0.0".to_string(),
106                created_date: Some(created),
107                updated_date: Some(updated),
108            },
109        };
110
111        // Test serialization
112        let json = serde_json::to_string(&metadata).unwrap();
113        assert!(json.contains("cedar_version"));
114        assert!(json.contains("4.4.0"));
115
116        // Test deserialization - compare whole structure
117        let deserialized: PolicyStoreMetadata = serde_json::from_str(&json).unwrap();
118        assert_eq!(deserialized, metadata);
119    }
