6//! Policy store loader with format detection and directory loading support.
7//!
8//! # Internal API Note
9//!
10//! This module is part of the internal implementation. External users should use the
11//! `Cedarling` API with `BootstrapConfig` to load policy stores.
12//!
13//! # Loading Archives (.cjar files)
14//!
15//! Archives are loaded using `ArchiveVfs`, which implements the `VfsFileSystem` trait.
16//! This design:
17//! - Works in WASM (no temp file extraction needed)
18//! - Is efficient (reads files on-demand from archive)
19//! - Is secure (no temp file cleanup concerns)
20//!
21//! # Blocking
22//!
23//! Every function here is synchronous and does blocking I/O and CPU work
24//! (unzipping, parsing, building the runtime store). Callers on an async
25//! runtime run it off the executor as one blocking task.
26

38/// Read a policy store from a directory path.
39///
40/// This function uses `PhysicalVfs` to read from the local filesystem.
41/// It is only available on native platforms (not WASM).
42#[cfg(not(target_arch = "wasm32"))]
43pub(crate) fn read_policy_store_directory(
44    path: &Path,
45    strict: bool,
46) -> Result<ParsedPolicyStore, ReadStoreError> {
47    let path_str = path
48        .to_str()
49        .ok_or_else(|| PolicyStoreError::PathNotFound {
50            path: path.display().to_string(),
51        })?;
52
53    let vfs = super::vfs_adapter::PhysicalVfs::new();
54    read_from_vfs(&vfs, path_str, strict, None)
55}

82/// Read a policy store from archive bytes.
83///
84/// Takes the bytes by value so the archive is read in place rather than
85/// copied. This function is useful for:
86/// - WASM environments where file system access is not available
87/// - Loading archives fetched from URLs
88/// - Loading archives from any byte source
89pub(crate) fn read_policy_store_archive_bytes(
90    bytes: Vec<u8>,
91    strict: bool,
92    limits: ArchiveLimits,
93) -> Result<ParsedPolicyStore, ReadStoreError> {
94    // Hashed here, the one place `.cjar` bytes become a store, so no archive
95    // store exists without its `bundle_hash`.
96    let bundle_hash = compute_bundle_hash(&bytes);
97    let archive_vfs = ArchiveVfs::from_buffer(bytes, limits).map_err(PolicyStoreError::from)?;
98    read_from_vfs(&archive_vfs, ".", strict, Some(bundle_hash))
99}

101/// Read a policy store rooted at `dir` from any VFS backend into its runtime
102/// form. The store's spec version decides where its files live and how they
103/// parse (see [`formats::read_policy_store`]).
104pub(super) fn read_from_vfs(
105    vfs: &dyn VfsFileSystem,
106    dir: &str,
107    strict: bool,
108    bundle_hash: Option<Digest>,
109) -> Result<ParsedPolicyStore, ReadStoreError> {
110    formats::read_policy_store(&StoreFs::open(vfs, dir)?, strict, bundle_hash)
111}

113/// One file read from a store: a policy, template, entity or issuer file.
114///
115/// All four are a name plus text; the directory a file came from is what gives
116/// it meaning, so they share one type rather than four identical ones.
117#[derive(Debug, Clone)]
118pub(crate) struct StoreFile {
119    /// File name, without the directory it was read from.
120    pub name: String,
121    /// File content as UTF-8 text.
122    pub content: String,
123}

125/// Describes where a schema source was found in the policy store.
126enum SchemaSource {
127    /// The single schema file exists at this path.
128    SingleFile { path: String },
129    /// The schema directory exists at this path.
130    Directory(String),
131    /// No schema source found.
132    None {
133        searched_file: String,
134        searched_dir: String,
135    },
136}
137
138impl SchemaSource {
139    fn exists(&self) -> bool {
140        matches!(
141            self,
142            SchemaSource::SingleFile { .. } | SchemaSource::Directory(_)
143        )
144    }
145}

147/// The schema of a store, as read by [`StoreFs::schema`].
148pub(super) struct LoadedSchema {
149    /// Parsed schema (optional — absent when running without schema)
150    pub(super) schema: Option<ParsedSchema>,
151    /// Whether a schema source was present in the store, regardless of
152    /// whether it was loaded/parsed.
153    pub(super) source_exists: bool,
154}

156/// Read-only view of a store's files, rooted at the store's top directory.
157///
158/// Version modules reach the files only through this type: it owns path
159/// joining, UTF-8 decoding, extension checks and the recursion limit, so a
160/// spec version decides where its files live but not how they are read.
161///
162/// Holds the backend as `dyn` so that version modules stay free of the
163/// backend type parameter.
164pub(super) struct StoreFs<'a> {
165    vfs: &'a dyn VfsFileSystem,
166    root: &'a str,
167}
168
169impl<'a> StoreFs<'a> {
170    /// Maximum directory recursion depth to prevent stack overflow from deeply
171    /// nested `.cjar` archives or filesystem trees.
172    const MAX_RECURSION_DEPTH: usize = 64;
173
174    /// Opens the store rooted at `root`, which must be an existing directory.
175    pub(super) fn open(
176        vfs: &'a dyn VfsFileSystem,
177        root: &'a str,
178    ) -> Result<Self, PolicyStoreError> {
179        if !vfs.exists(root) {
180            return Err(PolicyStoreError::PathNotFound {
181                path: root.to_string(),
182            });
183        }
184
185        if !vfs.is_dir(root) {
186            return Err(PolicyStoreError::NotADirectory {
187                path: root.to_string(),
188            });
189        }
190
191        Ok(Self { vfs, root })
192    }
193
194    /// Joins a store-relative path onto the root, handling "." correctly.
195    fn path(&self, relative: &str) -> String {
196        if self.root == "." || self.root.is_empty() {
197            relative.to_string()
198        } else {
199            format!("{}/{relative}", self.root)
200        }
201    }
202
203    /// Reads a file the store must contain.
204    pub(super) fn read_required_file(&self, relative: &str) -> Result<String, PolicyStoreError> {
205        let path = self.path(relative);
206        if !self.vfs.exists(&path) {
207            return Err(ValidationError::MissingRequiredFile {
208                file: relative.to_string(),
209            }
210            .into());
211        }
212        self.read_text(&path)
213    }
214
215    /// Reads a file the store may contain; `None` when it is absent.
216    pub(super) fn optional_file(&self, relative: &str) -> Result<Option<String>, PolicyStoreError> {
217        let path = self.path(relative);
218        if !self.vfs.exists(&path) {
219            return Ok(None);
220        }
221        self.read_text(&path).map(Some)
222    }
223
224    /// Whether the store contains the directory.
225    pub(super) fn has_dir(&self, relative: &str) -> bool {
226        self.vfs.is_dir(&self.path(relative))
227    }
228
229    /// Fails unless the store contains the directory.
230    pub(super) fn require_dir(&self, relative: &str) -> Result<(), PolicyStoreError> {
231        let path = self.path(relative);
232        if !self.vfs.exists(&path) {
233            return Err(ValidationError::MissingRequiredDirectory {
234                directory: relative.to_string(),
235            }
236            .into());
237        }
238
239        if !self.vfs.is_dir(&path) {
240            return Err(PolicyStoreError::NotADirectory { path });
241        }
242
243        Ok(())
244    }
245
246    /// Reads and parses the schema from a single file, or failing that, from
247    /// every `.cedarschema` file in a directory. With `strict`, a store that
248    /// has neither is an error.
249    pub(super) fn schema(
250        &self,
251        file: &str,
252        dir: &str,
253        strict: bool,
254    ) -> Result<LoadedSchema, PolicyStoreError> {
255        let source = self.resolve_schema_source(file, dir);
256        let source_exists = source.exists();
257
258        if strict
259            && let SchemaSource::None {
260                searched_file,
261                searched_dir,
262            } = source
263        {
264            return Err(ValidationError::MissingSchemaSource {
265                searched_file,
266                searched_dir,
267            }
268            .into());
269        }
270
271        Ok(LoadedSchema {
272            schema: self.load_schema(&source, file)?,
273            source_exists,
274        })
275    }

277    /// Reads every `.cedar` file under a directory the store must contain,
278    /// recursively.
279    pub(super) fn cedar_files(&self, relative: &str) -> Result<Vec<StoreFile>, PolicyStoreError> {
280        let mut files = Vec::new();
281        self.load_cedar_files_recursive(&self.path(relative), &mut files, 0)?;
282        Ok(files)
283    }
284
285    /// Like [`cedar_files`](Self::cedar_files), but an absent directory yields
286    /// no files.
287    pub(super) fn optional_cedar_files(
288        &self,
289        relative: &str,
290    ) -> Result<Vec<StoreFile>, PolicyStoreError> {
291        if !self.vfs.exists(&self.path(relative)) {
292            return Ok(Vec::new());
293        }
294        self.cedar_files(relative)
295    }
296
297    /// Reads every `.json` file directly in a directory, if the store has it;
298    /// subdirectories are skipped.
299    pub(super) fn optional_json_files(
300        &self,
301        relative: &str,
302    ) -> Result<Vec<StoreFile>, PolicyStoreError> {
303        let dir = self.path(relative);
304        if !self.vfs.exists(&dir) {
305            return Ok(Vec::new());
306        }
307
308        let entries =
309            self.vfs
310                .read_dir(&dir)
311                .map_err(|source| PolicyStoreError::DirectoryReadError {
312                    path: dir.clone(),
313                    source,
314                })?;
315
316        let mut files = Vec::new();
317        for entry in entries {
318            if entry.is_dir {
319                continue;
320            }
321            Self::check_extension(&entry, ".json")?;
322            let content = self.read_text(&entry.path)?;
323            files.push(StoreFile {
324                name: entry.name,
325                content,
326            });
327        }
328
329        Ok(files)
330    }
331
332    /// Reads a file as UTF-8 text.
333    fn read_text(&self, path: &str) -> Result<String, PolicyStoreError> {
334        let bytes = self
335            .vfs
336            .read_file(path)
337            .map_err(|source| PolicyStoreError::FileReadError {
338                path: path.to_string(),
339                source,
340            })?;
341
342        String::from_utf8(bytes).map_err(|e| PolicyStoreError::FileReadError {
343            path: path.to_string(),
344            source: std::io::Error::new(std::io::ErrorKind::InvalidData, e),
345        })
346    }
347
348    fn check_extension(
349        entry: &super::vfs_adapter::DirEntry,
350        expected: &str,
351    ) -> Result<(), PolicyStoreError> {
352        if entry.name.to_lowercase().ends_with(expected) {
353            return Ok(());
354        }
355        Err(ValidationError::InvalidFileExtension {
356            file: entry.path.clone(),
357            expected: expected.to_string(),
358            actual: Path::new(&entry.name)
359                .extension()
360                .and_then(|s| s.to_str())
361                .unwrap_or("(none)")
362                .to_string(),
363        }
364        .into())
365    }
366
367    /// Resolve where the schema lives (or that it's absent), without any I/O
368    /// beyond lightweight existence checks.
369    fn resolve_schema_source(&self, file: &str, dir: &str) -> SchemaSource {
370        let schema_path = self.path(file);
371        if self.vfs.exists(&schema_path) {
372            return SchemaSource::SingleFile { path: schema_path };
373        }
374        let schemas_dir = self.path(dir);
375        if self.vfs.exists(&schemas_dir) {
376            return SchemaSource::Directory(schemas_dir);
377        }
378        SchemaSource::None {
379            searched_file: schema_path,
380            searched_dir: schemas_dir,
381        }
382    }
383
384    /// Load and parse schema from a pre-resolved [`SchemaSource`].
385    /// Returns `Ok(None)` when no schema source exists.
386    fn load_schema(
387        &self,
388        source: &SchemaSource,
389        file_name: &str,
390    ) -> Result<Option<ParsedSchema>, PolicyStoreError> {
391        match source {
392            SchemaSource::SingleFile { path, .. } => {
393                let content = self.read_schema_file_content(path)?.ok_or_else(|| {
394                    PolicyStoreError::Validation(ValidationError::MissingSchemaSource {
395                        searched_file: path.clone(),
396                        searched_dir: String::new(),
397                    })
398                })?;
399                Ok(Some(ParsedSchema::parse(&content, file_name)?))
400            },
401            SchemaSource::Directory(path) => self.load_schema_from_directory(path),
402            SchemaSource::None { .. } => Ok(None),
403        }
404    }
405
406    /// Read the raw content of a single schema file.
407    /// Returns `None` if the file does not exist (no parsing).
408    fn read_schema_file_content(&self, path: &str) -> Result<Option<String>, PolicyStoreError> {
409        match self.vfs.read_file(path) {
410            Ok(bytes) => {
411                String::from_utf8(bytes)
412                    .map(Some)
413                    .map_err(|e| PolicyStoreError::FileReadError {
414                        path: path.to_string(),
415                        source: std::io::Error::new(std::io::ErrorKind::InvalidData, e),
416                    })
417            },
418            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(None),
419            Err(source) => Err(PolicyStoreError::FileReadError {
420                path: path.to_string(),
421                source,
422            }),
423        }
424    }

426    /// Load schema from the schema directory, combining all `.cedarschema` files.
427    fn load_schema_from_directory(
428        &self,
429        path: &str,
430    ) -> Result<Option<ParsedSchema>, PolicyStoreError> {
431        if !self.vfs.exists(path) {
432            return Ok(None);
433        }
434        if !self.vfs.is_dir(path) {
435            return Err(PolicyStoreError::NotADirectory {
436                path: path.to_string(),
437            });
438        }
439
440        let entries =
441            self.vfs
442                .read_dir(path)
443                .map_err(|source| PolicyStoreError::DirectoryReadError {
444                    path: path.to_string(),
445                    source,
446                })?;
447
448        let raw_files = self.read_schema_files(entries)?;
449        if raw_files.is_empty() {
450            return Err(ValidationError::EmptySchemaDirectory {
451                path: path.to_string(),
452            }
453            .into());
454        }
455
456        let parsed = ParsedSchema::parse_multiple(&raw_files)?;
457        Ok(Some(parsed))
458    }

460    /// Read and validate all `.cedarschema` files from directory entries.
461    fn read_schema_files(
462        &self,
463        entries: Vec<super::vfs_adapter::DirEntry>,
464    ) -> Result<Vec<SchemaFile>, PolicyStoreError> {
465        let mut files: Vec<SchemaFile> = Vec::new();
466        for entry in entries {
467            if entry.is_dir {
468                continue;
469            }
470            if !entry.name.to_lowercase().ends_with(".cedarschema") {
471                continue;
472            }
473            let content = self.read_text(&entry.path)?;
474            files.push(SchemaFile {
475                name: entry.name,
476                content,
477            });
478        }
479        files.sort_by(|a, b| a.name.cmp(&b.name));
480        Ok(files)
481    }

483    /// Recursive helper to load .cedar files from a directory and its subdirectories.
484    fn load_cedar_files_recursive(
485        &self,
486        dir: &str,
487        files: &mut Vec<StoreFile>,
488        depth: usize,
489    ) -> Result<(), PolicyStoreError> {
490        if depth > Self::MAX_RECURSION_DEPTH {
491            return Err(PolicyStoreError::MaxDepthExceeded {
492                path: dir.to_string(),
493                max_depth: Self::MAX_RECURSION_DEPTH,
494            });
495        }
496
497        let entries =
498            self.vfs
499                .read_dir(dir)
500                .map_err(|source| PolicyStoreError::DirectoryReadError {
501                    path: dir.to_string(),
502                    source,
503                })?;
504
505        for entry in entries {
506            if entry.is_dir {
507                self.load_cedar_files_recursive(&entry.path, files, depth + 1)?;
508            } else {
509                Self::check_extension(&entry, ".cedar")?;
510                let content = self.read_text(&entry.path)?;
511                files.push(StoreFile {
512                    name: entry.name,
513                    content,
514                });
515            }
516        }
517
518        Ok(())
519    }
