6//! In-memory migration between adjacent policy store formats.
7//!
8//! Each `vN_to_vM` step implements [`Upgrade`] for the older version's doc and
9//! depends only on its two version modules. [`IntoCurrent`] chains the steps,
10//! so every supported version reaches the current one without a hand-written
11//! path per version, and a missing step fails to compile where that version
12//! is dispatched.
13
14pub(crate) mod v0_to_v1;
15pub(crate) mod v1_to_v2;
16
17use std::convert::Infallible;
18
19use crate::common::policy_store::formats::{PolicyStoreWarning, current};
20
21/// The doc of the current spec version, the only one that converts into
22/// runtime types.
23pub(super) type CurrentDoc = current::PolicyStoreDoc;
24
25/// A store that cannot be expressed in a later spec version, by the step that
26/// failed. Each step owns its error type, so dropping a version removes its
27/// step module and its variant here. The message names the step, so a store
28/// migrated through several of them shows where it stopped.
29#[derive(Debug, thiserror::Error)]
30pub enum MigrationError {
31    #[error("spec version 1 to 2: {0}")]
32    V1ToV2(#[from] v1_to_v2::V1ToV2Error),
33}
34
35/// A step that cannot fail converts with no variant here.
36impl From<Infallible> for MigrationError {
37    fn from(never: Infallible) -> Self {
38        match never {}
39    }
40}
41
42/// One migration step, from a spec version's doc to the next version's.
43///
44/// Written by hand for each step: how fields map between two versions is a
45/// decision, not something to derive. A step reports what it had to change in
46/// behavior through `warnings`, and fails when the store cannot be expressed.
47/// Both use the step's own types, so they leave with the step.
48pub(super) trait Upgrade {
49    type Next;
50    type Error: Into<MigrationError>;
51    type Warning: Into<PolicyStoreWarning>;
52
53    fn upgrade(self, warnings: &mut Vec<Self::Warning>) -> Result<Self::Next, Self::Error>;
54}
55
56/// Brings the doc of any supported spec version up to [`CurrentDoc`] by
57/// applying each [`Upgrade`] step in turn.
58pub(super) trait IntoCurrent {
59    fn into_current(
60        self,
61        warnings: &mut Vec<PolicyStoreWarning>,
62    ) -> Result<CurrentDoc, MigrationError>;
63}
64
65/// The chain ends at the current version.
66///
67/// This impl and the blanket one below would overlap if [`CurrentDoc`] ever
68/// implemented [`Upgrade`]. It cannot: the current version is the newest, so
69/// there is no version to upgrade it to. Each bump moves both the `Upgrade`
70/// impl and this one forward together.
71impl IntoCurrent for CurrentDoc {
72    fn into_current(
73        self,
74        _warnings: &mut Vec<PolicyStoreWarning>,
75    ) -> Result<CurrentDoc, MigrationError> {
76        Ok(self)
77    }
78}
79
80impl<T> IntoCurrent for T
81where
82    T: Upgrade,
83    T::Next: IntoCurrent,
84{
85    fn into_current(
86        self,
87        warnings: &mut Vec<PolicyStoreWarning>,
88    ) -> Result<CurrentDoc, MigrationError> {
89        let mut step_warnings = Vec::new();
90        let next = self.upgrade(&mut step_warnings).map_err(Into::into)?;
91        warnings.extend(step_warnings.into_iter().map(Into::into));
92        next.into_current(warnings)
93    }
94}
