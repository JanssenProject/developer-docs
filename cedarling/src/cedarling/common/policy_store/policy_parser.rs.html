6//! Cedar policy and template parsing and validation.
7//!
8//! This module handles parsing Cedar policy files (.cedar) and extracting
9//! policy IDs from `@id()` annotations. It provides validation and error
10//! reporting with file names and line numbers.
11//!
12//! A file may contain multiple policies. Single-policy files use the first
13//! `@id()` annotation or the filename (without extension) as the policy id.
14//! When a file contains more than one policy, each policy must use Cedar's
15//! `@id("...")` policy annotation before `permit` / `forbid`; filename-based ids
16//! are not applied in that case. Policies are renamed so their [`PolicyId`] matches
17//! the `@id` value (Cedar 4 defaults internal ids to `policy0`, `policy1`, ...).
18
19use cedar_policy::{Policy, PolicyId, PolicySet, Template};
20use std::collections::HashMap;
21use std::str::FromStr;
22
23use super::errors::{CedarParseErrorDetail, PolicyStoreError, ValidationError};

25/// Represents a parsed Cedar policy with metadata.
26#[derive(Debug, Clone)]
27pub(super) struct ParsedPolicy {
28    /// The policy ID (from Cedar engine or @id annotation)
29    pub id: PolicyId,
30    /// The original filename
31    pub filename: String,
32    /// The parsed Cedar policy
33    pub policy: Policy,
34}

36/// Represents a parsed Cedar template with metadata.
37#[derive(Debug, Clone)]
38pub(super) struct ParsedTemplate {
39    /// The original filename
40    pub filename: String,
41    /// The parsed Cedar template
42    pub template: Template,
43}

45/// Cedar policy and template parser.
46///
47/// Provides methods for parsing Cedar policies and templates from text,
48/// extracting `@id()` annotations, and validating syntax.
49pub(super) struct PolicyParser;

51impl PolicyParser {
52    /// Parse Cedar policy text from one file.
53    ///
54    /// Returns one [`ParsedPolicy`] for single-policy files, or one entry per policy
55    /// when the file contains multiple policies (see module docs).
56    ///
57    /// The policy ID for single-policy files is determined by:
58    /// 1. Extracting from `@id()` annotation in the policy text, OR
59    /// 2. Deriving from the filename (without .cedar extension)
60    ///
61    /// Multi-policy files are parsed with [`PolicySet`]; each policy must have a
62    /// Cedar `@id("...")` annotation, then policies are given [`PolicyId`]s equal to
63    /// those annotation values.
64    pub(super) fn parse_policy(
65        content: &str,
66        filename: &str,
67    ) -> Result<Vec<ParsedPolicy>, PolicyStoreError> {
68        let policy_set =
69            PolicySet::from_str(content).map_err(|e| PolicyStoreError::CedarParsing {
70                file: filename.to_string(),
71                detail: CedarParseErrorDetail::ParseError(e.to_string()),
72            })?;
73
74        // Reject policy files that also contain templates (slots like `?principal`).
75        // Cedar parses them into a separate `templates()` collection, so iterating only
76        // `policies()` would silently drop them.
77        let template_count = policy_set.templates().count();
78        if template_count > 0 {
79            return Err(PolicyStoreError::CedarParsing {
80                file: filename.to_string(),
81                detail: CedarParseErrorDetail::TemplatesInPolicyFile {
82                    count: template_count,
83                },
84            });
85        }
86
87        let policies: Vec<Policy> = policy_set.policies().cloned().collect();
88
89        match policies.len() {
90            0 => Err(PolicyStoreError::CedarParsing {
91                file: filename.to_string(),
92                detail: CedarParseErrorDetail::ParseError("no policies found in file".to_string()),
93            }),
94            1 => {
95                let policy = policies.into_iter().next().expect("len == 1 checked above");
96                // Prefer Cedar's `@id("...")` annotation (same as multi-policy branch);
97                // fall back to scanning `@id(...)` in comments, then to the filename.
98                let id_str = policy
99                    .annotation("id")
100                    .map(str::to_owned)
101                    .or_else(|| Self::extract_id_annotation(content))
102                    .or_else(|| Self::derive_id_from_filename(filename))
103                    .ok_or_else(|| PolicyStoreError::CedarParsing {
104                        file: filename.to_string(),
105                        detail: CedarParseErrorDetail::MissingIdAnnotation,
106                    })?;
107                Self::validate_policy_id(&id_str, filename)
108                    .map_err(PolicyStoreError::Validation)?;
109                let policy = policy.new_id(PolicyId::new(&id_str));
110                Ok(vec![ParsedPolicy {
111                    id: policy.id().clone(),
112                    filename: filename.to_string(),
113                    policy,
114                }])
115            },
116            n => {
117                let mut out = Vec::with_capacity(n);
118                let mut seen_ids = std::collections::HashSet::with_capacity(n);
119                for policy in policies {
120                    // In Cedar 4+, `@id("...")` is a policy annotation named `id`; the AST
121                    // `PolicyId` defaults to `policy0`, `policy1`, ... until renamed.
122                    let id_str = policy.annotation("id").map(str::to_owned).ok_or_else(|| {
123                        // Cedar's `policies()` iteration order is not guaranteed,
124                        // so `policy.id()` here (e.g. `policy0`) may not match file
125                        // order. Scan the source text to point at the first policy
126                        // start that has no preceding `@id(...)`.
127                        let (line, snippet) = Self::find_first_policy_without_id(content)
128                            .unwrap_or((0, String::new()));
129                        PolicyStoreError::CedarParsing {
130                            file: filename.to_string(),
131                            detail: CedarParseErrorDetail::MultiPolicyMissingExplicitId {
132                                line,
133                                snippet,
134                            },
135                        }
136                    })?;
137                    Self::validate_policy_id(&id_str, filename)
138                        .map_err(PolicyStoreError::Validation)?;
139                    if !seen_ids.insert(id_str.clone()) {
140                        return Err(PolicyStoreError::CedarParsing {
141                            file: filename.to_string(),
142                            detail: CedarParseErrorDetail::DuplicatePolicyIdInFile { id: id_str },
143                        });
144                    }
145                    let policy = policy.new_id(PolicyId::new(&id_str));
146                    out.push(ParsedPolicy {
147                        id: policy.id().clone(),
148                        filename: filename.to_string(),
149                        policy,
150                    });
151                }
152                Ok(out)
153            },
154        }
155    }

157    /// Parse Cedar template text from one file.
158    ///
159    /// Symmetric to [`parse_policy`](Self::parse_policy):
160    /// returns one [`ParsedTemplate`] for single-template files, or one entry per
161    /// template when the file contains multiple. Templates use slots (e.g. `?principal`).
162    ///
163    /// Single-template files use the `@id("...")` annotation, then the comment-style
164    /// `@id(...)`, then the filename (without `.cedar`) as the template id.
165    /// Multi-template files require a Cedar `@id("...")` annotation on each template.
166    /// Files that mix templates with non-template policies are rejected.
167    pub(super) fn parse_template(
168        content: &str,
169        filename: &str,
170    ) -> Result<Vec<ParsedTemplate>, PolicyStoreError> {
171        let policy_set =
172            PolicySet::from_str(content).map_err(|e| PolicyStoreError::CedarParsing {
173                file: filename.to_string(),
174                detail: CedarParseErrorDetail::ParseError(e.to_string()),
175            })?;
176
177        // Reject template files that also contain non-template policies.
178        let policy_count = policy_set.policies().count();
179        if policy_count > 0 {
180            return Err(PolicyStoreError::CedarParsing {
181                file: filename.to_string(),
182                detail: CedarParseErrorDetail::PoliciesInTemplateFile {
183                    count: policy_count,
184                },
185            });
186        }
187
188        let templates: Vec<Template> = policy_set.templates().cloned().collect();
189
190        match templates.len() {
191            0 => Err(PolicyStoreError::CedarParsing {
192                file: filename.to_string(),
193                detail: CedarParseErrorDetail::ParseError("no templates found in file".to_string()),
194            }),
195            1 => {
196                let template = templates
197                    .into_iter()
198                    .next()
199                    .expect("len == 1 checked above");
200                let id_str = template
201                    .annotation("id")
202                    .filter(|s| !s.is_empty())
203                    .map(str::to_owned)
204                    .or_else(|| Self::extract_id_annotation(content))
205                    .or_else(|| Self::derive_id_from_filename(filename))
206                    .ok_or_else(|| PolicyStoreError::CedarParsing {
207                        file: filename.to_string(),
208                        detail: CedarParseErrorDetail::MissingIdAnnotation,
209                    })?;
210                Self::validate_policy_id(&id_str, filename)
211                    .map_err(PolicyStoreError::Validation)?;
212                let template = template.new_id(PolicyId::new(&id_str));
213                Ok(vec![ParsedTemplate {
214                    filename: filename.to_string(),
215                    template,
216                }])
217            },
218            n => {
219                let mut out = Vec::with_capacity(n);
220                let mut seen_ids = std::collections::HashSet::with_capacity(n);
221                for template in templates {
222                    let id_str = template
223                        .annotation("id")
224                        .filter(|s| !s.is_empty())
225                        .map(str::to_owned)
226                        .ok_or_else(|| {
227                            let (line, snippet) = Self::find_first_policy_without_id(content)
228                                .unwrap_or((0, String::new()));
229                            PolicyStoreError::CedarParsing {
230                                file: filename.to_string(),
231                                detail: CedarParseErrorDetail::MultiTemplateMissingExplicitId {
232                                    line,
233                                    snippet,
234                                },
235                            }
236                        })?;
237                    Self::validate_policy_id(&id_str, filename)
238                        .map_err(PolicyStoreError::Validation)?;
239                    if !seen_ids.insert(id_str.clone()) {
240                        return Err(PolicyStoreError::CedarParsing {
241                            file: filename.to_string(),
242                            detail: CedarParseErrorDetail::DuplicateTemplateIdInFile { id: id_str },
243                        });
244                    }
245                    let template = template.new_id(PolicyId::new(&id_str));
246                    out.push(ParsedTemplate {
247                        filename: filename.to_string(),
248                        template,
249                    });
250                }
251                Ok(out)
252            },
253        }
254    }

256    /// Parse multiple policies and return a map of policy ID to filename.
257    ///
258    /// Useful for batch processing of policy files in tests and tooling.
259    #[cfg(test)]
260    pub(super) fn parse_policies<'a, I>(
261        policy_files: I,
262    ) -> Result<HashMap<PolicyId, String>, PolicyStoreError>
263    where
264        I: IntoIterator<Item = (&'a str, &'a str)>,
265    {
266        let policy_files_vec: Vec<_> = policy_files.into_iter().collect();
267        let mut policy_map = HashMap::with_capacity(policy_files_vec.len());
268
269        for (filename, content) in policy_files_vec {
270            for parsed in Self::parse_policy(content, filename)? {
271                policy_map.insert(parsed.id, parsed.filename);
272            }
273        }
274
275        Ok(policy_map)
276    }

278    /// Create a [`PolicySet`] from parsed policies and templates.
279    ///
280    /// Validates that all policies and templates can be successfully added
281    /// to the policy set, ensuring no ID conflicts or other issues.
282    pub(super) fn create_policy_set(
283        policies: Vec<ParsedPolicy>,
284        templates: Vec<ParsedTemplate>,
285    ) -> Result<PolicySet, PolicyStoreError> {
286        // Cedar's PolicySet namespaces policies and templates together. Check
287        // cross-file duplicates up front so the error names both offending files
288        // instead of surfacing a generic Cedar "duplicate id" on add().
289        let mut seen: HashMap<PolicyId, String> =
290            HashMap::with_capacity(policies.len() + templates.len());
291        let check = |id: &PolicyId,
292                     filename: &str,
293                     seen: &mut HashMap<PolicyId, String>|
294         -> Result<(), PolicyStoreError> {
295            if let Some(first) = seen.get(id) {
296                if first != filename {
297                    return Err(PolicyStoreError::CedarParsing {
298                        file: filename.to_string(),
299                        detail: CedarParseErrorDetail::DuplicatePolicyIdAcrossFiles {
300                            id: id.to_string(),
301                            first_file: first.clone(),
302                            second_file: filename.to_string(),
303                        },
304                    });
305                }
306            } else {
307                seen.insert(id.clone(), filename.to_string());
308            }
309            Ok(())
310        };
311        for parsed in &policies {
312            check(&parsed.id, &parsed.filename, &mut seen)?;
313        }
314        for parsed in &templates {
315            check(parsed.template.id(), &parsed.filename, &mut seen)?;
316        }
317
318        let mut policy_set = PolicySet::new();
319
320        for parsed in policies {
321            policy_set
322                .add(parsed.policy)
323                .map_err(|e| PolicyStoreError::CedarParsing {
324                    file: parsed.filename,
325                    detail: CedarParseErrorDetail::AddPolicyFailed(e.to_string()),
326                })?;
327        }
328
329        for parsed in templates {
330            policy_set.add_template(parsed.template).map_err(|e| {
331                PolicyStoreError::CedarParsing {
332                    file: parsed.filename,
333                    detail: CedarParseErrorDetail::AddTemplateFailed(e.to_string()),
334                }
335            })?;
336        }
337
338        Ok(policy_set)
339    }

341    /// Derive a policy ID from a filename.
342    ///
343    /// Removes the .cedar extension, sanitizes characters, and returns the ID.
344    /// Returns None if the filename is empty or invalid.
345    fn derive_id_from_filename(filename: &str) -> Option<String> {
346        // Extract just the filename without path
347        let base_name = filename.rsplit('/').next().unwrap_or(filename);
348
349        // Remove .cedar extension
350        let without_ext = base_name.strip_suffix(".cedar").unwrap_or(base_name);
351
352        // If empty after stripping, return None
353        if without_ext.is_empty() {
354            return None;
355        }
356
357        // Replace invalid characters with underscores
358        let sanitized: String = without_ext
359            .chars()
360            .map(|c| {
361                if c.is_alphanumeric() || c == '_' || c == '-' || c == ':' {
362                    c
363                } else {
364                    '_'
365                }
366            })
367            .collect();
368
369        Some(sanitized)
370    }

372    /// Extract the first `@id()` annotation from Cedar policy text.
373    ///
374    /// Looks for `@id("...")` or `@id('...')` on each line.
375    fn extract_id_annotation(content: &str) -> Option<String> {
376        Self::collect_id_annotations(content).into_iter().next()
377    }

379    /// Collect every `@id("...")` / `@id('...')` value in file order.
380    fn collect_id_annotations(content: &str) -> Vec<String> {
381        let mut ids = Vec::new();
382        for line in content.lines() {
383            if let Some(id) = Self::parse_id_from_line_trimmed(line.trim()) {
384                ids.push(id);
385            }
386        }
387        ids
388    }
389
390    fn parse_id_from_line_trimmed(trimmed: &str) -> Option<String> {
391        let start_idx = trimmed.find("@id(")?;
392        let after_id = &trimmed[start_idx + 4..];
393        // Pair the byte offset with the quote we matched on. `str::find` returns a
394        // byte offset, not a char index, so recovering the quote via
395        // `chars().nth(open_quote)` would pick the wrong character once any
396        // multi-byte UTF-8 appears before the quote (e.g. `@id(🙂"x")`).
397        // Prefer whichever opening quote appears first so a later `"` (e.g. in
398        // `User::"alice"`) does not steal parsing from an earlier `'...'`.
399        let dq = after_id.find('"');
400        let sq = after_id.find('\'');
401        let (open_quote, quote_char) = match (dq, sq) {
402            (Some(i), Some(j)) => {
403                if i < j {
404                    (i, '"')
405                } else {
406                    (j, '\'')
407                }
408            },
409            (Some(i), None) => (i, '"'),
410            (None, Some(j)) => (j, '\''),
411            (None, None) => return None,
412        };
413        let after_open = &after_id[open_quote + 1..];
414        let close_quote = after_open.find(quote_char)?;
415        Some(after_open[..close_quote].to_string())
416    }

418    /// Byte offset in `trimmed` of the first `permit` / `forbid` policy keyword,
419    /// or `None` if this line does not contain a policy statement start.
420    fn policy_keyword_byte_offset(trimmed: &str) -> Option<usize> {
421        const KEYWORDS: [&str; 4] = ["permit(", "forbid(", "permit ", "forbid "];
422        KEYWORDS.iter().filter_map(|kw| trimmed.find(kw)).min()
423    }
424
425    /// Scan source text for the first `permit` / `forbid` policy whose preceding
426    /// lines (back to the start of the file or the previous policy) contain no
427    /// `@id(...)` annotation. Returns `(1-based line, trimmed-and-truncated snippet)`.
428    ///
429    /// Treats a line as containing a policy start if any of `permit(` / `forbid(` /
430    /// `permit ` / `forbid ` appears in the line after leading whitespace (e.g.
431    /// `@id("p1") permit(...)` on one line). The snippet starts at that keyword.
432    fn find_first_policy_without_id(content: &str) -> Option<(usize, String)> {
433        const MAX_SNIPPET: usize = 80;
434        let lines: Vec<&str> = content.lines().collect();
435        let mut window_start = 0usize;
436        for (idx, line) in lines.iter().enumerate() {
437            let trimmed = line.trim_start();
438            let Some(token_off) = Self::policy_keyword_byte_offset(trimmed) else {
439                continue;
440            };
441            let has_id_on_prior_lines = lines[window_start..idx]
442                .iter()
443                .any(|l| Self::parse_id_from_line_trimmed(l.trim()).is_some());
444            let has_id_on_same_line_before_kw = if token_off == 0 {
445                false
446            } else {
447                Self::parse_id_from_line_trimmed(trimmed.get(..token_off).unwrap_or("").trim_end())
448                    .is_some()
449            };
450            let has_id = has_id_on_prior_lines || has_id_on_same_line_before_kw;
451            if !has_id {
452                let mut snippet = trimmed[token_off..].trim_end().to_string();
453                if snippet.chars().count() > MAX_SNIPPET {
454                    snippet = snippet.chars().take(MAX_SNIPPET).collect::<String>() + "...";
455                }
456                return Some((idx + 1, snippet));
457            }
458            window_start = idx + 1;
459        }
460        None
461    }

463    /// Validate policy ID format (alphanumeric, underscore, hyphen, colon only).
464    fn validate_policy_id(id: &str, filename: &str) -> Result<(), ValidationError> {
465        if id.is_empty() {
466            return Err(ValidationError::EmptyPolicyId {
467                file: filename.to_string(),
468            });
469        }
470
471        // Check for valid characters (alphanumeric, underscore, hyphen, colon)
472        if !id
473            .chars()
474            .all(|c| c.is_alphanumeric() || c == '_' || c == '-' || c == ':')
475        {
476            return Err(ValidationError::InvalidPolicyIdCharacters {
477                file: filename.to_string(),
478                id: id.to_string(),
479            });
480        }
481
482        Ok(())
483    }
