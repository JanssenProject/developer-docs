6//! RFC 3339 timestamp deserialization shared by every format version.
7//!
8//! A stateless codec rather than part of any version's shape, so versions share
9//! it instead of each freezing a copy.
10
11use chrono::{DateTime, Utc};
12use serde::{Deserialize, Deserializer};
13
14/// Deserializes an optional RFC 3339 timestamp.
15pub(super) fn deserialize_option<'de, D>(deserializer: D) -> Result<Option<DateTime<Utc>>, D::Error>
16where
17    D: Deserializer<'de>,
18{
19    let opt: Option<String> = Option::deserialize(deserializer)?;
20    match opt {
21        Some(s) => DateTime::parse_from_rfc3339(&s)
22            .map(|dt| Some(dt.with_timezone(&Utc)))
23            .map_err(serde::de::Error::custom),
24        None => Ok(None),
25    }
26}
27
28#[cfg(test)]
29mod tests {
30    use super::*;
31
32    #[derive(Debug, Deserialize)]
33    struct Holder {
34        #[serde(default, deserialize_with = "deserialize_option")]
35        at: Option<DateTime<Utc>>,
36    }
37
38    fn at(json: &str) -> Option<DateTime<Utc>> {
39        serde_json::from_str::<Holder>(json)
40            .expect("timestamp should parse")
41            .at
42    }
43
44    #[test]
45    fn reads_rfc3339_and_treats_absent_or_null_as_none() {
46        assert_eq!(
47            at(r#"{ "at": "2024-01-02T03:04:05Z" }"#),
48            Some(
49                DateTime::parse_from_rfc3339("2024-01-02T03:04:05Z")
50                    .expect("valid timestamp")
51                    .with_timezone(&Utc)
52            ),
53            "an RFC 3339 timestamp should be parsed into UTC"
54        );
55        assert_eq!(at("{}"), None, "an absent timestamp should read as None");
56        assert_eq!(
57            at(r#"{ "at": null }"#),
58            None,
59            "an explicit null should read as None"
60        );
61    }
62
63    #[test]
64    fn rejects_a_non_rfc3339_timestamp() {
65        serde_json::from_str::<Holder>(r#"{ "at": "01/02/2024" }"#)
66            .expect_err("a non-RFC-3339 timestamp must be rejected");
67    }
68}
