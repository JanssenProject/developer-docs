26pub(crate) const SPEC_VERSION: u32 = 1;
27
28// Where each part of a v1 store lives. Frozen with the version: a store laid
29// out differently is a new spec version, so this is never shared with or
30// edited for another version.
31const LAYOUT: Layout = Layout {
32    schema_file: "schema.cedarschema",
33    schema_dir: "schemas",
34    policies_dir: "policies",
35    templates_dir: "templates",
36    entities_dir: "entities",
37};
38
39const ISSUER_DIRS: IssuerDirs = IssuerDirs {
40    trusted_issuers: "trusted-issuers",
41    custom_issuers: "custom-issuers",
42};
43
44/// Reads a v1 store and parses its files; `metadata_json` has already been
45/// read by the caller to pick this version.
46pub(super) fn read(
47    fs: &StoreFs<'_>,
48    metadata_json: String,
49    strict: bool,
50) -> Result<PolicyStoreDoc, ReadStoreError> {
51    let files = super::read_files(fs, &LAYOUT, metadata_json, strict)?;
52    let issuers = super::read_issuer_dirs(fs, &ISSUER_DIRS)?;
53    Ok(parse(files, issuers)?)
54}
55
56/// A whole v1 store, parsed but not yet converted into runtime types.
57#[derive(Debug)]
58pub(crate) struct PolicyStoreDoc {
59    pub(crate) metadata: MetadataDoc,
60    pub(crate) trusted_issuers: Vec<ParsedIssuer>,
61    pub(crate) custom_issuers: Vec<ParsedCustomIssuer>,
62    pub(crate) content: StoreContent,
63}
64
65/// Parses the raw files of a v1 store.
66fn parse(files: Files, issuers: IssuerDirFiles) -> Result<PolicyStoreDoc, ParseStoreError> {
67    let Files {
68        metadata_json,
69        schema,
70        schema_source_exists,
71        policies,
72        templates,
73        entities,
74    } = files;
75    let IssuerDirFiles {
76        trusted_issuers,
77        custom_issuers,
78    } = issuers;
79
80    let metadata = metadata::parse(&metadata_json)?;
81
82    let mut parsed_issuers = Vec::with_capacity(trusted_issuers.len());
83    for file in &trusted_issuers {
84        let parsed = IssuerParser::parse_issuer(&file.content, &file.name).map_err(|e| {
85            ConversionError::TrustedIssuerParse {
86                file: file.name.clone(),
87                source: Box::new(e),
88            }
89        })?;
90        parsed_issuers.extend(parsed);
91    }
92
93    let parsed_custom_issuers = custom_issuers
94        .iter()
95        .map(|file| {
96            CustomIssuerParser::parse(&file.content, &file.name)
97                .map_err(ConversionError::CustomIssuerParse)
98        })
99        .collect::<Result<Vec<_>, _>>()?;
100
101    Ok(PolicyStoreDoc {
102        metadata,
103        trusted_issuers: parsed_issuers,
104        custom_issuers: parsed_custom_issuers,
105        content: StoreContent {
106            schema,
107            schema_source_exists,
108            policies,
109            templates,
110            entities,
111        },
112    })
113}

115#[cfg(test)]
116mod tests {
117    use super::*;
118    use crate::common::policy_store::errors::PolicyStoreError;
119    use crate::common::policy_store::loader::StoreFile;
120
121    const METADATA: &str = r#"{
122        "policy_store_spec_version": 1,
123        "cedar_version": "4.4.0",
124        "policy_store": { "id": "abc123def456", "name": "Golden v1", "version": "1.0.0" }
125    }"#;
126
127    fn issuer_file(name: &str, id: &str) -> StoreFile {
128        StoreFile {
129            name: name.to_string(),
130            content: format!(
131                r#"{{
132                    "id": "{id}",
133                    "name": "{id}",
134                    "openid_configuration_endpoint": "https://{id}.test/.well-known/openid-configuration",
135                    "token_metadata": {{
136                        "access_token": {{ "entity_type_name": "{id}::Access_token" }}
137                    }}
138                }}"#
139            ),
140        }
141    }
142
143    fn loaded(
144        trusted_issuers: Vec<StoreFile>,
145        custom_issuers: Vec<StoreFile>,
146    ) -> (Files, IssuerDirFiles) {
147        let files = Files {
148            metadata_json: METADATA.to_string(),
149            schema: None,
150            schema_source_exists: false,
151            policies: vec![StoreFile {
152                name: "allow.cedar".to_string(),
153                content: "permit(principal, action, resource);".to_string(),
154            }],
155            templates: vec![],
156            entities: vec![],
157        };
158        let issuers = IssuerDirFiles {
159            trusted_issuers,
160            custom_issuers,
161        };
162        (files, issuers)
163    }
164
165    /// Golden v1 store: every version-specific file parses with v1 semantics.
166    #[test]
167    fn golden_store_parses() {
168        let (files, issuers) = loaded(
169            vec![issuer_file("a.json", "Alpha")],
170            vec![StoreFile {
171                name: "acme.json".to_string(),
172                content:
173                    r#"{ "tokens_mappings": { "Acme::Custom": { "required_claims": ["sub"] } } }"#
174                        .to_string(),
175            }],
176        );
177        let doc = parse(files, issuers).expect("golden v1 store should parse");
178
179        assert_eq!(
180            doc.metadata.policy_store_spec_version, SPEC_VERSION,
181            "spec version should be read from metadata.json"
182        );
183        assert_eq!(
184            doc.trusted_issuers[0].issuer.token_metadata["access_token"].entity_type_name,
185            "Alpha::Access_token",
186            "trusted issuer token metadata should be parsed"
187        );
188        assert_eq!(
189            doc.custom_issuers[0].id, "acme",
190            "custom issuer id comes from the file"
191        );
192        assert!(
193            doc.custom_issuers[0].meta.tokens_mappings["Acme::Custom"]
194                .required_claims
195                .contains("sub"),
196            "custom issuer required_claims should be parsed"
197        );
198    }
199
200    #[test]
201    fn invalid_issuer_file_is_reported_with_file_name() {
202        let err = {
203            let (files, issuers) = loaded(
204                vec![StoreFile {
205                    name: "bad.json".to_string(),
206                    content: "{ not json".to_string(),
207                }],
208                vec![],
209            );
210            parse(files, issuers)
211        }
212        .expect_err("malformed issuer file must be rejected");
213        assert!(
214            matches!(
215                &err,
216                ParseStoreError::Conversion(ConversionError::TrustedIssuerParse { file, source })
217                    if file == "bad.json"
218                        && matches!(**source, PolicyStoreError::JsonParsing { .. })
219            ),
220            "expected TrustedIssuerParse naming the file, got: {err:?}"
221        );
222    }
