6use serde::Serialize;
7
8/// What happened to the TRACE record of one decision, as the decision log and the
9/// metrics report it. `Emitted` means handed to the emitter: whether Lock accepted
10/// the record shows in the emitter's own logs and metrics.
11#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
12#[serde(rename_all = "snake_case")]
13pub(crate) enum TraceRecordStatus {
14    /// Handed to the emitter.
15    Emitted,
16    /// The host passed no TRACE context, or left a mandatory field empty.
17    NotEmittedMissingContext,
18    /// A value is longer, or a list larger, than Lock accepts.
19    NotEmittedOverLimit,
20    /// A value Lock rejects, or a policy store that cannot back a record.
21    NotEmittedInvalidField,
22    /// The emitter queue is full.
23    NotEmittedQueueFull,
24    /// The emitter has stopped, as after `Cedarling::shut_down`.
25    NotEmittedStopped,
26    /// There is no TRACE endpoint to send the record to.
27    NotEmittedNoEndpoint,
28}
29
30impl TraceRecordStatus {
31    /// Every status, each with its own counter.
32    pub(crate) const ALL: [Self; 7] = [
33        Self::Emitted,
34        Self::NotEmittedMissingContext,
35        Self::NotEmittedOverLimit,
36        Self::NotEmittedInvalidField,
37        Self::NotEmittedQueueFull,
38        Self::NotEmittedStopped,
39        Self::NotEmittedNoEndpoint,
40    ];
41
42    /// Key of the status counter in the metrics `operational_stats`.
43    pub(crate) fn metric_key(self) -> &'static str {
44        match self {
45            Self::Emitted => "trace.emitted",
46            Self::NotEmittedMissingContext => "trace.not_emitted_missing_context",
47            Self::NotEmittedOverLimit => "trace.not_emitted_over_limit",
48            Self::NotEmittedInvalidField => "trace.not_emitted_invalid_field",
49            Self::NotEmittedQueueFull => "trace.not_emitted_queue_full",
50            Self::NotEmittedStopped => "trace.not_emitted_stopped",
51            Self::NotEmittedNoEndpoint => "trace.not_emitted_no_endpoint",
52        }
53    }
54}

56/// What the TRACE emitter did with records after the authorize path handed them
57/// over, counted in the metrics.
58#[derive(Debug, Clone, Copy, PartialEq, Eq)]
59pub(crate) enum EmitterEvent {
60    /// Lock accepted the record.
61    Accepted,
62    /// Lock refused the record for good, alone or with its whole request.
63    Rejected,
64    /// The record was given up after its retries, or discarded with its lost chain.
65    Dropped,
66    /// The draft could not be signed.
67    SignFailed,
68    /// A chain opened to replace one whose genesis record was lost or that
69    /// Lock no longer knows.
70    ChainOpened,
71}
72
73impl EmitterEvent {
74    /// Every event, each with its own counter.
75    pub(crate) const ALL: [Self; 5] = [
76        Self::Accepted,
77        Self::Rejected,
78        Self::Dropped,
79        Self::SignFailed,
80        Self::ChainOpened,
81    ];
82
83    /// Key of the event counter in the metrics `operational_stats`.
84    pub(crate) fn metric_key(self) -> &'static str {
85        match self {
86            Self::Accepted => "trace.accepted",
87            Self::Rejected => "trace.rejected",
88            Self::Dropped => "trace.dropped",
89            Self::SignFailed => "trace.sign_failed",
90            Self::ChainOpened => "trace.chains_opened",
91        }
92    }
