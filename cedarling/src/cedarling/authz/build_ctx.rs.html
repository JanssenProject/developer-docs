18/// Constructs the authorization context by dispatching to schema or no-schema implementation.
19///
20/// When a schema is available, entity references from tokens are injected into the context
21/// based on the action's context shape. Without a schema, only request context and pushed
22/// data are merged.
23pub(super) fn build_context(
24    config: &AuthzConfig,
25    request_context: Value,
26    build_entities: &BuiltEntities,
27    action: &cedar_policy::EntityUid,
28    pushed_data: &HashMap<String, Value>,
29) -> Result<cedar_policy::Context, BuildContextError> {
30    if config.policy_store.schema.is_some() {
31        build_context_with_schema(config, request_context, build_entities, action, pushed_data)
32    } else {
33        build_context_no_schema(request_context, pushed_data)
34    }
35}

37/// Constructs the authorization context using schema-based entity references.
38///
39/// Builds context by looking up the action's expected context shape in the Cedar schema
40/// and injecting matching entity references from the built entities.
41pub(super) fn build_context_with_schema(
42    config: &AuthzConfig,
43    request_context: Value,
44    build_entities: &BuiltEntities,
45    action: &cedar_policy::EntityUid,
46    pushed_data: &HashMap<String, Value>,
47) -> Result<cedar_policy::Context, BuildContextError> {
48    let store_schema = config.policy_store.schema.as_ref().ok_or_else(|| {
49        BuildContextError::ContextCreation(
50            "build_context_with_schema called but policy store has no schema".to_string(),
51        )
52    })?;
53    let json_schema = &store_schema.json;
54    let schema = &store_schema.schema;
55    let namespace = action.type_name().namespace();
56    let action_name = &action.id().escaped();
57
58    let action_schema = json_schema
59        .get_action(&namespace, action_name)
60        .ok_or(BuildContextError::UnknownAction(action_name.to_string()))?;
61
62    let mut ctx_entity_refs = json!({});
63
64    if let Some(ctx) = action_schema.applies_to.context.as_ref() {
65        match ctx {
66            Attribute::Record { attrs, .. } => {
67                for (key, attr) in attrs {
68                    if let Some(entity_ref) =
69                        build_entity_refs_from_attr(&namespace, attr, build_entities, json_schema)?
70                    {
71                        ctx_entity_refs[key] = entity_ref;
72                    }
73                }
74            },
75            Attribute::EntityOrCommon { name, .. } => {
76                if let Some((_entity_type_name, attr)) = json_schema
77                    .get_common_type(name, Some(&namespace))
78                    .map_err(|err| BuildContextError::ParseEntityName(name.clone(), err))?
79                {
80                    match attr {
81                        Attribute::Record { attrs, .. } => {
82                            for (key, attr) in attrs {
83                                if let Some(entity_ref) = build_entity_refs_from_attr(
84                                    &namespace,
85                                    attr,
86                                    build_entities,
87                                    json_schema,
88                                )? {
89                                    ctx_entity_refs[key] = entity_ref;
90                                }
91                            }
92                        },
93                        attr => {
94                            return Err(BuildContextError::InvalidKind(
95                                attr.kind_str().to_string(),
96                                "record".to_string(),
97                            ));
98                        },
99                    }
100                }
101            },
102            attr => {
103                return Err(BuildContextError::InvalidKind(
104                    attr.kind_str().to_string(),
105                    "record or common".to_string(),
106                ));
107            },
108        }
109    }
110
111    // Inject pushed data under context.data namespace, merging so key conflicts are surfaced.
112    if !pushed_data.is_empty() {
113        let data_value = Value::Object(
114            pushed_data
115                .iter()
116                .map(|(k, v)| (k.clone(), v.clone()))
117                .collect(),
118        );
119        let pushed_wrapper = json!({ "data": data_value });
120        ctx_entity_refs = merge_json_values(ctx_entity_refs, &pushed_wrapper)?;
121    }
122
123    let context = merge_json_values(request_context, &ctx_entity_refs)?;
124    let context: cedar_policy::Context =
125        cedar_policy::Context::from_json_value(context, Some((schema, action)))
126            .map_err(|e| BuildContextError::ContextCreation(e.to_string()))?;
127
128    Ok(context)
129}

131/// Constructs the authorization context without schema-based entity references.
132///
133/// Merges request context and pushed data directly. Entity references are not injected
134/// because the schema is unavailable.
135pub(super) fn build_context_no_schema(
136    request_context: Value,
137    pushed_data: &HashMap<String, Value>,
138) -> Result<cedar_policy::Context, BuildContextError> {
139    let mut context_json = request_context;
140    if !pushed_data.is_empty() {
141        let data_value = Value::Object(
142            pushed_data
143                .iter()
144                .map(|(k, v)| (k.clone(), v.clone()))
145                .collect(),
146        );
147        let pushed_wrapper = json!({ "data": data_value });
148        context_json = merge_json_values(context_json, &pushed_wrapper)?;
149    }
150
151    cedar_policy::Context::from_json_value(context_json, None)
152        .map_err(|e| BuildContextError::ContextCreation(e.to_string()))
153}

155/// Constructs the authorization context for multi-issuer requests with token collection
156///
157/// This function implements the design document's token collection context structure:
158/// - Individual token entities are accessible via `context.tokens.{issuer}_{token_type}`
159/// - All JWT claims are stored as entity tags (Set of String by default)
160/// - Provides ergonomic policy syntax for cross-token validation
161///
162/// ## Context Merging
163///
164/// Values are merged in the following order:
165/// 1. **Request Context** - Values passed directly in the authorization request
166/// 2. **Token References** - Built token entity references
167/// 3. **Pushed Data** - Data from the `DataStore` (injected under `context.data`)
168///
169/// **Important:** Merges will error on key conflicts via `merge_json_values`, which returns
170/// `BuildContextError::KeyConflict` when any key collision occurs. Callers must resolve
171/// conflicts before invoking this function.
172pub(super) fn build_multi_issuer_context(
173    request_context: Value,
174    token_entities: &HashMap<String, Entity>,
175    schema: Option<&cedar_policy::Schema>,
176    action: &cedar_policy::EntityUid,
177    pushed_data: &HashMap<String, Value>,
178) -> Result<cedar_policy::Context, BuildContextError> {
179    // Start with the request context
180    let mut context_json = request_context;
181
182    // Create the tokens context as specified in the design document
183    let mut tokens_context = serde_json::Map::new();
184
185    // Add individual token entity references to context
186    // Format: {issuer}_{token_type} -> {"type": "EntityType", "id": "entity_id"}
187    for (field_name, entity) in token_entities {
188        // Create entity reference like the existing build_context function
189        let entity_ref = json!({
190            "type": entity.uid().type_name().to_string(),
191            "id": entity.uid().id().as_ref() as &str
192        });
193        tokens_context.insert(field_name.clone(), entity_ref);
194    }
195
196    // Add total token count as specified in design
197    tokens_context.insert("total_token_count".to_string(), json!(token_entities.len()));
198
199    // Merge tokens into request context, checking for conflicts
200    let tokens_value = Value::Object(tokens_context);
201    let tokens_wrapper = json!({ "tokens": tokens_value });
202    context_json = merge_json_values(context_json, &tokens_wrapper)?;
203
204    // Inject pushed data under context.data namespace, merging so key conflicts are surfaced.
205    if !pushed_data.is_empty() {
206        let data_value = Value::Object(
207            pushed_data
208                .iter()
209                .map(|(k, v)| (k.clone(), v.clone()))
210                .collect(),
211        );
212        let pushed_wrapper = json!({ "data": data_value });
213        context_json = merge_json_values(context_json, &pushed_wrapper)?;
214    }
215
216    // Create Cedar context
217    let context = cedar_policy::Context::from_json_value(context_json, schema.map(|s| (s, action)))
218        .map_err(|e| BuildContextError::ContextCreation(e.to_string()))?;
219
220    Ok(context)
221}

223/// Builds the JSON entity references from a given attribute.
224///
225/// Returns `Ok(None)` if the attr is not an Entity Reference
226fn build_entity_refs_from_attr(
227    namespace: &str,
228    attr: &Attribute,
229    built_entities: &BuiltEntities,
230    schema: &CedarSchemaJson,
231) -> Result<Option<Value>, BuildContextError> {
232    match attr {
233        Attribute::Entity { name, required } => {
234            let Some(type_name) = schema
235                .get_entity_type_name(name, Some(namespace))
236                .map_err(|e| BuildContextError::ParseEntityName(name.clone(), e))?
237            else {
238                return Ok(None);
239            };
240
241            match map_entity_id(&type_name, built_entities) {
242                Ok(res) => Ok(res),
243                Err(err) => {
244                    if *required {
245                        Err(err)
246                    } else {
247                        Ok(None)
248                    }
249                },
250            }
251        },
252        Attribute::EntityOrCommon { name, required } => {
253            let Some(type_name) = schema
254                .get_entity_type_name(name, Some(namespace))
255                .map_err(|e| BuildContextError::ParseEntityName(name.clone(), e))?
256            else {
257                return Ok(None);
258            };
259
260            match map_entity_id(&type_name, built_entities) {
261                Ok(res) => Ok(res),
262                Err(err) => {
263                    if *required {
264                        Err(err)
265                    } else {
266                        Ok(None)
267                    }
268                },
269            }
270        },
271        _ => Ok(None),
272    }
273}

275/// Maps a known entity ID to the entity reference
276fn map_entity_id(
277    name: &EntityTypeName,
278    built_entities: &BuiltEntities,
279) -> Result<Option<Value>, BuildContextError> {
280    if let Some(type_id) = built_entities.get_single(&name.to_smolstr()) {
281        Ok(Some(json!({"type": name.to_string(), "id": type_id})))
282    } else {
283        Err(BuildContextError::MissingEntityId(name.to_string()))
284    }
285}

287fn merge_json_values(mut base: Value, other: &Value) -> Result<Value, BuildContextError> {
288    if let (Some(base_map), Some(additional_map)) = (base.as_object_mut(), other.as_object()) {
289        for (key, value) in additional_map {
290            if let Entry::Vacant(entry) = base_map.entry(key) {
291                entry.insert(value.clone());
292            } else {
293                return Err(BuildContextError::KeyConflict(key.clone()));
294            }
295        }
296    }
297    Ok(base)
298}

307    fn can_merge_json_objects() {
308        let obj1 = json!({ "a": 1, "b": 2 });
309        let obj2 = json!({ "c": 3, "d": 4 });
310        let expected = json!({"a": 1, "b": 2, "c": 3, "d": 4});
311
312        let result = merge_json_values(obj1, &obj2).expect("Should merge JSON objects");
313
314        assert_eq!(result, expected);
315    }

322
323        assert!(
324            matches!(result, Err(BuildContextError::KeyConflict(key)) if key.as_str() == "b"),
