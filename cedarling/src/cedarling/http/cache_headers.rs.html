6//! Parses HTTP cache validation headers used by the policy-store refresh worker.
7//!
8//! Recognized headers (RFC 7234 / RFC 9111):
9//! - `Cache-Control` — `max-age=N`, `no-cache`, `no-store` (both directives
10//!   force the refresh worker's in-memory freshness window to zero so the
11//!   next tick revalidates against the upstream)
12//! - `Expires` — RFC 7231 IMF-fixdate
13//! - `ETag` — opaque validator quoted-string
14//! - `Last-Modified` — RFC 7231 IMF-fixdate (stored verbatim for echoing back)
15//!
16//! All malformed values are treated as absent. The worker never panics on a
17//! surprising header value — it falls back to its configured interval.
18
19use chrono::{DateTime, Utc};
20use reqwest::header::HeaderMap;
21use std::time::Duration;
22
23/// Cache state extracted from a single HTTP response.
24#[derive(Debug, Clone, Default, PartialEq, Eq)]
25pub(crate) struct CacheHeadersState {
26    /// Strong or weak `ETag`, including the quote marks and any `W/` prefix.
27    pub etag: Option<String>,
28    /// `Last-Modified` value as the server sent it. Stored verbatim for echoing
29    /// in a subsequent `If-Modified-Since` request.
30    pub last_modified: Option<String>,
31    /// How long the response is fresh according to the server, after the
32    /// `Date` of the response. Derived from `Cache-Control: max-age` (preferred)
33    /// or `Expires` minus the current time.
34    pub fresh_for: Option<Duration>,
35    /// `Cache-Control: no-cache` was present — caller should always revalidate.
36    pub no_cache: bool,
37}
38
39impl CacheHeadersState {
40    /// Parse a [`HeaderMap`] into [`CacheHeadersState`].
41    ///
42    /// `now` is injected so callers can test deterministically. In production
43    /// pass [`Utc::now()`].
44    pub(crate) fn from_headers(headers: &HeaderMap, now: DateTime<Utc>) -> Self {
45        let etag = headers
46            .get(reqwest::header::ETAG)
47            .and_then(|v| v.to_str().ok())
48            .map(str::trim)
49            .filter(|s| !s.is_empty())
50            .map(str::to_owned);
51
52        let last_modified = headers
53            .get(reqwest::header::LAST_MODIFIED)
54            .and_then(|v| v.to_str().ok())
55            .map(str::trim)
56            .filter(|s| !s.is_empty())
57            .map(str::to_owned);
58
59        // RFC 9110 §5.3: a recipient MUST handle multiple field-values for the
60        // same header (sent either as one comma-separated line or as separate
61        // header lines). Fold across all `Cache-Control` field-values so a
62        // directive on the second line still gets honored — `headers.get(...)`
63        // alone would silently drop them.
64        let (cc_max_age, no_cache) = headers.get_all(reqwest::header::CACHE_CONTROL).iter().fold(
65            (None::<u64>, false),
66            |(prev_max_age, prev_no_cache), v| {
67                let Some(s) = v.to_str().ok() else {
68                    return (prev_max_age, prev_no_cache);
69                };
70                let (this_max_age, this_no_cache) = parse_cache_control(s);
71                // `no-cache` is sticky — any field-value carrying it wins.
72                // For `max-age`, the most-restrictive (smallest) value wins:
73                // RFC 9111 §5.2.1.1 lets caches use any value but the
74                // safer choice is the minimum.
75                let merged_max_age = match (prev_max_age, this_max_age) {
76                    (Some(a), Some(b)) => Some(a.min(b)),
77                    (Some(a), None) => Some(a),
78                    (None, b) => b,
79                };
80                (merged_max_age, prev_no_cache || this_no_cache)
81            },
82        );
83
84        // Cache-Control: max-age wins over Expires when both are present.
85        let mut fresh_for = if let Some(secs) = cc_max_age {
86            Some(Duration::from_secs(secs))
87        } else if let Some(expires_at) = headers
88            .get(reqwest::header::EXPIRES)
89            .and_then(|v| v.to_str().ok())
90            .and_then(parse_http_date)
91        {
92            // RFC 9111 §5.3: freshness lifetime from `Expires` is computed
93            // against the response's `Date` header — not the client's wall
94            // clock — so clock skew between client and origin doesn't
95            // (mis)classify a fresh response as stale or vice versa. Fall back
96            // to `now` only when the server omitted `Date` (allowed by RFC
97            // 7231 §7.1.1.2 for origins with no reliable clock).
98            let reference_time = headers
99                .get(reqwest::header::DATE)
100                .and_then(|v| v.to_str().ok())
101                .and_then(parse_http_date)
102                .unwrap_or(now);
103            let delta = expires_at.signed_duration_since(reference_time);
104            // Negative or zero ⇒ already stale, treat as fresh_for=0 so the
105            // worker will revalidate immediately on its next tick.
106            Some(if delta.num_seconds() > 0 {
107                Duration::from_secs(delta.num_seconds().cast_unsigned())
108            } else {
109                Duration::ZERO
110            })
111        } else {
112            None
113        };
114
115        // no-cache forces revalidation regardless of any max-age value.
116        if no_cache {
117            fresh_for = Some(Duration::ZERO);
118        }
119
120        Self {
121            etag,
122            last_modified,
123            fresh_for,
124            no_cache,
125        }
126    }
127
128    /// True if any conditional-request validator (`ETag` or `Last-Modified`) is
129    /// present. Used by the HEAD-then-GET refresh strategy to decide whether
130    /// HEAD added enough information to skip the body fetch.
131    pub(crate) fn has_validator(&self) -> bool {
132        self.etag.is_some() || self.last_modified.is_some()
133    }
134
135    /// Apply cache headers from a fresh response (typically a `304`) to
136    /// `self`. Field semantics differ on purpose:
137    ///
138    /// - **Validators** (`etag`, `last_modified`) identify the *resource
139    ///   version*. Per RFC 9110 §15.4.5 a `304` only carries them when
140    ///   refreshed; an omitted field must not erase our cached value, or the
141    ///   next conditional GET would have nothing to send.
142    ///
143    /// - **`fresh_for`** is a *per-response* property — "this specific
144    ///   response is fresh for N seconds". A new response that omits
145    ///   `Cache-Control` / `Expires` is making no freshness claim, and
146    ///   carrying forward an old value would pin the worker cadence to a
147    ///   long-ago `max-age` indefinitely. We always overwrite, including
148    ///   with `None`.
149    ///
150    /// - **`no_cache`** is a sticky directive: once a response asks for
151    ///   revalidation, a later response that omits the directive does not
152    ///   automatically unset it.
153    pub(crate) fn merge_from(&mut self, other: Self) {
154        if other.etag.is_some() {
155            self.etag = other.etag;
156        }
157        if other.last_modified.is_some() {
158            self.last_modified = other.last_modified;
159        }
160        // Per-response, not per-resource — overwrite unconditionally (a `None`
161        // means "no freshness claim this time", which the worker should honor
162        // rather than pretending an earlier max-age still applies).
163        self.fresh_for = other.fresh_for;
164        if other.no_cache {
165            self.no_cache = true;
166        }
167    }
168}
169
170/// Parse a `Cache-Control` header value into (`max_age`, `no_cache`).
171/// Tolerant: malformed directives are skipped. `no-store` is folded into the
172/// `no_cache` flag: although the directive's RFC 9111 meaning is "don't
173/// persist," a server emitting it is explicitly asking caches to revalidate
174/// instead of using stored values — so we treat the in-memory freshness
175/// window as zero and let the next tick re-check.
176fn parse_cache_control(value: &str) -> (Option<u64>, bool) {
177    let mut max_age: Option<u64> = None;
178    let mut no_cache = false;
179
180    for raw in value.split(',') {
181        let directive = raw.trim();
182        if directive.is_empty() {
183            continue;
184        }
185        // Directives are case-insensitive per RFC 9111
186        let lower = directive.to_ascii_lowercase();
187        if lower == "no-cache" || lower == "no-store" {
188            no_cache = true;
189        } else if let Some(rest) = lower.strip_prefix("max-age=") {
190            // Strip optional quotes; ignore unparseable / negative values.
191            let trimmed = rest.trim().trim_matches('"');
192            if let Ok(secs) = trimmed.parse::<u64>() {
193                max_age = Some(secs);
194            }
195        }
196    }
197
198    (max_age, no_cache)
199}

201/// Parse an HTTP date (RFC 7231 IMF-fixdate, RFC 850, or ANSI C `asctime`).
202/// Returns `None` on any parse failure.
203fn parse_http_date(value: &str) -> Option<DateTime<Utc>> {
204    let trimmed = value.trim();
205    // Try in order: RFC 2822 (close to IMF-fixdate), RFC 3339, then a couple of
206    // explicit format strings for the alternates.
207    if let Ok(dt) = DateTime::parse_from_rfc2822(trimmed) {
208        return Some(dt.with_timezone(&Utc));
209    }
210    if let Ok(dt) = DateTime::parse_from_rfc3339(trimmed) {
211        return Some(dt.with_timezone(&Utc));
212    }
213    // RFC 850, e.g. "Sunday, 06-Nov-94 08:49:37 GMT"
214    if let Ok(naive) = chrono::NaiveDateTime::parse_from_str(trimmed, "%A, %d-%b-%y %H:%M:%S GMT") {
215        return Some(DateTime::<Utc>::from_naive_utc_and_offset(naive, Utc));
216    }
217    // ANSI C asctime(), e.g. "Sun Nov  6 08:49:37 1994"
218    if let Ok(naive) = chrono::NaiveDateTime::parse_from_str(trimmed, "%a %b %e %H:%M:%S %Y") {
219        return Some(DateTime::<Utc>::from_naive_utc_and_offset(naive, Utc));
220    }
221    None
222}

581        let v = CacheHeadersState::from_headers(&h, t0());
582        // In the past, so fresh_for is zero (not None).
