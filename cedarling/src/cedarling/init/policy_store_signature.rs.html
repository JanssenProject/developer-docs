20/// Bundles are a few KB; anything near this is not a bundle.
21const MAX_BUNDLE_BYTES: u64 = 1024 * 1024;
22
23/// Why a policy store failed its signature check.
24#[derive(Debug, thiserror::Error)]
25pub enum SignatureError {
26    /// The bundle could not be read from disk, or is too large to be a bundle.
27    #[error("failed to load the policy store signature bundle from {location}: {reason}")]
28    Bundle { location: String, reason: String },
29    /// The bundle could not be fetched over HTTP.
30    #[error("failed to fetch the policy store signature bundle from {location}: {source}")]
31    BundleFetch {
32        location: String,
33        #[source]
34        source: HttpClientError,
35    },
36    /// The policy store is not a `.cjar` archive, so it cannot carry a signature.
37    #[error(
38        "`CEDARLING_POLICY_STORE_SIGSTORE_MODE` is `enforce` but the policy store is not a `.cjar` archive"
39    )]
40    UnsupportedSource,
41    /// An in-memory archive has no location to derive its bundle location from.
42    #[error(
43        "no signature bundle location for an in-memory policy store; set CEDARLING_POLICY_STORE_SIGSTORE_BUNDLE_URI"
44    )]
45    NoBundleLocation,
46    /// The bundle does not verify the archive for the configured signer.
47    #[error(
48        "policy store signature does not verify for identity '{identity}' and issuer '{issuer}': {source}"
49    )]
50    Verification {
51        identity: String,
52        issuer: String,
53        #[source]
54        source: SigstoreVerificationError,
55    },
56}
57
58/// Who signed a verified archive, and its digest.
59#[derive(Debug)]
60pub(crate) struct VerifiedArchive {
61    pub identity: String,
62    pub issuer: String,
63    /// Rekor `integratedTime`, UNIX seconds.
64    pub integrated_time: i64,
65    /// Lowercase hex SHA-256 of the archive bytes.
66    pub sha256: String,
67}
68
69impl std::fmt::Display for VerifiedArchive {
70    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
71        write!(
72            f,
73            "identity={} issuer={} archive_sha256={} rekor_integrated_time={}",
74            self.identity, self.issuer, self.sha256, self.integrated_time
75        )
76    }
77}

79/// Runs [`ArchiveVerifier::check`] when signature checking is enabled.
80pub(crate) async fn check_if_enabled(
81    verifier: Option<&ArchiveVerifier>,
82    http: &HttpClient,
83    store_location: Option<&str>,
84    archive: &[u8],
85) -> Result<Option<VerifiedArchive>, SignatureError> {
86    match verifier {
87        Some(verifier) => verifier
88            .check(http, store_location, archive)
89            .await
90            .map(Some),
91        None => Ok(None),
92    }
93}

95/// Verifies `.cjar` archives against the configured signer.
96pub(crate) struct ArchiveVerifier {
97    verifier: SigstoreBlobVerifier,
98    signer: SigstoreSigner,
99}
100
101impl ArchiveVerifier {
102    /// Returns `None` when signature checking is disabled.
103    pub(crate) fn from_config(signature: &PolicyStoreSignature) -> Option<Self> {
104        match signature {
105            PolicyStoreSignature::Disabled => None,
106            PolicyStoreSignature::Enforce(signer) => Some(Self::new(
107                signer,
108                SigstoreBlobVerifier::with_static_trust_root(),
109            )),
110        }
111    }
112
113    pub(crate) fn new(signer: &SigstoreSigner, verifier: SigstoreBlobVerifier) -> Self {
114        Self {
115            verifier,
116            signer: signer.clone(),
117        }
118    }
119
120    /// Loads the bundle for a store at `store_location` and verifies `archive` with it.
121    /// `store_location` is `None` for in-memory archives.
122    pub(crate) async fn check(
123        &self,
124        http: &HttpClient,
125        store_location: Option<&str>,
126        archive: &[u8],
127    ) -> Result<VerifiedArchive, SignatureError> {
128        let location = self
129            .bundle_location(store_location)
130            .ok_or(SignatureError::NoBundleLocation)?;
131        let bundle = self.load_bundle(http, &location).await?;
132        self.verify(archive, &bundle)
133    }
134
135    /// The configured override, else the store location plus `.sigstore.json`.
136    fn bundle_location(&self, store_location: Option<&str>) -> Option<String> {
137        self.signer
138            .bundle_uri
139            .clone()
140            .or_else(|| store_location.map(default_bundle_location))
141    }
142
143    /// Reads the bundle at `location`, over HTTP for `http(s)` URLs, else from disk.
144    async fn load_bundle(
145        &self,
146        http: &HttpClient,
147        location: &str,
148    ) -> Result<Vec<u8>, SignatureError> {
149        if is_http_url(location) {
150            return fetch_bundle(http, location).await;
151        }
152        read_bundle_file(location).map_err(|reason| SignatureError::Bundle {
153            location: location.to_string(),
154            reason,
155        })
156    }
157
158    /// Verifies `bundle` over `archive`.
159    fn verify(&self, archive: &[u8], bundle: &[u8]) -> Result<VerifiedArchive, SignatureError> {
160        let policy = VerificationPolicy {
161            cert_identity: IdentityMatch::Exact(self.signer.cert_identity.clone()),
162            cert_issuer: self.signer.cert_oidc_issuer.clone(),
163        };
164        let verified = self
165            .verifier
166            .verify(archive, bundle, &policy)
167            .map_err(|source| SignatureError::Verification {
168                identity: self.signer.cert_identity.clone(),
169                issuer: self.signer.cert_oidc_issuer.clone(),
170                source,
171            })?;
172        Ok(VerifiedArchive {
173            identity: verified.subject_alternative_name,
174            issuer: verified.issuer,
175            integrated_time: verified.verified_at,
176            sha256: sha256_hex(archive),
177        })
178    }
179}
180
181fn is_http_url(location: &str) -> bool {
182    url::Url::parse(location).is_ok_and(|url| matches!(url.scheme(), "http" | "https"))
183}
184
185/// Appends `.sigstore.json` to the path, so a URL's query string stays at the end.
186fn default_bundle_location(store_location: &str) -> String {
187    match url::Url::parse(store_location) {
188        Ok(mut url) if matches!(url.scheme(), "http" | "https") => {
189            let path = format!("{}.sigstore.json", url.path());
190            url.set_path(&path);
191            url.into()
192        },
193        _ => format!("{store_location}.sigstore.json"),
194    }
195}
196
197/// Transport and HTTP failures stay typed so refresh can count them as network errors.
198async fn fetch_bundle(http: &HttpClient, url: &str) -> Result<Vec<u8>, SignatureError> {
199    let fetch_err = |source| SignatureError::BundleFetch {
200        location: url.to_string(),
201        source,
202    };
203    let response = http.get_with_retry(url).await.map_err(fetch_err)?;
204    let status = response.status();
205    crate::http_utils::read_response_capped(response, Some(MAX_BUNDLE_BYTES))
206        .await
207        .map_err(|reason| match reason {
208            HttpRequestReasonError::ResponseTooLarge { .. } => SignatureError::Bundle {
209                location: url.to_string(),
210                reason: format!("bundle exceeds {MAX_BUNDLE_BYTES} bytes"),
211            },
212            reason => fetch_err(HttpRequestError::new(reason, Some(status))),
213        })
214}
215
216fn read_bundle_file(path: &str) -> Result<Vec<u8>, String> {
217    let mut bundle = Vec::new();
218    std::fs::File::open(path)
219        .and_then(|file| file.take(MAX_BUNDLE_BYTES + 1).read_to_end(&mut bundle))
220        .map_err(|e| e.to_string())?;
221    if bundle.len() as u64 > MAX_BUNDLE_BYTES {
222        return Err(format!("bundle exceeds {MAX_BUNDLE_BYTES} bytes"));
223    }
224    Ok(bundle)
225}
226
227fn sha256_hex(bytes: &[u8]) -> String {
228    Sha256::digest(bytes)
229        .iter()
230        .fold(String::with_capacity(64), |mut hex, byte| {
231            let _ = write!(hex, "{byte:02x}");
232            hex
233        })
234}
