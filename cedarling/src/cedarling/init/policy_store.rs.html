22#[derive(Debug, thiserror::Error)]
23pub enum PolicyStoreLoadError {
24    /// Legacy JSON policy store format is no longer supported.
25    #[error(
26        "Legacy JSON policy store format is no longer supported. Please migrate to the \
27         folder-based policy store format (.cjar archive or directory)."
28    )]
29    LegacyJsonNotSupported,
30    /// Failed to parse policy store from YAML string.
31    #[error("failed to parse the policy store from policy_store yaml: {0}")]
32    ParseYaml(#[from] serde_yaml_ng::Error),
33    /// Failed to fetch the policy store from the lock server.
34    #[error("failed to fetch the policy store from the lock server")]
35    FetchFromLockServer(#[from] HttpClientError),
36    /// Invalid structure in the policy store.
37    #[error("Policy Store does not contain correct structure: {0}")]
38    InvalidStore(String),
39    /// Failed to read or parse the policy store file.
40    #[error("Failed to load policy store from {0}: {1}")]
41    ParseFile(Box<Path>, io::Error),
42    /// Failed to parse the policy store from file.
43    #[error("Failed to convert loaded policy store: {0}")]
44    Conversion(#[from] ConversionError),
45    /// Failed to load policy store from archive.
46    #[error("Failed to load policy store from archive: {0}")]
47    Archive(String),
48    /// Failed to load policy store from directory.
49    #[error("Failed to load policy store from directory: {0}")]
50    Directory(String),
51    /// Policy store validation error.
52    #[error("Policy store validation error: {0}")]
53    Validation(#[from] ValidationError),
54}

56// LegacyAgamaPolicyStore contains the structure to accommodate several policies,
57// and this code for now assumes that there is only ever one policy store,
58// extract the first 'policy_stores' entry.
59fn extract_first_policy_store(
60    agama_policy_store: &LegacyAgamaPolicyStore,
61    strict_schema_validation: bool,
62) -> Result<PolicyStoreWithID, PolicyStoreLoadError> {
63    if agama_policy_store.policy_stores.len() != 1 {
64        return Err(PolicyStoreLoadError::InvalidStore(format!(
65            "expected exactly one 'policy_stores' entry, but found {:?}",
66            agama_policy_store.policy_stores.len()
67        )));
68    }
69
70    // extract exactly the first policy store in the struct
71    let policy_store_option = agama_policy_store
72        .policy_stores
73        .iter()
74        .take(1)
75        .map(|(k, v)| {
76            let store: PolicyStore = v.to_owned().into();
77
78            let metadata = crate::common::policy_store::metadata::PolicyStoreMetadata {
79                cedar_version: agama_policy_store
80                    .cedar_version
81                    .strip_prefix('v')
82                    .unwrap_or(&agama_policy_store.cedar_version)
83                    .to_string(),
84                policy_store: crate::common::policy_store::metadata::PolicyStoreInfo {
85                    id: String::new(),
86                    name: k.clone(),
87                    version: v.version.clone().unwrap_or_default(),
88                    description: None,
89                    created_date: None,
90                    updated_date: None,
91                },
92            };
93
94            PolicyStoreWithID {
95                id: k.to_owned(),
96                store,
97                metadata: Some(metadata),
98            }
99        })
100        .next();
101
102    match policy_store_option {
103        Some(policy_store) => {
104            if strict_schema_validation && policy_store.schema.is_none() {
105                return Err(PolicyStoreLoadError::InvalidStore(
106                    "missing required schema in policy store".to_string(),
107                ));
108            }
109            Ok(policy_store)
110        },
111        None => Err(PolicyStoreLoadError::InvalidStore(
112            "error retrieving first policy_stores element".into(),
113        )),
114    }
115}

117/// Outcome of [`load_policy_store`]: the parsed store plus, for URL-based
118/// sources, both the body-hash and the parsed cache validators (`ETag`,
119/// `Last-Modified`, `max-age` / `Expires`) captured from the bootstrap
120/// response. The refresh worker seeds **both** `last_body_hash` and
121/// `validators` with these values so the first periodic tick can:
122///
123/// - send a conditional GET that may return `304 Not Modified` (no body
124///   downloaded at all — the optimal path), and
125/// - failing that, still short-circuit on the body-hash compare when the
126///   upstream returns a byte-identical body despite ignored conditional
127///   headers.
128///
129/// Both are `None` / empty for non-URL sources (local files, inline JSON/YAML,
130/// in-memory archive bytes) — the refresh worker doesn't spawn there anyway.
131#[cfg_attr(test, derive(Debug))]
132pub(crate) struct LoadedPolicyStore {
133    pub store: PolicyStoreWithID,
134    pub body_hash: Option<u64>,
135    pub validators: CacheHeadersState,
136}

138/// Loads the policy store based on the provided configuration.
139///
140/// This function supports multiple sources for loading policies. The
141/// `strict_schema_validation` flag is forwarded to every parser path so that
142/// the bootstrap-time invariant "policy store must declare a schema" is
143/// enforced consistently regardless of which source the operator picked
144/// (file, URL, archive, etc.).
145pub(crate) async fn load_policy_store(
146    config: &PolicyStoreConfig,
147    http_client: &HttpClient,
148    strict_schema_validation: bool,
149) -> Result<LoadedPolicyStore, PolicyStoreLoadError> {
150    let limits = config.archive_limits();
151
152    let loaded = match &config.source {
153        PolicyStoreSource::Yaml(policy_yaml) => {
154            if crate::common::policy_store::is_json_content(policy_yaml) {
155                return Err(PolicyStoreLoadError::LegacyJsonNotSupported);
156            }
157            let agama_policy_store = serde_yaml_ng::from_str::<LegacyAgamaPolicyStore>(policy_yaml)
158                .map_err(PolicyStoreLoadError::ParseYaml)?;
159            process_legacy_agama_store(&agama_policy_store, strict_schema_validation)?
160        },
161        PolicyStoreSource::LockServer(policy_store_uri) => {
162            load_policy_store_from_uri(
163                policy_store_uri,
164                http_client,
165                strict_schema_validation,
166                limits,
167            )
168            .await?
169        },
170        PolicyStoreSource::FileYaml(path) => {
171            let policy_yaml = fs::read_to_string(path)
172                .map_err(|e| PolicyStoreLoadError::ParseFile(path.clone().into(), e))?;
173            if crate::common::policy_store::is_json_content(&policy_yaml) {
174                return Err(PolicyStoreLoadError::LegacyJsonNotSupported);
175            }
176            let agama_policy_store =
177                serde_yaml_ng::from_str::<LegacyAgamaPolicyStore>(&policy_yaml)
178                    .map_err(PolicyStoreLoadError::ParseYaml)?;
179            process_legacy_agama_store(&agama_policy_store, strict_schema_validation)?
180        },
181        #[cfg(not(target_arch = "wasm32"))]
182        PolicyStoreSource::CjarFile(path) => LoadedPolicyStore {
183            store: load_policy_store_from_cjar_file(path, strict_schema_validation, limits).await?,
184            body_hash: None,
185            validators: CacheHeadersState::default(),
186        },
187        #[cfg(target_arch = "wasm32")]
188        PolicyStoreSource::CjarFile(path) => LoadedPolicyStore {
189            store: load_policy_store_from_cjar_file(path, strict_schema_validation, limits)?,
190            body_hash: None,
191            validators: CacheHeadersState::default(),
192        },
193        PolicyStoreSource::CjarUrl(url) => {
194            load_policy_store_from_cjar_url(url, http_client, strict_schema_validation, limits)
195                .await?
196        },
197        #[cfg(not(target_arch = "wasm32"))]
198        PolicyStoreSource::Directory(path) => LoadedPolicyStore {
199            store: load_policy_store_from_directory(path, strict_schema_validation).await?,
200            body_hash: None,
201            validators: CacheHeadersState::default(),
202        },
203        #[cfg(target_arch = "wasm32")]
204        PolicyStoreSource::Directory(path) => LoadedPolicyStore {
205            store: load_policy_store_from_directory(path, strict_schema_validation)?,
206            body_hash: None,
207            validators: CacheHeadersState::default(),
208        },
209        PolicyStoreSource::ArchiveBytes(bytes) => LoadedPolicyStore {
210            store: load_policy_store_from_archive_bytes(bytes, strict_schema_validation, limits)?,
211            body_hash: None,
212            validators: CacheHeadersState::default(),
213        },
214        PolicyStoreSource::Uri(uri) => {
215            load_policy_store_from_uri(uri, http_client, strict_schema_validation, limits).await?
216        },
217    };
218
219    Ok(loaded)
220}

222fn process_legacy_agama_store(
223    agama_policy_store: &LegacyAgamaPolicyStore,
224    strict_schema_validation: bool,
225) -> Result<LoadedPolicyStore, PolicyStoreLoadError> {
226    MetadataValidator::validate_legacy_store(agama_policy_store)
227        .map_err(PolicyStoreLoadError::Validation)?;
228    Ok(LoadedPolicyStore {
229        store: extract_first_policy_store(agama_policy_store, strict_schema_validation)?,
230        body_hash: None,
231        validators: CacheHeadersState::default(),
232    })
233}

235/// Loads the policy store from a URI with ZIP magic byte-based format detection.
236async fn load_policy_store_from_uri(
237    uri: &str,
238    http_client: &HttpClient,
239    strict_schema_validation: bool,
240    limits: ArchiveLimits,
241) -> Result<LoadedPolicyStore, PolicyStoreLoadError> {
242    let response = http_client.get_with_retry(uri).await?;
243
244    let validators = CacheHeadersState::from_headers(response.headers(), chrono::Utc::now());
245
246    let bytes = http_client.read_response_capped(response).await?;
247    let body_hash = crate::init::policy_store_refresh::body_hash(&bytes);
248
249    if bytes.starts_with(&ZIP_MAGIC) {
250        return Ok(LoadedPolicyStore {
251            store: parse_cjar_bytes(&bytes, strict_schema_validation, limits).await?,
252            body_hash: Some(body_hash),
253            validators,
254        });
255    }
256
257    if crate::common::policy_store::is_json_bytes(&bytes) {
258        return Err(PolicyStoreLoadError::LegacyJsonNotSupported);
259    }
260
261    Err(PolicyStoreLoadError::Archive(
262        "Response body from URI is not a valid Cedar Archive (.cjar)".to_string(),
263    ))
264}

266/// Parses already-fetched `.cjar` archive bytes into a [`PolicyStoreWithID`].
267/// The cfg gate around the `convert_to_legacy` step is localized via the
268/// `convert_archive_to_legacy` helper so the load → metadata → convert flow
269/// can be shared. Native callers offload the CPU-heavy conversion to a
270/// blocking thread; WASM is single-threaded and calls it inline.
271pub(crate) async fn parse_cjar_bytes(
272    bytes: &[u8],
273    strict_schema_validation: bool,
274    limits: ArchiveLimits,
275) -> Result<PolicyStoreWithID, PolicyStoreLoadError> {
276    use crate::common::policy_store::loader;
277
278    let loaded = loader::load_policy_store_archive_bytes(bytes, strict_schema_validation, limits)
279        .map_err(|e| {
280        PolicyStoreLoadError::Archive(format!("Failed to load from archive: {e}"))
281    })?;
282
283    let store_id = loaded.metadata.policy_store.id.clone();
284    let store_metadata = loaded.metadata.clone();
285
286    #[cfg(not(target_arch = "wasm32"))]
287    let legacy_store = tokio::task::spawn_blocking(move || {
288        convert_archive_to_legacy(loaded, strict_schema_validation)
289    })
290    .await
291    .map_err(|e| PolicyStoreLoadError::Archive(format!("Conversion task panicked: {e}")))??;
292    #[cfg(target_arch = "wasm32")]
293    let legacy_store = convert_archive_to_legacy(loaded, strict_schema_validation)?;
294
295    Ok(PolicyStoreWithID {
296        id: store_id,
297        store: legacy_store,
298        metadata: Some(store_metadata),
299    })
300}
301
302/// Synchronous shared helper: runs `PolicyStoreManager::convert_to_legacy`
303/// and lifts the error into [`PolicyStoreLoadError`]. Always synchronous —
304/// the platform-specific "offload to a blocking thread vs. run inline"
305/// decision is made by callers, since WASM has no `spawn_blocking` and an
306/// `async` wrapper there would have nothing to await.
307fn convert_archive_to_legacy(
308    loaded: crate::common::policy_store::loader::LoadedPolicyStore,
309    strict_schema_validation: bool,
310) -> Result<crate::common::policy_store::PolicyStore, PolicyStoreLoadError> {
311    PolicyStoreManager::convert_to_legacy(loaded, strict_schema_validation).map_err(Into::into)
312}

314/// Loads the policy store from a Cedar Archive (.cjar) file.
315///
316/// Uses the `load_policy_store_archive` function from the loader module
317/// and converts to legacy format for backward compatibility.
318#[cfg(not(target_arch = "wasm32"))]
319async fn load_policy_store_from_cjar_file(
320    path: &Path,
321    strict_schema_validation: bool,
322    limits: ArchiveLimits,
323) -> Result<PolicyStoreWithID, PolicyStoreLoadError> {
324    use crate::common::policy_store::loader;
325
326    let loaded = loader::load_policy_store_archive(path, strict_schema_validation, limits)
327        .await
328        .map_err(|e| map_policy_store_err(e, true))?;
329
330    // Get the policy store ID and metadata
331    let store_id = loaded.metadata.policy_store.id.clone();
332    let store_metadata = loaded.metadata.clone();
333
334    // Convert to legacy format in a blocking task (schema parsing is CPU-heavy)
335    let legacy_store = tokio::task::spawn_blocking(move || {
336        convert_archive_to_legacy(loaded, strict_schema_validation)
337    })
338    .await
339    .map_err(|e| PolicyStoreLoadError::Archive(format!("Conversion task panicked: {e}")))??;
340
341    Ok(PolicyStoreWithID {
342        id: store_id,
343        store: legacy_store,
344        metadata: Some(store_metadata),
345    })
346}

367/// Loads the policy store from a Cedar Archive (.cjar) URL.
368///
369/// Fetches the archive via HTTP (capturing response headers for the
370/// `validators` seed and body bytes for the `body_hash` seed), loads it via
371/// `load_policy_store_archive_bytes`, and converts to legacy format. The CPU-
372/// heavy schema-parsing step in `convert_to_legacy` is shared via the
373/// `convert_archive_to_legacy` helper; the only platform-specific concern is
374/// whether to wrap that call in `tokio::task::spawn_blocking` (native) or
375/// run it inline (WASM, which is single-threaded).
376async fn load_policy_store_from_cjar_url(
377    url: &str,
378    http_client: &HttpClient,
379    strict_schema_validation: bool,
380    limits: ArchiveLimits,
381) -> Result<LoadedPolicyStore, PolicyStoreLoadError> {
382    use crate::common::policy_store::loader;
383
384    // Fetch via `get_with_retry` to capture response headers (for the refresh
385    // worker's initial `validators` seed) alongside the body bytes (for the
386    // `last_body_hash` seed).
387    let response = http_client
388        .get_with_retry(url)
389        .await
390        .map_err(|e| PolicyStoreLoadError::Archive(format!("Failed to fetch archive: {e}")))?;
391    let validators = CacheHeadersState::from_headers(response.headers(), chrono::Utc::now());
392    // Cap-aware body read: `CEDARLING_HTTP_MAX_RESPONSE_SIZE` bounds the
393    // archive download so an oversized `.cjar` URL can't exhaust memory.
394    let bytes = http_client
395        .read_response_capped(response)
396        .await
397        .map_err(|e| PolicyStoreLoadError::Archive(format!("Failed to read archive body: {e}")))?;
398
399    let body_hash = crate::init::policy_store_refresh::body_hash(&bytes);
400
401    let loaded = loader::load_policy_store_archive_bytes(&bytes, strict_schema_validation, limits)
402        .map_err(|e| map_policy_store_err(e, true))?;
403
404    let store_id = loaded.metadata.policy_store.id.clone();
405    let store_metadata = loaded.metadata.clone();
406
407    #[cfg(not(target_arch = "wasm32"))]
408    let legacy_store = tokio::task::spawn_blocking(move || {
409        convert_archive_to_legacy(loaded, strict_schema_validation)
410    })
411    .await
412    .map_err(|e| PolicyStoreLoadError::Archive(format!("Conversion task panicked: {e}")))??;
413    #[cfg(target_arch = "wasm32")]
414    let legacy_store = convert_archive_to_legacy(loaded, strict_schema_validation)?;
415
416    Ok(LoadedPolicyStore {
417        store: PolicyStoreWithID {
418            id: store_id,
419            store: legacy_store,
420            metadata: Some(store_metadata),
421        },
422        body_hash: Some(body_hash),
423        validators,
424    })
425}

478/// Loads the policy store directly from archive bytes.
479///
480/// This is useful for:
481/// - WASM environments with custom fetch logic (e.g., auth headers)
482/// - Embedding archives in applications
483/// - Loading from non-standard sources (databases, S3, etc.)
484///
485/// Works on all platforms including WASM.
486fn load_policy_store_from_archive_bytes(
487    bytes: &[u8],
488    strict_schema_validation: bool,
489    limits: ArchiveLimits,
490) -> Result<PolicyStoreWithID, PolicyStoreLoadError> {
491    use crate::common::policy_store::loader;
492
493    // Load from bytes (works in both native and WASM)
494    let loaded = loader::load_policy_store_archive_bytes(bytes, strict_schema_validation, limits)
495        .map_err(|e| map_policy_store_err(e, true))?;
496
497    // Get the policy store ID and metadata
498    let store_id = loaded.metadata.policy_store.id.clone();
499    let store_metadata = loaded.metadata.clone();
500
501    // Convert to legacy format using the shared helper
502    let legacy_store = convert_archive_to_legacy(loaded, strict_schema_validation)?;
503
504    Ok(PolicyStoreWithID {
505        id: store_id,
506        store: legacy_store,
507        metadata: Some(store_metadata),
508    })
509}
510
511fn map_policy_store_err(e: PolicyStoreError, is_archive: bool) -> PolicyStoreLoadError {
512    match e {
513        PolicyStoreError::Validation(ve) => PolicyStoreLoadError::Validation(ve),
514        PolicyStoreError::CedarParsing { file, detail } => {
515            PolicyStoreLoadError::InvalidStore(format!("Cedar parse error in {file}: {detail}"))
516        },
517        PolicyStoreError::CedarSchemaError { file, err } => {
518            PolicyStoreLoadError::InvalidStore(format!("Cedar schema error in {file}: {err}"))
519        },
520        _ => {
521            if is_archive {
522                PolicyStoreLoadError::Archive(e.to_string())
523            } else {
524                PolicyStoreLoadError::Directory(e.to_string())
525            }
526        },
527    }
528}
