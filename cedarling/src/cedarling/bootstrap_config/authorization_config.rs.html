6use serde::{Deserialize, Serialize};
7
8/// Configuration to specify authorization workflow.
9#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
10pub struct AuthorizationConfig {
11    /// Claim name used for decision logging (e.g. which JWT claim identifies the token).
12    /// `CEDARLING_DECISION_LOG_DEFAULT_JWT_ID` in bootstrap properties documentation.
13    pub decision_log_default_jwt_id: String,
14
15    /// Whether to enforce Cedar schema validation.
16    ///
17    /// When `true` (default), a schema is required and policies/entities are validated
18    /// against it. When `false`, Cedarling runs without schema-based validation.
19    pub strict_schema_validation: bool,
20
21    /// Timeout in milliseconds applied to a
22    /// [`CustomTokenProcessor::process`](crate::CustomTokenProcessor::process)
23    /// call for **non-JWT** custom tokens. `0` (default) disables the timeout;
24    /// a positive value races `process` against the deadline, producing a
25    /// distinct timeout error.
26    ///
27    /// Corresponds to `CEDARLING_CUSTOM_TOKEN_PROCESSOR_TIMEOUT_MILLIS`.
28    #[serde(default)]
29    pub custom_token_processor_timeout_millis: u64,
30}
31
32impl Default for AuthorizationConfig {
33    fn default() -> Self {
34        Self {
35            decision_log_default_jwt_id: "jti".to_string(),
36            strict_schema_validation: true,
37            custom_token_processor_timeout_millis: 0,
38        }
39    }
40}
