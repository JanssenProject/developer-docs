473    /// Upper bound on the Status List JWT refresh interval, in seconds.
474    ///
475    /// Caps how long Cedarling waits between Status List refreshes. When the Status
476    /// List JWT carries a `ttl` claim, the effective refresh interval is
477    /// `min(jwt_ttl, status_list_refresh_interval_max)`, so the issuer can request a
478    /// *more frequent* refresh but never a less frequent one. When the JWT omits
479    /// `ttl`, this value is used directly. A value of `0` or an unset variable is
480    /// treated as "use the default" (300 seconds) so the status list cannot silently
481    /// go stale forever. Non-zero values below `5` are clamped to `5`.
482    ///
483    /// Fail-closed behavior: if any background refresh fails — fetch error, 5xx
484    /// response, or the status list body is invalid (JWT validation fails,
485    /// deserialization fails, or bit-string parsing fails) — the cached status list
486    /// is dropped and all tokens that reference it are rejected until the next
487    /// successful refresh. This prevents a revoked token from being accepted based on
488    /// stale data at the cost of temporarily denying valid tokens when the issuer's
489    /// status endpoint is unreachable or returns a malformed payload.
490    #[serde(
491        rename = "CEDARLING_JWT_STATUS_LIST_REFRESH_INTERVAL_MAX",
492        default = "default_status_list_refresh_interval_max"
493    )]

495    pub status_list_refresh_interval_max: u64,

497    /// Base refresh interval, in seconds, for periodic background refresh of
498    /// remote policy stores (`CjarUrl` / `LockServer`). `0` disables refresh and
499    /// preserves the load-once-at-startup behavior. Non-zero values below the
500    /// `MIN_REFRESH_INTERVAL_SECS` floor are clamped at service-init time (with
501    /// a `WARN` log emitted) so the worker never busy-polls the upstream — see
502    /// [`PolicyStoreConfig::effective_refresh_interval`]. A server
503    /// `Cache-Control: max-age` / `Expires` hint can *shorten* the next
504    /// interval but never extends it.

527impl BootstrapConfigRaw {
528    /// Construct `BootstrapConfig` from environment variables and `BootstrapConfigRaw` config.
529    /// Environment variables have bigger priority.
530    //
531    // Simple implementation that map input structure to JSON map
532    // and map environment variables with prefix `CEDARLING_` to JSON map. And merge it.
533    #[cfg(not(target_arch = "wasm32"))]
534    pub fn from_raw_config_and_env(
535        raw: Option<BootstrapConfigRaw>,
536    ) -> Result<Self, BootstrapConfigLoadingError> {
537        let mut json_config_params = serde_json::json!(raw.unwrap_or_default())
538            .as_object()
539            .map(std::borrow::ToOwned::to_owned)
540            .unwrap_or_default();
541
542        for (k, v) in get_cedarling_env_vars() {
543            // update map with values from env variables
544            json_config_params.insert(k, v);
545        }
546
547        Ok(BootstrapConfigRaw::deserialize(json_config_params)?)
548    }
549}
550
551/// Get environment variables related to `Cedarling`
552#[cfg(not(target_arch = "wasm32"))]
553fn get_cedarling_env_vars() -> HashMap<String, serde_json::Value> {
554    env::vars()
555        .filter_map(|(k, v)| {
556            k.starts_with("CEDARLING_")
557                .then_some((k, serde_json::json!(v)))
558        })
559        .collect()
560}

575    fn with_env_vars<F>(vars: &[(&str, &str)], test: F)
576    where
577        F: FnOnce(),
578    {
579        // Ensure only one test modifies env vars at a time
580        let _lock = ENV_MUTEX.lock().unwrap();
581
582        // Create a fresh environment by clearing all CEDARLING_* vars
583        let mut all_vars = Vec::new();
584        for (key, value) in env::vars() {
585            let key_clone = key.clone();
586            all_vars.push((key, value));
587            unsafe {
588                env::remove_var(&key_clone);
589            }
590        }
591
592        // Set new env vars
593        for (key, value) in vars {
594            unsafe {
595                env::set_var(key, value);
596            }
597        }
598
599        test();
600
601        // Clean up
602        for (key, _) in vars {
603            unsafe {
604                env::remove_var(key);
605            }
606        }
607
608        // Restore original environment
609        for (key, value) in all_vars {
610            unsafe {
611                env::set_var(&key, value);
612            }
613        }
614    }
