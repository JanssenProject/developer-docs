12/// `PolicyStoreConfig` - Configuration for the policy store.
13///
14/// Defines where the policy will be retrieved from.
15#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
16pub struct PolicyStoreConfig {
17    /// Specifies the source from which the policy will be read.
18    pub source: PolicyStoreSource,
19
20    /// Base refresh interval in seconds for URL-based policy store sources
21    /// (`CjarUrl`, `LockServer`, `Uri`). `0` disables background refresh and preserves
22    /// the load-once-at-startup behavior. Ignored for local sources. A server
23    /// `Cache-Control: max-age` / `Expires` hint may *shorten* the next
24    /// interval but never lengthens it.
25    #[serde(default)]
26    pub refresh_interval_secs: u64,
27
28    /// `CEDARLING_POLICY_STORE_MAX_FILE_SIZE` — cap on the decompressed size of
29    /// a single `.cjar` entry, in bytes. `0` disables the cap.
30    #[serde(default = "default_policy_store_max_file_size")]
31    pub max_file_size: u64,
32}
33
34/// Serde default for [`PolicyStoreConfig::max_file_size`], so a config
35/// deserialized without the field still gets a bounded archive loader.
36pub(crate) fn default_policy_store_max_file_size() -> u64 {
37    ArchiveLimits::DEFAULT_MAX_ENTRY_SIZE
38}
39
40impl PolicyStoreConfig {
41    /// Minimum refresh interval, in seconds — anything smaller is clamped up to
42    /// this value to avoid a busy-poll against the upstream.
43    pub(crate) const MIN_REFRESH_INTERVAL_SECS: u64 = 5;
44
45    /// True if the source is a remote URL and refresh is enabled.
46    #[must_use]
47    pub fn refresh_enabled(&self) -> bool {
48        self.refresh_interval_secs > 0
49            && matches!(
50                self.source,
51                PolicyStoreSource::CjarUrl(_)
52                    | PolicyStoreSource::LockServer(_)
53                    | PolicyStoreSource::Uri(_)
54            )
55    }
56
57    /// Returns the effective refresh interval after applying the
58    /// `MIN_REFRESH_INTERVAL_SECS` floor, plus a boolean indicating whether
59    /// clamping occurred. `0` (disabled) passes through unchanged. Callers
60    /// should emit a `WARN` log when `clamped` is true so operators see the
61    /// silent normalization. Single point of normalization so deserializer
62    /// inputs (env vars, JSON, dict) can't drift apart.
63    #[must_use]
64    pub(crate) fn effective_refresh_interval(&self) -> (u64, bool) {
65        let raw = self.refresh_interval_secs;
66        if raw == 0 || raw >= Self::MIN_REFRESH_INTERVAL_SECS {
67            (raw, false)
68        } else {
69            (Self::MIN_REFRESH_INTERVAL_SECS, true)
70        }
71    }
72}
73
74impl PolicyStoreConfig {
75    /// Resource limits for `.cjar` loading, derived from [`Self::max_file_size`].
76    #[must_use]
77    pub(crate) fn archive_limits(&self) -> ArchiveLimits {
78        ArchiveLimits::from_max_file_size(self.max_file_size)
79    }
80}
81
82impl Default for PolicyStoreConfig {
83    fn default() -> Self {
84        Self {
85            source: PolicyStoreSource::Yaml(
86                "cedar_version: v4.0.0\npolicy_stores: {}\n".to_string(),
87            ),
88            refresh_interval_secs: 0,
89            max_file_size: default_policy_store_max_file_size(),
90        }
91    }
92}

102/// `PolicyStoreSource` represents the source from which policies will be retrieved.
103#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
104pub enum PolicyStoreSource {
105    /// Read the policy directly from a raw YAML string.
106    ///
107    /// The string contains the raw YAML data representing the policy.
108    /// Mostly used only for testing purposes.
109    Yaml(String),
110
111    /// Fetch the policies from the Lock Master service using a specified identifier.
112    ///
113    /// The string contains a URI where the policy store can be retrieved.
114    LockServer(String),
115
116    /// Read policy from a YAML File.
117    FileYaml(PathBuf),
118
119    /// Read policy from a Cedar Archive (.cjar) file.
120    ///
121    /// The path points to a `.cjar` archive containing the policy store
122    /// in the new directory structure format.
123    CjarFile(PathBuf),
124
125    /// Read policy from a Cedar Archive (.cjar) fetched from a URL.
126    ///
127    /// The string contains a URL where the `.cjar` archive can be downloaded.
128    CjarUrl(String),
129
130    /// Read policy from a directory structure.
131    ///
132    /// The path points to a directory containing the policy store
133    /// in the directory structure format (`metadata.json`, `schema.cedarschema`, `policies/`, etc.).
134    Directory(PathBuf),
135
136    /// An unresolved URI whose source type (archive vs lock server) is detected
137    /// at load time via magic byte checking.
138    ///
139    /// During loading, [`load_policy_store`](crate::init::policy_store::load_policy_store)
140    /// resolves this by making an HTTP request to determine whether the URI points
141    /// to a Cedar Archive (`.cjar`) or a Lock Master endpoint.
142    Uri(String),
143
144    /// Read policy from Cedar Archive bytes directly.
145    ///
146    /// The bytes contain a `.cjar` archive (ZIP format) with the policy store.
147    /// This is particularly useful for:
148    /// - WASM environments with custom fetch logic
149    /// - Embedding archives in applications
150    /// - Loading from non-standard sources (databases, S3, etc.)
151    ArchiveBytes(Vec<u8>),
152}
153
154/// Raw policy store source
155pub enum PolicyStoreSourceRaw {
156    /// YAML
157    Yaml(String),
158    /// Lock server
159    LockServer(String),
160    /// File YAML
161    FileYaml(String),
162    /// Cedar Archive file (.cjar)
163    CjarFile(String),
164    /// Cedar Archive URL (.cjar)
165    CjarUrl(String),
166    /// Directory structure
167    Directory(String),
168}
169
170impl TryFrom<PolicyStoreConfigRaw> for PolicyStoreConfig {
171    type Error = BootstrapConfigLoadingError;
172
173    fn try_from(raw: PolicyStoreConfigRaw) -> Result<Self, Self::Error> {
174        let source = match raw.source.as_str() {
175            "json" | "file_json" => {
176                return Err(BootstrapConfigLoadingError::LegacyJsonNotSupported);
177            },
178            "yaml" => PolicyStoreSource::Yaml(raw.path.unwrap_or_default()),
179
180            "lock_server" => PolicyStoreSource::LockServer(raw.path.unwrap_or_default()),
181            "file_yaml" => PolicyStoreSource::FileYaml(raw.path.unwrap_or_default().into()),
182            "cjar_file" => PolicyStoreSource::CjarFile(
183                raw.path
184                    .filter(|p| !p.is_empty())
185                    .unwrap_or_else(|| "policy-store.cjar".to_string())
186                    .into(),
187            ),
188            "cjar_url" => {
189                let url = raw.path.filter(|p| !p.is_empty()).unwrap_or_default();
190                if url.is_empty() {
191                    return Err(BootstrapConfigLoadingError::MissingCjarUrl);
192                }
193                PolicyStoreSource::CjarUrl(url)
194            },
195            "directory" => PolicyStoreSource::Directory(
196                raw.path
197                    .filter(|p| !p.is_empty())
198                    .unwrap_or_else(|| "policy-store".to_string())
199                    .into(),
200            ),
201            _ => PolicyStoreSource::FileYaml("policy-store.yaml".into()),
202        };
203        // Explicit field init rather than `..Default::default()`, which would
204        // allocate and immediately discard the default YAML source string.
205        Ok(Self {
206            source,
207            refresh_interval_secs: 0,
208            max_file_size: default_policy_store_max_file_size(),
209        })
210    }
211}
