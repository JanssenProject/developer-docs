7use jsonwebtoken::Algorithm;
8use serde::{Deserialize, Serialize};
9use std::{collections::HashSet, num::NonZeroUsize};

11/// The set of Bootstrap properties related to JWT validation.
12#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
13pub struct JwtConfig {
14    /// A Json Web Key Store (JWKS) with public keys.
15    ///
16    /// If this is used, Cedarling will no longer try to fetch JWK Stores from
17    /// a trustede identity provider and stick to using the local JWKS.
18    pub jwks: Option<String>,
19    /// Check the signature for all the Json Web Tokens.
20    ///
21    /// This Requires the `iss` claim to be present in all the tokens and
22    /// and the scheme must be `https`.
23    ///
24    /// This setting overrides the `iss` validation settings in the following:
25    ///
26    /// - `access_token_config`
27    /// - `id_token_config`
28    /// - `userinfo_token_config`
29    pub jwt_sig_validation: bool,
30    /// Whether to check the status of the JWT.
31    ///
32    /// On startup, the Cedarling will fetch and retreive the latest Status List
33    /// JWT from the `.well-known/openid-configuration` via the `status_list_endpoint`
34    /// claim and cache it.
35    ///
36    /// See the [`IETF Draft`] for more info.
37    ///
38    /// [`IETF Draft`]: https://datatracker.ietf.org/doc/draft-ietf-oauth-status-list/
39    pub jwt_status_validation: bool,
40    /// Only tokens signed with algorithms in this list can be valid.
41    pub signature_algorithms_supported: HashSet<Algorithm>,
42    /// Maximum TTL (in seconds) for cached tokens.
43    /// Zero disables the token cache entirely.
44    ///
45    /// Defaults to [`Self::DEFAULT_TOKEN_CACHE_MAX_TTL_SECS`] (5 seconds): small
46    /// enough that revocation / status-list changes are picked up promptly while
47    /// still serving repeated requests for the same token from cache. Keep this
48    /// value short (seconds, not minutes) to bound cache growth and staleness.
49    pub token_cache_max_ttl_secs: usize,
50    /// Maximum number of tokens the cache can store.
51    pub token_cache_capacity: usize,
52    /// Enables eviction policy based on the earliest expiration time.
53    ///
54    /// When the cache reaches its capacity, the entry with the nearest
55    /// expiration timestamp will be removed to make room for a new one.
56    pub token_cache_earliest_expiration_eviction: bool,
57    /// Configuration for loading trusted issuers.
58    pub trusted_issuer_loader: TrustedIssuerLoaderConfig,
59    /// Optional override for JWKS periodic refresh interval in seconds.
60    /// Set to `None` to use the server-driven or fallback interval.
61    pub jwks_refresh_interval: Option<u64>,
62    /// Minimum interval in seconds between on-demand JWKS re-fetches per issuer.
63    pub jwks_refresh_min_interval: u64,
64    /// Upper bound on the Status List JWT refresh interval in seconds.
65    ///
66    /// Caps how long Cedarling waits between Status List refreshes. When the Status
67    /// List JWT carries a `ttl` claim (per the IETF `oauth-status-list` spec), the
68    /// effective refresh interval is `min(jwt_ttl, status_list_refresh_interval_max)`
69    /// so the issuer can always request a *more frequent* refresh, but never a less
70    /// frequent one. When the JWT omits `ttl`, this value is used directly.
71    ///
72    /// A value of `0` is treated as "use the default" so the cache cannot be left to
73    /// go stale forever. Non-zero values below `MIN_STATUS_LIST_REFRESH_SECS` (5) are
74    /// clamped to that minimum.
75    pub status_list_refresh_interval_max: u64,
76}
77
78/// Default periodic JWKS refresh interval when neither `Cache-Control: max-age`
79/// nor the bootstrap config provides a value.
80pub(crate) const DEFAULT_JWKS_REFRESH_INTERVAL_SECS: u64 = 3600;
81
82/// Minimum allowed refresh interval, in seconds. Values below this are clamped.
83pub(crate) const MIN_JWKS_REFRESH_SECS: u64 = 5;
84
85/// Minimum allowed Status List JWT refresh interval (in seconds). Non-zero values
86/// below this are clamped up to this floor.
87pub(crate) const MIN_STATUS_LIST_REFRESH_SECS: u64 = 5;
88
89/// Normalize a candidate Status List JWT refresh maximum (seconds) into a safe
90/// interval. Single source of truth shared by the bootstrap deserializer and
91/// [`JwtConfig::normalize`]:
92///
93/// - `0` -> [`JwtConfig::DEFAULT_STATUS_LIST_REFRESH_INTERVAL_MAX_SECS`]
94///   (300s): the cache must never be left to go stale forever, so "disabled"
95///   is not a supported state.
96/// - Non-zero values below [`MIN_STATUS_LIST_REFRESH_SECS`] are clamped up to
97///   that floor to avoid hammering the IDP.
98/// - Any other value is returned unchanged.
99#[must_use]
100pub(crate) fn normalize_status_list_refresh_interval_max(value: u64) -> u64 {
101    match value {
102        0 => JwtConfig::DEFAULT_STATUS_LIST_REFRESH_INTERVAL_MAX_SECS,
103        v => v.max(MIN_STATUS_LIST_REFRESH_SECS),
104    }
105}
106
107impl Default for JwtConfig {
108    /// Cedarling uses strict-by-default validation.
109    ///
110    /// The returned value is identical to what is produced when parsing an
111    /// empty bootstrap configuration (`BootstrapConfigRaw::default()` →
112    /// `JwtConfig`). Signature and status validation are **on**. To opt out
113    /// for testing or trusted environments, set the flags explicitly or use
114    /// [`JwtConfig::new_without_validation`].
115    fn default() -> Self {
116        let config = Self {
117            jwks: None,
118            jwt_sig_validation: true,
119            jwt_status_validation: true,
120            signature_algorithms_supported: HashSet::new(),
121            token_cache_capacity: Self::DEFAULT_TOKEN_CACHE_CAPACITY,
122            token_cache_earliest_expiration_eviction: true,
123            token_cache_max_ttl_secs: Self::DEFAULT_TOKEN_CACHE_MAX_TTL_SECS,
124            trusted_issuer_loader: TrustedIssuerLoaderConfig::default(),
125            jwks_refresh_interval: None,
126            jwks_refresh_min_interval: Self::DEFAULT_JWKS_REFRESH_MIN_INTERVAL,
127            status_list_refresh_interval_max:
128                JwtConfig::DEFAULT_STATUS_LIST_REFRESH_INTERVAL_MAX_SECS,
129        };
130        config.allow_all_algorithms()
131    }
132}
133
134impl JwtConfig {
135    /// Default maximum number of tokens the token cache can store.
136    pub const DEFAULT_TOKEN_CACHE_CAPACITY: usize = 100;
137
138    /// Default maximum TTL (seconds) for cached tokens.
139    ///
140    /// Chosen as a small value so that repeated requests for the same token are
141    /// served from cache, while revocation / status-list changes are still
142    /// picked up within a few seconds. Prevents unbounded cache growth for
143    /// tokens without an `exp` claim.
144    pub const DEFAULT_TOKEN_CACHE_MAX_TTL_SECS: usize = 5;
145
146    /// Default minimum interval (seconds) between on-demand JWKS re-fetches per issuer.
147    pub const DEFAULT_JWKS_REFRESH_MIN_INTERVAL: u64 = 30;
148
149    /// Default upper bound for the Status List JWT refresh interval (in seconds),
150    /// applied when the Status List JWT has no `ttl` claim and as a cap on JWT-
151    /// provided values. Also used when the bootstrap property is set to `0`.
152    pub const DEFAULT_STATUS_LIST_REFRESH_INTERVAL_MAX_SECS: u64 = 300;
153
154    /// Enforce all field-level invariants on the config in place. Called once
155    /// during service initialization on a cloned `JwtConfig`, so that callers
156    /// who construct the struct programmatically cannot bypass the bootstrap
157    /// deserializer's normalization.
158    pub(crate) fn normalize(&mut self) {
159        self.status_list_refresh_interval_max =
160            normalize_status_list_refresh_interval_max(self.status_list_refresh_interval_max);
161    }
162
163    /// Creates a new `JwtConfig` instance with validation turned off for all tokens.
164    ///
165    /// Intended for tests and trusted-environment embedders. Production
166    /// deployments should use [`JwtConfig::default`] (strict).
167    #[must_use]
168    pub fn new_without_validation() -> Self {
169        Self {
170            jwks: None,
171            jwt_sig_validation: false,
172            jwt_status_validation: false,
173            signature_algorithms_supported: HashSet::new(),
174            ..Default::default()
175        }
176        .allow_all_algorithms()
177    }
178
179    pub(crate) fn supported_algorithms() -> HashSet<Algorithm> {
180        HashSet::from_iter([
181            Algorithm::HS256,
182            Algorithm::HS384,
183            Algorithm::HS512,
184            Algorithm::ES256,
185            Algorithm::ES384,
186            Algorithm::RS256,
187            Algorithm::RS384,
188            Algorithm::RS512,
189            Algorithm::PS256,
190            Algorithm::PS384,
191            Algorithm::PS512,
192            Algorithm::EdDSA,
193        ])
194    }
195
196    /// Adds all supported algorithms to `signature_algorithms_supported`.
197    #[must_use]
198    pub fn allow_all_algorithms(mut self) -> Self {
199        self.signature_algorithms_supported = Self::supported_algorithms();
200        self
201    }
202}

204/// Raw representation of trusted issuer loader type from environment variable.
205/// Is used in [`BootstrapConfigRaw`].
206#[derive(Debug, Default, Clone, PartialEq, Serialize, Deserialize)]
207pub enum TrustedIssuerLoaderTypeRaw {
208    /// Synchronous loading
209    #[default]
210    #[serde(rename = "SYNC")]
211    Sync,
212    /// Asynchronous loading
213    #[serde(rename = "ASYNC")]
214    Async,
215}
216
217impl TrustedIssuerLoaderTypeRaw {
218    /// Converts raw representation to `TrustedIssuerLoaderConfig`.
219    pub(crate) fn to_config(&self, workers: WorkersCount) -> TrustedIssuerLoaderConfig {
220        match self {
221            TrustedIssuerLoaderTypeRaw::Sync => TrustedIssuerLoaderConfig::Sync { workers },
222            TrustedIssuerLoaderTypeRaw::Async => TrustedIssuerLoaderConfig::Async { workers },
223        }
224    }
225}
226
227/// Config structure that define how trusted issuers will be loaded.
228///
229/// Default is `Sync` with 1 worker.
230#[derive(Debug, Copy, Clone, PartialEq, Serialize, Deserialize)]
231pub enum TrustedIssuerLoaderConfig {
232    /// Synchronous loading, on start program.
233    /// The Cedarling will load all entities on start in "blocking mode" (you need to wait).
234    Sync {
235        /// Workers count
236        workers: WorkersCount,
237    },
238    /// Asynchronous loading, on start program.
239    /// The Cedarling will load all entities on the background.
240    /// You need specify workers count.
241    Async {
242        /// Workers count
243        workers: WorkersCount,
244    },
245}
246
247impl Default for TrustedIssuerLoaderConfig {
248    fn default() -> Self {
249        Self::Sync {
250            workers: WorkersCount::MIN,
251        }
252    }
253}

255/// A wrapper around `NonZeroUsize` that enforces a maximum value.
256/// This is used to ensure that the number of workers for loading trusted issuers does not exceed a reasonable limit.
257/// The maximum value is defined based on the target architecture to prevent excessive resource usage.
258///
259/// On non-WebAssembly targets, the maximum is set to 1000, while on WebAssembly targets,
260/// it is set to 4 to align with typical browser limits on concurrent HTTP connections.
261#[derive(
262    Debug, Copy, Clone, PartialEq, Eq, PartialOrd, Ord, derive_more::Deref, serde::Serialize,
263)]
264pub struct WorkersCount(NonZeroUsize);
265
266#[cfg(not(target_arch = "wasm32"))]
267impl WorkersCount {
268    /// Maximum number of workers is 1000 to prevent excessive resource usage.
269    pub const MAX: WorkersCount = WorkersCount(NonZeroUsize::new(1000).unwrap());
270
271    /// For native architecture default value is 10. Should cover most of cases.
272    pub const DEFAULT: WorkersCount = WorkersCount(NonZeroUsize::new(10).unwrap());
273}
274
275#[cfg(target_arch = "wasm32")]
276impl WorkersCount {
277    /// On WebAssembly targets, we set workers limit 6.
278    ///
279    /// Most of web browsers have limit of 6 concurrent http connections, so we respect that.
280    pub const MAX: WorkersCount = WorkersCount(NonZeroUsize::new(6).unwrap());
281
282    /// For WASM architecture default value is 2
283    pub const DEFAULT: WorkersCount = WorkersCount(NonZeroUsize::new(2).unwrap());
284}
285
286impl WorkersCount {
287    /// Minimum number of workers is 1.
288    pub const MIN: WorkersCount = WorkersCount(NonZeroUsize::MIN);
289}
290
291impl WorkersCount {
292    /// Creates a new `NonZeroUsizeLimited` instance, ensuring the value is non-zero and does not exceed the defined maximum.
293    #[must_use]
294    pub fn new(value: usize) -> Self {
295        let value = NonZeroUsize::new(value)
296            .unwrap_or(NonZeroUsize::MIN)
297            .min(Self::MAX.0);
298
299        Self(value)
300    }
301}
302
303impl Default for WorkersCount {
304    fn default() -> Self {
305        Self::DEFAULT
306    }
307}

309impl<'de> serde::Deserialize<'de> for WorkersCount {
310    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
311    where
312        D: serde::Deserializer<'de>,
313    {
314        let value = usize::deserialize(deserializer)?;
315        Ok(Self::new(value))
316    }
317}

319impl PartialEq<usize> for WorkersCount {
320    fn eq(&self, other: &usize) -> bool {
321        self.0.get() == *other
322    }
