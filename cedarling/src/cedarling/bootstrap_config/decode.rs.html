35    /// Construct `BootstrapConfig` from environment variables and `BootstrapConfigRaw` config.
36    /// Environment variables have bigger priority.
37    //
38    // Simple implementation that map input structure to JSON map
39    // and map environment variables with prefix `CEDARLING_` to JSON map. And merge it.
40    #[cfg(not(target_arch = "wasm32"))]
41    pub fn from_raw_config_and_env(
42        raw: Option<BootstrapConfigRaw>,
43    ) -> Result<Self, BootstrapConfigLoadingError> {
44        let config_raw = BootstrapConfigRaw::from_raw_config_and_env(raw)?;
45        Self::from_raw_config(&config_raw)
46    }

48    /// Construct an instance from [`BootstrapConfigRaw`]
49    pub fn from_raw_config(raw: &BootstrapConfigRaw) -> Result<Self, BootstrapConfigLoadingError> {
50        let lock_config = raw.lock.is_enabled().then(|| raw.try_into()).transpose()?;
51
52        // Decode LogCofig
53        let log_config = LogConfig {
54            log_type: resolve_log_type(raw)?,
55            log_level: raw.log_level,
56        };
57
58        let policy_store_config = build_policy_store_config(raw)?;
59        let trace_config = build_trace_config(raw, &policy_store_config.source)?;
60
61        // Load the jwks from a local file
62        let jwks = raw
63            .local_jwks
64            .as_ref()
65            .map(|path| {
66                fs::read_to_string(path).map_err(|e| {
67                    BootstrapConfigLoadingError::LoadLocalJwks(path.clone(), e.to_string())
68                })
69            })
70            .transpose()?;
71
72        // JWT Config
73        let jwt_config = JwtConfig {
74            jwks,
75            jwt_sig_validation: raw.jwt_sig_validation.into(),
76            jwt_status_validation: raw.jwt_status_validation.into(),
77            signature_algorithms_supported: raw.jwt_signature_algorithms_supported.clone(),
78            token_cache_max_ttl_secs: raw.token_cache_max_ttl,
79            token_cache_capacity: raw.token_cache_capacity,
80            token_cache_earliest_expiration_eviction: raw.token_cache_earliest_expiration_eviction,
81            trusted_issuer_loader: raw
82                .trusted_issuer_loader_type
83                .to_config(raw.trusted_issuer_loader_workers),
84            jwks_refresh_interval: raw.jwks_refresh_interval,
85            jwks_refresh_min_interval: raw.jwks_refresh_min_interval,
86            status_list_refresh_interval_max: raw.status_list_refresh_interval_max,
87        };
88
89        let authorization_config = AuthorizationConfig {
90            decision_log_default_jwt_id: raw.decision_log_default_jwt_id.clone(),
91            strict_schema_validation: raw.strict_schema_validation.into(),
92            custom_token_processor_timeout_millis: raw.custom_token_processor_timeout_millis,
93            metrics_collection: raw.metrics_collection.into(),
94        };
95
96        // Build `DataStoreConfig` from raw config, using defaults if not specified
97        let data_store_config = build_data_store_config(raw);
98
99        let http_client_config = HttpClientConfig {
100            max_retries: raw.http_client_request_max_retries,
101            retry_delay: Duration::from_secs(raw.http_client_request_retry_delay),
102            #[cfg(not(target_arch = "wasm32"))]
103            request_timeout: Duration::from_secs(raw.http_client_request_timeout),
104            // `0` is the documented "no cap" sentinel.
105            max_response_size_bytes: match raw.http_client_max_response_size_bytes {
106                0 => None,
107                n => Some(n),
108            },
109        };
110
111        Ok(Self {
112            application_name: raw.application_name.clone(),
113            log_config,
114            policy_store_config,
115            jwt_config,
116            authorization_config,
117            lock_config,
118            max_default_entities: raw.max_default_entities,
119            max_base64_size: raw.max_base64_size,
120            data_store_config,
121            http_client_config,
122            trace_config,
123        })
124    }
125}

127/// Build [`PolicyStoreConfig`] from the three mutually-exclusive raw source
128/// fields (`CEDARLING_POLICY_STORE_LOCAL`, `_URI`, `_LOCAL_FN`).
129/// Returns an error if none or more than one are set
130fn build_policy_store_config(
131    raw: &BootstrapConfigRaw,
132) -> Result<PolicyStoreConfig, BootstrapConfigLoadingError> {
133    let source = match (
134        raw.local_policy_store.clone(),
135        raw.policy_store_uri.clone(),
136        raw.policy_store_local_fn.clone(),
137        raw.policy_store_cjar_url.clone(),
138    ) {
139        // Case: no policy store provided
140        (None, None, None, None) => return Err(BootstrapConfigLoadingError::MissingPolicyStore),
141        // Case: get the policy store from an inline string (YAML supported for test suites; legacy JSON rejected)
142        (Some(policy_store), None, None, None) => {
143            let trimmed = policy_store.trim();
144            if trimmed.is_empty() {
145                return Err(BootstrapConfigLoadingError::MissingPolicyStore);
146            }
147            if crate::common::policy_store::is_json_content(&policy_store) {
148                return Err(BootstrapConfigLoadingError::LegacyJsonNotSupported);
149            }
150            PolicyStoreSource::Yaml(policy_store)
151        },
152        // Case: get the policy store from a URI
153        (None, Some(policy_store_uri), None, None) => PolicyStoreSource::Uri(policy_store_uri),
154        // Case: get the policy store from a CjarUrl
155        (None, None, None, Some(policy_store_cjar_url)) => {
156            PolicyStoreSource::CjarUrl(policy_store_cjar_url)
157        },
158        // Case: get the policy store from a local file or directory
159        (None, None, Some(raw_path), None) => {
160            let path = Path::new(&raw_path);
161            if path.is_dir() {
162                PolicyStoreSource::Directory(path.into())
163            } else {
164                let file_ext = path
165                    .extension()
166                    .and_then(|ext| ext.to_str())
167                    .map(str::to_lowercase);
168                match file_ext.as_deref() {
169                    Some("json") => {
170                        return Err(BootstrapConfigLoadingError::LegacyJsonNotSupported);
171                    },
172                    Some("yaml" | "yml") => PolicyStoreSource::FileYaml(path.into()),
173                    Some("cjar") => PolicyStoreSource::CjarFile(path.into()),
174                    _ => {
175                        return Err(
176                            BootstrapConfigLoadingError::UnsupportedPolicyStoreFileFormat(raw_path),
177                        );
178                    },
179                }
180            }
181        },
182        // Case: multiple policy stores were set
183        _ => return Err(BootstrapConfigLoadingError::ConflictingPolicyStores),
184    };
185
186    let signature = build_policy_store_signature(raw)?;
187    Ok(PolicyStoreConfig {
188        source,
189        refresh_interval_secs: raw.policy_store_refresh_interval_secs,
190        max_file_size: raw.policy_store_max_file_size,
191        signature,
192    })
193}

195/// Build the Sigstore check; `enforce` needs a signer. The loader checks the source.
196fn build_policy_store_signature(
197    raw: &BootstrapConfigRaw,
198) -> Result<PolicyStoreSignature, BootstrapConfigLoadingError> {
199    if raw.policy_store_sigstore_mode == SigstoreMode::Disabled {
200        return Ok(PolicyStoreSignature::Disabled);
201    }
202    // A blank value would never match a certificate, so treat it as unset.
203    let required = |value: &Option<String>, property| {
204        value.clone().filter(|v| !v.trim().is_empty()).ok_or(
205            BootstrapConfigLoadingError::MissingSigstoreProperty(property),
206        )
207    };
208    let cert_identity = required(
209        &raw.policy_store_sigstore_cert_identity,
210        "CEDARLING_POLICY_STORE_SIGSTORE_CERT_IDENTITY",
211    )?;
212    let cert_oidc_issuer = required(
213        &raw.policy_store_sigstore_cert_oidc_issuer,
214        "CEDARLING_POLICY_STORE_SIGSTORE_CERT_OIDC_ISSUER",
215    )?;
216    Ok(PolicyStoreSignature::Enforce(SigstoreSigner {
217        cert_identity,
218        cert_oidc_issuer,
219        bundle_uri: raw.policy_store_sigstore_bundle_uri.clone(),
220    }))
221}

223/// Build `DataStoreConfig` from raw config fields.
224/// Uses default values for any fields that are not specified.
225fn build_data_store_config(raw: &BootstrapConfigRaw) -> DataStoreConfig {
226    let defaults = DataStoreConfig::default();
227
228    DataStoreConfig {
229        max_entries: raw.data_store_max_entries.unwrap_or(defaults.max_entries),
230        max_entry_size: raw
231            .data_store_max_entry_size
232            .unwrap_or(defaults.max_entry_size),
233        default_ttl: raw
234            .data_store_default_ttl
235            .map(std::time::Duration::from_secs)
236            .or(defaults.default_ttl),
237        max_ttl: raw
238            .data_store_max_ttl
239            .map(std::time::Duration::from_secs)
240            .or(defaults.max_ttl),
241        enable_metrics: raw
242            .data_store_enable_metrics
243            .unwrap_or(defaults.enable_metrics),
244        memory_alert_threshold: raw
245            .data_store_memory_alert_threshold
246            .unwrap_or(defaults.memory_alert_threshold),
247    }
248}

250/// Helper function to resolve log type from raw config
251fn resolve_log_type(
252    raw_config: &BootstrapConfigRaw,
253) -> Result<LogTypeConfig, BootstrapConfigLoadingError> {
254    let log_type_config = match raw_config.log_type {
255        LoggerType::Off => LogTypeConfig::Off,
256        LoggerType::Memory => LogTypeConfig::Memory(MemoryLogConfig {
257            log_ttl: raw_config
258                .log_ttl
259                .ok_or(BootstrapConfigLoadingError::MissingLogTTL)?,
260            max_item_size: raw_config.log_max_item_size,
261            max_items: raw_config.log_max_items,
262        }),
263        LoggerType::StdOut => {
264            let std_out_logger_conf = match raw_config.stdout_mode {
265                // WASM does not support async
266                #[cfg(not(target_arch = "wasm32"))]
267                super::log_config::StdOutMode::Async => StdOutLoggerMode::Async {
268                    timeout_millis: raw_config.stdout_timeout_millis,
269                    buffer_limit: raw_config.stdout_buffer_limit,
270                },
271                super::log_config::StdOutMode::Immediate => StdOutLoggerMode::Immediate,
272            };
273            LogTypeConfig::StdOut(std_out_logger_conf)
274        },
275    };
276    Ok(log_type_config)
277}

297    fn decode(extra: &str) -> BootstrapConfig {
298        let raw: BootstrapConfigRaw = serde_json::from_str(&raw_config_json(extra))
299            .expect("raw bootstrap config should deserialize");
300        BootstrapConfig::try_from(raw).expect("raw config should decode")
301    }

303    #[test]
304    fn archive_cap_does_not_move_the_http_cap() {
305        // The two caps bound different memory (decompressed output vs. buffered
306        // response body), so tuning the archive cap must leave every other
307        // HTTP fetch on its own default.
308        let http_default = Some(HttpClientConfig::DEFAULT_MAX_RESPONSE_SIZE_BYTES);
309
310        let neither = decode("");
311        assert_eq!(
312            neither.http_client_config.max_response_size_bytes, http_default,
313            "With neither property set the HTTP cap must use its own default"
314        );
315        assert_eq!(
316            neither.policy_store_config.max_file_size,
317            ArchiveLimits::DEFAULT_MAX_ENTRY_SIZE,
318            "With neither property set the archive cap must use its own default"
319        );
320
321        let small = decode(r#", "CEDARLING_POLICY_STORE_MAX_FILE_SIZE": 4096"#);
322        assert_eq!(
323            small.http_client_config.max_response_size_bytes, http_default,
324            "A small archive cap must not shrink unrelated HTTP responses"
325        );
326        assert_eq!(
327            small.policy_store_config.max_file_size, 4096,
328            "The archive cap must carry the configured value"
329        );
330
331        let disabled = decode(r#", "CEDARLING_POLICY_STORE_MAX_FILE_SIZE": 0"#);
332        assert_eq!(
333            disabled.http_client_config.max_response_size_bytes, http_default,
334            "Disabling the archive cap must not remove the HTTP cap"
335        );
336    }
