6//! Shared health check registry for cedarling subsystems.
7//!
8//! Subsystems register health check callbacks when they are initialized.
9//! The `HealthTicker` collects statuses from all registered checks on each tick.
10
11use std::collections::HashMap;
12use std::fmt::{Display, Formatter};
13use std::sync::{Arc, RwLock};
14
15use serde::{Deserialize, Serialize};
16
17type HealthCheckFn = Arc<dyn Fn() -> HealthStatus + Send + Sync>;
18
19#[derive(Debug, Copy, Clone, PartialEq, Eq, Deserialize, Serialize)]
20#[serde(rename_all = "lowercase")]
21pub(crate) enum HealthStatus {
22    Success,
23    Failure,
24}
25
26impl Display for HealthStatus {
27    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
28        match self {
29            HealthStatus::Success => write!(f, "success"),
30            HealthStatus::Failure => write!(f, "failure"),
31        }
32    }
33}
34
35/// Overall system health derived from registered checks
36#[derive(Debug, Clone, Copy, PartialEq, Eq)]
37pub(crate) enum SystemHealth {
38    Unknown,
39    Running,
40    Degraded,
41}
42
43impl SystemHealth {
44    pub(crate) fn as_str(self) -> &'static str {
45        match self {
46            Self::Unknown => "unknown",
47            Self::Running => "running",
48            Self::Degraded => "degraded",
49        }
50    }
51}
52
53impl Display for SystemHealth {
54    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
55        f.write_str(self.as_str())
56    }
57}
58
59/// A registered health check with a name and callback.
60struct RegisteredCheck {
61    name: String,
62    check: HealthCheckFn,
63}
64
65/// Thread-safe registry of health check callbacks.
66///
67/// Created during `LockService` initialization, shared with cedarling subsystems
68/// so they can register their health status callbacks.
69#[derive(Clone)]
70pub(crate) struct HealthRegistry {
71    checks: Arc<RwLock<Vec<RegisteredCheck>>>,
72}
73
74impl Default for HealthRegistry {
75    fn default() -> Self {
76        Self::new()
77    }
78}
79
80impl std::fmt::Debug for HealthRegistry {
81    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
82        f.debug_struct("HealthRegistry").finish_non_exhaustive()
83    }
84}
85
86impl HealthRegistry {
87    pub(crate) fn new() -> Self {
88        Self {
89            checks: Arc::new(RwLock::new(Vec::new())),
90        }
91    }
92
93    /// Register a health check callback.
94    ///
95    /// The callback should return [`HealthStatus::Success`] if the subsystem is healthy,
96    /// or [`HealthStatus::Failure`] if not.
97    pub(crate) fn register<F>(&self, name: &str, check: F)
98    where
99        F: Fn() -> HealthStatus + Send + Sync + 'static,
100    {
101        let mut checks = self
102            .checks
103            .write()
104            .expect("health registry write lock poisoned");
105        checks.retain(|c| c.name != name);
106        checks.push(RegisteredCheck {
107            name: name.to_string(),
108            check: Arc::new(check),
109        });
110    }
111
112    /// Collect health statuses from all registered checks.
113    ///
114    /// Returns a map of component name to status.
115    pub(crate) fn collect(&self) -> HashMap<String, HealthStatus> {
116        let checks = self
117            .checks
118            .read()
119            .expect("health registry read lock poisoned");
120        checks
121            .iter()
122            .map(|c| (c.name.clone(), (c.check)()))
123            .collect()
124    }
125
126    /// Compute overall health by reading registered checks directly
127    pub(crate) fn compute_status(&self) -> SystemHealth {
128        let checks = self
129            .checks
130            .read()
131            .expect("health registry read lock poisoned");
132        if checks.is_empty() {
133            SystemHealth::Unknown
134        } else if checks
135            .iter()
136            .all(|c| matches!((c.check)(), HealthStatus::Success))
137        {
138            SystemHealth::Running
139        } else {
140            SystemHealth::Degraded
141        }
142    }
143}

145#[cfg(test)]
146mod test {
147    use super::*;
148
149    #[test]
150    fn test_collect_empty() {
151        let registry = HealthRegistry::new();
152        let result = registry.collect();
153        assert!(result.is_empty(), "expected empty map for empty registry");
154    }
155
156    #[test]
157    fn test_register_and_collect() {
158        let registry = HealthRegistry::new();
159        registry.register("core", || HealthStatus::Success);
160        let result = registry.collect();
161        assert_eq!(result.len(), 1, "expected one entry");
162        assert_eq!(
163            result.get("core"),
164            Some(&HealthStatus::Success),
165            "expected Success status for core"
166        );
167    }
168
169    #[test]
170    fn test_register_multiple_distinct() {
171        let registry = HealthRegistry::new();
172        registry.register("core", || HealthStatus::Success);
173        registry.register("data", || HealthStatus::Success);
174        registry.register("policy_store", || HealthStatus::Failure);
175        let result = registry.collect();
176        assert_eq!(result.len(), 3, "expected three distinct entries");
177        assert_eq!(result.get("core"), Some(&HealthStatus::Success));
178        assert_eq!(result.get("data"), Some(&HealthStatus::Success));
179        assert_eq!(result.get("policy_store"), Some(&HealthStatus::Failure));
180    }
181
182    #[test]
183    fn test_register_same_name_twice() {
184        let registry = HealthRegistry::new();
185        registry.register("core", || HealthStatus::Success);
186        registry.register("core", || HealthStatus::Failure);
187        let result = registry.collect();
188        // the last insertion wins
189        assert_eq!(result.len(), 1, "expected only one entry after dedup");
190        assert_eq!(
191            result.get("core"),
192            Some(&HealthStatus::Failure),
193            "expected the second registration to take precedence"
194        );
195    }
196
197    #[test]
198    fn test_register_same_name_mixed_statuses() {
199        let registry = HealthRegistry::new();
200        registry.register("core", || HealthStatus::Failure);
201        registry.register("data", || HealthStatus::Success);
202        registry.register("core", || HealthStatus::Success);
203        let result = registry.collect();
204
205        assert_eq!(result.len(), 2, "expected two deduplicated entries");
206        assert_eq!(result.get("core"), Some(&HealthStatus::Success));
207        assert_eq!(result.get("data"), Some(&HealthStatus::Success));
208    }
209}
