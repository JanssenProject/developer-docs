32pub(super) struct HealthTicker<T: AuditTransport> {
33    transport: Arc<T>,
34    health_url: Url,
35    pdp_id: PdpID,
36    app_name: Option<ApplicationName>,
37    interval: Duration,
38    logger: Option<LoggerWeak>,
39    registry: HealthRegistry,
40}
41
42impl<T: AuditTransport + 'static> HealthTicker<T> {
43    pub(super) fn spawn(
44        transport: Arc<T>,
45        params: HealthTickerParams,
46    ) -> (CancellationToken, JoinHandle<()>) {
47        let cancel_tkn = CancellationToken::new();
48        let ticker = Self {
49            transport,
50            health_url: params.health_url,
51            pdp_id: params.pdp_id,
52            app_name: params.app_name,
53            interval: params.health_interval,
54            logger: params.logger,
55            registry: params.registry,
56        };
57        let handle = spawn_task(ticker.run(cancel_tkn.clone()));
58        (cancel_tkn, handle)
59    }
60
61    async fn run(self, cancel_tkn: CancellationToken) {
62        let mut ticker = interval(self.interval);
63        ticker.set_missed_tick_behavior(MissedTickBehavior::Delay);
64        ticker.tick().await;
65        loop {
66            tokio::select! {
67                _ = ticker.tick() => {
68                    self.send_health().await;
69                }
70                () = cancel_tkn.cancelled() => {
71                    return;
72                }
73            }
74        }
75    }
76
77    async fn send_health(&self) {
78        let entry = self.build_health_entry();
79        let logger = self.logger.as_ref().and_then(std::sync::Weak::upgrade);
80
81        let item = AuditItem {
82            payload: AuditPayload::Health(Box::new(entry)),
83            pdp_id: self.pdp_id,
84            app_name: self.app_name.clone(),
85            status: None,
86        };
87
88        let result = self
89            .transport
90            .send(&[item], &AuditKind::Health(self.health_url.clone()))
91            .await;
92
93        if let Err(e) = result {
94            logger.log_any(LockLogEntry::error(format!(
95                "failed to send health check to lock server: {e}"
96            )));
97        }
98    }
99
100    fn build_health_entry(&self) -> LockServerHealthEntry {
101        let now = Utc::now().to_rfc3339();
102        let engine_status = self.registry.collect();
103
104        let overall_status = if engine_status.is_empty() {
105            SystemHealth::Unknown
106        } else if engine_status.values().all(|s| *s == HealthStatus::Success) {
107            SystemHealth::Running
108        } else {
109            SystemHealth::Degraded
110        };
111
112        LockServerHealthEntry {
113            creation_date: now.clone(),
114            event_time: now,
115            service: self
116                .app_name
117                .as_ref()
118                .map_or_else(String::new, |n| n.0.to_string()),
119            node_name: self.pdp_id.to_string(),
120            status: overall_status.to_string(),
121            engine_status,
122        }
123    }
124}

126#[cfg(test)]
127mod test {
128    use super::*;
129    use crate::lock::{health_registry::HealthStatus, transport::TransportResult};

133    /// No-op transport that does nothing on send.
134    struct NoopTransport;
135
136    #[async_trait::async_trait]
137    impl AuditTransport for NoopTransport {
138        async fn send(
139            &self,
140            _entries: &[AuditItem],
141            _audit_kind: &AuditKind,
142        ) -> TransportResult<()> {
143            Ok(())
144        }
145    }
146
147    /// Build a minimal [`HealthTicker`] with the given registry and a noop transport.
148    /// The other fields use dummy values since `build_health_entry` ignores them.
149    fn ticker_with_registry(registry: HealthRegistry) -> HealthTicker<NoopTransport> {
150        HealthTicker {
151            transport: Arc::new(NoopTransport),
152            health_url: Url::parse("http://localhost/health").unwrap(),
153            pdp_id: PdpID::new(),
154            app_name: None,
155            interval: Duration::from_secs(10),
156            logger: None,
157            registry,
158        }
159    }
160
161    #[test]
162    fn test_build_health_entry_empty() {
163        let registry = HealthRegistry::new();
164        let ticker = ticker_with_registry(registry);
165        let entry = ticker.build_health_entry();
166        assert_eq!(
167            entry.status, "unknown",
168            "empty registry should report 'unknown'"
169        );
170        assert!(
171            entry.engine_status.is_empty(),
172            "empty registry should have empty engine_status"
173        );
174    }
175
176    #[test]
177    fn test_build_health_entry_all_success() {
178        let registry = HealthRegistry::new();
179        registry.register("core", || HealthStatus::Success);
180        registry.register("data", || HealthStatus::Success);
181        let ticker = ticker_with_registry(registry);
182        let entry = ticker.build_health_entry();
183        assert_eq!(
184            entry.status, "running",
185            "all success should report 'running'"
186        );
187        assert_eq!(entry.engine_status.len(), 2);
188    }
189
190    #[test]
191    fn test_build_health_entry_any_failure() {
192        let registry = HealthRegistry::new();
193        registry.register("core", || HealthStatus::Success);
194        registry.register("data", || HealthStatus::Failure);
195        let ticker = ticker_with_registry(registry);
196        let entry = ticker.build_health_entry();
197        assert_eq!(
198            entry.status, "degraded",
199            "any failure should report 'degraded'"
200        );
201    }

203    #[test]

205        let registry = HealthRegistry::new();

207        registry.register("data", || HealthStatus::Failure);
208        let ticker = ticker_with_registry(registry);

211            entry.status, "degraded",

214    }
215
216    #[test]
217    fn test_build_health_entry_mixed() {
218        let registry = HealthRegistry::new();
219        registry.register("core", || HealthStatus::Success);
220        registry.register("policy_store", || HealthStatus::Success);
221        registry.register("trusted_issuers", || HealthStatus::Failure);
222        registry.register("data", || HealthStatus::Success);
223        let ticker = ticker_with_registry(registry);
224        let entry = ticker.build_health_entry();
225        assert_eq!(
226            entry.status, "degraded",
227            "mixed status with any failure should report 'degraded'"
228        );
229        assert_eq!(entry.engine_status.len(), 4);
230        assert_eq!(
231            entry.engine_status.get("core"),
232            Some(&HealthStatus::Success)
233        );
234        assert_eq!(
235            entry.engine_status.get("trusted_issuers"),
236            Some(&HealthStatus::Failure)
237        );
238    }
