6use std::sync::Arc;
7use std::time::Duration;
8
9use tokio::time::{MissedTickBehavior, interval};
10use tokio_util::sync::CancellationToken;
11
12use crate::authz::metrics::MetricsCollector;
13use crate::http::{JoinHandle, spawn_task};
14use crate::log::interface::LogWriter;
15use crate::log::{BaseLogEntry, LoggerWeak, MetricsLogEntry};
16
17pub(crate) struct TelemetryTicker {
18    metrics: Arc<MetricsCollector>,
19    logger: Option<LoggerWeak>,
20    interval: Duration,
21}
22
23impl TelemetryTicker {
24    pub(crate) fn spawn(
25        metrics: Arc<MetricsCollector>,
26        logger: Option<LoggerWeak>,
27        interval: Duration,
28    ) -> (CancellationToken, JoinHandle<()>) {
29        let cancel_tkn = CancellationToken::new();
30        let ticker = Self {
31            metrics,
32            logger,
33            interval,
34        };
35        let handle = spawn_task(ticker.run(cancel_tkn.clone()));
36        (cancel_tkn, handle)
37    }
38
39    pub(crate) async fn run(self, cancel_tkn: CancellationToken) {
40        let mut ticker = interval(self.interval);
41        ticker.set_missed_tick_behavior(MissedTickBehavior::Delay);
42        ticker.tick().await;
43        loop {
44            tokio::select! {
45                _ = ticker.tick() => {
46                    self.emit_snapshot();
47                }
48                () = cancel_tkn.cancelled() => {
49                    self.emit_snapshot();
50                    return;
51                }
52            }
53        }
54    }
55
56    fn emit_snapshot(&self) {
57        // Attempt upgrade before resetting, so we don't discard data
58        // if the logger is gone.
59        let Some(logger) = self.logger.as_ref().and_then(std::sync::Weak::upgrade) else {
60            return;
61        };
62
63        let snapshot = self.metrics.snapshot_and_reset();
64        let entry = MetricsLogEntry {
65            base: BaseLogEntry::new_metric_opt_request_id(None),
66            policy_stats: snapshot.policy_stats,
67            error_counters: snapshot.error_counters,
68            operational_stats: snapshot.operational_stats,
69            interval_secs: snapshot.interval_secs,
70        };
71
72        logger.log_any(entry);
73    }
74}
